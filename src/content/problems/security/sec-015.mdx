    class Policy:
        """An egress policy for the agent's fetch tool, built once and used for every call.

        schemes: the URL schemes the tool may use, such as {"https"}.
        hosts:   entries of three kinds. An exact name ("docs.python.org"); "*.suffix", which matches any
                 name ending in "." + suffix at any depth but not suffix itself; and "*", which matches any
                 name (an open-web tool whose only protection is the address check). Names compare without
                 case and without a trailing dot. An IP-literal host matches only "*".
                 A policy can hold hundreds of thousands of entries.
        """

        def __init__(self, schemes, hosts):
            raise NotImplementedError


    def fetch(url, policy, resolve, send, max_hops=5):
        """Fetch url for the agent, or refuse. Return ("ok", final_url, body) or ("deny", reason).

        resolve(name) -> list of address strings, as getaddrinfo returns them; raises LookupError for
            a name that does not exist. Treat it as the attacker's DNS server: it can answer differently
            on every call and, like getaddrinfo, it accepts numeric spellings of addresses.
        send(address, host, url) -> (status, headers, body) makes one request to address, with host for
            the Host header and TLS name. Header names arrive lowercase. send never follows redirects.

        The request may go only to a global unicast address: no private, loopback, link-local, shared,
        reserved or multicast address, including one carried inside an IPv6 address. Refuse a URL with
        userinfo or a backslash; clients disagree about where such a URL points. Follow at most
        max_hops redirects (301, 302, 303, 307, 308), checking every hop as if the agent had asked for it.
        """
        raise NotImplementedError

    def _aton(name):
        # getaddrinfo's view of a numeric name: 1 to 4 parts, each decimal, 0x hex or 0-led octal.
        parts = name.split(".")
        if not 1 <= len(parts) <= 4:
            return None
        vals = []
        for p in parts:
            try:
                if p.lower().startswith("0x"):
                    v = int(p[2:] or "0", 16)
                elif len(p) > 1 and p.startswith("0"):
                    v = int(p, 8)
                else:
                    v = int(p, 10)
            except ValueError:
                return None
            vals.append(v)
        *head, last = vals
        if any(v > 255 for v in head) or last >= 256 ** (4 - len(head)):
            return None
        n = 0
        for v in head:
            n = n * 256 + v
        n = (n << (8 * (4 - len(head)))) | last
        return str(_ip.IPv4Address(n))


    def _blocked(text):
        try:
            a = _ip.ip_address(text)
        except ValueError:
            return True
        xs = [a]
        if a.version == 6:
            if getattr(a, "scope_id", None):
                return True
            for e in (a.ipv4_mapped, a.sixtofour):
                if e is not None:
                    xs.append(e)
            if a.teredo:
                xs.extend(a.teredo)
            if a in _ip.ip_network("64:ff9b::/96") or a in _ip.ip_network("::/96"):
                xs.append(_ip.IPv4Address(int(a) & 0xFFFFFFFF))
        return not all(x.is_global and not x.is_multicast for x in xs)


    class _DNS:
        """Like getaddrinfo: a table, then numeric spellings, then IPv6 literals. Records every answer."""

        def __init__(self, table):
            self.table = table
            self.answers = []

        def __call__(self, name):
            n = name.lower().rstrip(".")
            if n in self.table:
                v = self.table[n]
                ans = list(v(len(self.answers)) if callable(v) else v)
            else:
                a = _aton(n)
                if a is None:
                    try:
                        a = str(_ip.ip_address(n))
                    except ValueError:
                        raise LookupError(name)
                ans = [a]
            self.answers.append((n, ans))
            return list(ans)


    class _Net:
        def __init__(self, routes=None):
            self.routes = routes or {}
            self.sent = []

        def __call__(self, address, host, url):
            self.sent.append((address, host, url))
            r = self.routes.get(url)
            return r(len(self.sent)) if callable(r) else (r or (200, {}, b"body"))


    def _safe_sends(net, dns):
        # Every request went to an allowed address that DNS returned for that host, or to the literal in the URL.
        for address, host, url in net.sent:
            if _blocked(address):
                return False
            h = str(host).lower().rstrip(".")
            if not any(n == h and address in ans for n, ans in dns.answers):
                try:
                    if _ip.ip_address(address) != _ip.ip_address(h):
                        return False
                except ValueError:
                    return False
        return True

    import ipaddress
    import re
    from urllib.parse import urljoin, urlsplit

    REDIRECTS = {301, 302, 303, 307, 308}
    _NAT64 = ipaddress.ip_network("64:ff9b::/96")
    _V4COMPAT = ipaddress.ip_network("::/96")
    _NUMERIC_LABEL = re.compile(r"^(0x[0-9a-f]*|[0-9]+)$")
    _NAME = re.compile(r"^[a-z0-9_-]+(\.[a-z0-9_-]+)*$")


    class Denied(Exception):
        pass


    class Policy:
        def __init__(self, schemes, hosts):
            self.schemes = {s.lower() for s in schemes}
            self.any_host = False
            self.exact = set()
            self.suffixes = set()
            # Sets keyed by name, so a lookup costs one probe per label of the host, not one pass over every entry.
            for entry in hosts:
                e = entry.strip().lower().rstrip(".")
                if e == "*":
                    self.any_host = True
                elif e.startswith("*."):
                    self.suffixes.add(e[2:])
                elif e:
                    self.exact.add(e)

        def allows_name(self, host):
            if self.any_host or host in self.exact:
                return True
            i = host.find(".")
            while i != -1:
                if host[i + 1:] in self.suffixes:
                    return True
                i = host.find(".", i + 1)
            return False


    def _embedded_v4(a):
        """IPv4 addresses an IPv6 address can carry to a translator or tunnel."""
        if a.version == 4:
            return []
        out = []
        if a.ipv4_mapped is not None:
            out.append(a.ipv4_mapped)
        if a.sixtofour is not None:
            out.append(a.sixtofour)
        if a.teredo is not None:
            out.extend(a.teredo)
        if a in _NAT64 or a in _V4COMPAT:
            out.append(ipaddress.IPv4Address(int(a) & 0xFFFFFFFF))
        return out


    def permitted(text):
        try:
            a = ipaddress.ip_address(text)
        except ValueError:
            return False
        if getattr(a, "scope_id", None):
            return False
        # is_global alone is not enough: 224.0.0.1, ::127.0.0.1 and 64:ff9b::a9fe:a9fe all report global.
        return all(x.is_global and not x.is_multicast for x in [a] + _embedded_v4(a))


    def _parse(url, policy):
        if not isinstance(url, str) or "\\" in url or any(c in url for c in " \t\r\n"):
            raise Denied("malformed URL")
        parts = urlsplit(url)
        if parts.scheme.lower() not in policy.schemes:
            raise Denied("scheme not allowed: " + parts.scheme)
        if "@" in parts.netloc:
            raise Denied("userinfo in URL")
        host = parts.hostname
        try:
            parts.port
        except ValueError:
            raise Denied("bad port")
        if not host:
            raise Denied("no host")
        host = host.rstrip(".")
        try:
            literal = ipaddress.ip_address(host)
        except ValueError:
            literal = None
        if literal is not None:
            if not policy.any_host:
                raise Denied("IP literal not allowed by this policy")
            return host, literal
        if "%" in host or not _NAME.match(host):
            raise Denied("host is not a plain name")
        if _NUMERIC_LABEL.match(host.rsplit(".", 1)[-1]):
            # 0x7f.1, 2130706433 and 0177.0.0.1 are names to a strict parser and addresses to getaddrinfo.
            raise Denied("non-canonical numeric host")
        if not policy.allows_name(host):
            raise Denied("host not allowed: " + host)
        return host, None


    def _pin(host, literal, resolve):
        if literal is not None:
            if not permitted(str(literal)):
                raise Denied("address not allowed")
            return str(literal)
        try:
            answer = list(resolve(host))
        except Exception:
            raise Denied("does not resolve: " + host)
        if not answer:
            raise Denied("empty answer: " + host)
        # Policy choice: a mixed answer is refused outright rather than filtered, because it is what a rebinding attack looks like.
        if not all(permitted(a) for a in answer):
            raise Denied("answer contains a blocked address")
        return answer[0]


    def fetch(url, policy, resolve, send, max_hops=5):
        current = url
        for _ in range(max_hops + 1):
            try:
                host, literal = _parse(current, policy)
                address = _pin(host, literal, resolve)
            except Denied as e:
                return ("deny", str(e))
            # The connection uses the address that was checked; nothing resolves the name again.
            status, headers, body = send(address, host, current)
            if status not in REDIRECTS:
                return ("ok", current, body)
            location = (headers or {}).get("location")
            if not location:
                return ("deny", "redirect without a location")
            current = urljoin(current, location)
        return ("deny", "too many redirects")

## Question

<Callout kind="interview" label="Coding, 45 minutes">
Your agent has a fetch tool: the model passes a URL, the tool returns the body. The URLs come from the model, which reads pages an attacker can write. Write <code>Policy</code> and <code>fetch</code> to the contract in the starter, so that the tool reaches only hosts the policy allows and never an internal address, however the URL, the DNS answer or a redirect is shaped.
</Callout>

The tool runs on a server inside your cloud network, so it can reach the metadata service at 169.254.169.254, the cluster's internal services and anything on loopback. The MCP specification's security page (revision 2026-07-28) lists, for MCP clients that fetch URLs a server supplies, the same attacks this tool faces: internal IPs written directly, the metadata endpoint, localhost services, DNS rebinding and redirect chains. It also warns against hand-written IP validation, because octal, hex and IPv4-mapped IPv6 spellings slip past custom parsers.

`resolve` stands in for getaddrinfo with an attacker's DNS server behind it: it can answer differently on each call, and it resolves `0x7f.1` the way getaddrinfo does, to 127.0.0.1. `send` makes one request to an address and never follows redirects, so the loop that follows them is yours.

Part of the policy is yours to decide. When one DNS answer holds a public and a private address, you can refuse the request or connect to the public one. When a host is a numeric name that is not a canonical address, you can refuse it before any lookup or let the address check catch what it resolves to. The tests check invariants: every request goes to an address that the policy permits and that DNS returned for that host in that hop, every host sent to is on the allowlist, and a list of internal targets in many spellings is never reached, directly or by redirect.

One constraint is not in the visible tests. In this scenario the allowlist is generated from every tenant's configured integrations and holds 200,000 entries, half of them wildcards, and the tool runs on every fetch of every agent. Checking a host has to cost a handful of operations, not a pass over the list.

## Hints

<details class="hint"><summary>Hint 2</summary><p>Request a list of every address form your HTTP client and the system resolver will accept for 127.0.0.1, and check each against the parser your policy uses.</p></details>
<details class="hint"><summary>Hint 3</summary><p>Ask for the size of the largest tenant allowlist and the fetch rate at peak, then work out what one check costs if it walks every entry.</p></details>

## Answer

The reference solution has four parts: an index built once, a parser that refuses ambiguous URLs, an address rule that looks inside IPv6, and a loop that resolves once per hop and connects to what it checked.

**The index.** `Policy` splits entries into a set of exact names and a set of wildcard suffixes. Checking `a.b.tenant7.example.net` probes the exact set once and the suffix set once per dot: `b.tenant7.example.net`, `tenant7.example.net`, `example.net`, `net`. Starting each candidate after a dot is what keeps `evilgithubusercontent.com` from matching `*.githubusercontent.com`, and starting after the first dot is what keeps the apex out. Walking the list instead costs 200 × 200,000 = 4 × 10<sup>7</sup> comparisons for the hidden test's 200 fetches, which takes seconds in CPython and longer in Pyodide.

**The parser.** A URL with a backslash or userinfo is refused, because clients disagree on where it points: `urlsplit` reads `https://evil.example\@api.github.com/` as a request to api.github.com, and a WHATWG parser treats the backslash as a slash and goes to evil.example. A bad port, a missing host or a scheme off the list is refused rather than raised. An IP literal passes only a `*` policy. A host whose last label is numeric but which `ipaddress` will not parse (`2130706433`, `0x7f.1`, `0177.0.0.1`, `127.1`) is refused before any lookup. Since Python 3.9.5 `ipaddress` rejects leading zeros, so the strict parser calls these names, while getaddrinfo calls them addresses.

**The address rule.** `ipaddress` does most of the work, and its flags alone are not enough. Measured on CPython 3.12.6, `is_global` is true for `224.0.0.1` (multicast), for `::127.0.0.1` (IPv4-compatible) and for `64:ff9b::a9fe:a9fe` (NAT64 to the metadata address). The documentation also records that 3.13 changed how it classifies `2002::/16` and parts of `192.0.0.0/24`, so the answer depends on the Python version. The reference therefore extracts every IPv4 address an IPv6 address can carry and requires each to be global and not multicast:

```python
def permitted(text):
    a = ipaddress.ip_address(text)                      # ValueError: not permitted
    if getattr(a, "scope_id", None):                    # fe80::1%eth0
        return False
    carried = []
    if a.version == 6:
        carried = [x for x in (a.ipv4_mapped, a.sixtofour) if x]
        carried += list(a.teredo or ())
        if a in NAT64 or a in V4COMPAT:                 # 64:ff9b::/96, ::/96
            carried.append(ipaddress.IPv4Address(int(a) & 0xFFFFFFFF))
    return all(x.is_global and not x.is_multicast for x in [a] + carried)
```

**One lookup per hop, and the connection uses its answer.** For each hop the reference parses the URL, resolves the host once, checks every address in the answer, and calls `send` with the first address and the name. The check and the connection see the same answer because there is only one. A rebinding server that answers 93.184.215.14 and then 169.254.169.254 gets one question per hop. A redirect is a new hop and gets the whole check again, so a 302 to `http://169.254.169.254/` fails on the scheme and a 302 to an allowed name that now resolves to loopback fails on the address.

**The decision with two defensible answers: a mixed DNS answer.**

- **Refuse the whole answer** (the reference). A public and a private address for one name is what a rebinding setup looks like, and refusing it gives the attacker nothing to time. The cost is legitimate failures: split-horizon DNS for a name your tenants use internally and externally, or a resolver that returns a private address first for a public service, breaks the fetch, and the agent gets a refusal it cannot fix.
- **Drop the blocked addresses and connect to a permitted one.** Fetches to misconfigured but legitimate hosts keep working. The cost is that the policy now depends on every later step choosing from the filtered list. A client that falls back to the next address after a timeout, or a library that re-resolves on retry, reaches the private address the filter dropped.

The conflict is availability against an attack surface you cannot see. I would refuse, log the name and the answer, and revisit when the logs show which tenants hit it. The redirect limit is the number to defend. The reference uses 5. Every followed hop is a full request to a host the attacker may choose, so a high limit buys nothing for legitimate pages. Below 2, ordinary chains such as a bare domain to its www host and then to a trailing slash start to fail. Anything from 3 to 10 is reasonable; 20 or more is a budget given to the attacker.

**Where this fails.** The gate only holds if every packet passes through it. If the agent can run code in the same network namespace, `curl http://169.254.169.254/` never calls `fetch`; the gate belongs outside the sandbox, as an egress proxy with the agent's network limited to that proxy. Pinning also gives up what the resolver does for availability: connecting to the first address, with no fallback, turns one dead address into a failed fetch. And a policy keyed on names cannot see what a request carries: an allowed host that accepts uploads (a Git host, an object store, a model API) is still a way out for data.

## Follow-ups

- *What changes to run this in a real client?* Open the socket to the pinned address and pass the name separately for the Host header, SNI and certificate verification, so TLS still checks the name. Most HTTP libraries resolve internally, so this means a custom connection class or an egress proxy such as Smokescreen, which the MCP page names. Set connect and read timeouts, cap the response size, and decide whether the pinned address may change on a retry.

## Common mistakes

- Validating the hostname, then letting the HTTP client resolve it again, which is the rebinding window.
- Checking only the first address in the answer.
- Trusting `is_private` or `is_global` alone, and missing multicast, IPv4-compatible and NAT64 forms.
- Following redirects with the client's default, so only the first URL is checked.
- Matching wildcards with `endswith(suffix)`, which lets `evilexample.org` through `*.example.org`.
- Scanning the allowlist on every call; it passes the examples and stalls on a generated list.
