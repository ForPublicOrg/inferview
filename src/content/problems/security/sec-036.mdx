    def scan(data, allowed):
        """Decide whether a model file may be loaded, by reading its pickle opcodes without unpickling.

        data:    the file's bytes. Either raw pickle data, possibly several pickles back to back, or a zip
                 archive in the layout torch.save writes, whose members ending in ".pkl" are pickles.
        allowed: a set of (module, name) pairs the loader may import, such as ("collections", "OrderedDict").

        Return {"ok": bool, "imports": [...], "errors": [...]}:
          imports: (where, offset, module, name) for every import the unpickler would make, in file order;
                   where is "" for raw data or the zip member's name, offset is the opcode's position in it.
          errors:  (where, offset, message) for anything that stops the file from being read with certainty.
          ok:      True only when there are no errors and every import is in allowed.
        """
        raise NotImplementedError

    import io
    import pickletools
    import zipfile

    _MARK = object()
    _STRINGS = {"STRING", "BINSTRING", "SHORT_BINSTRING", "UNICODE", "BINUNICODE", "SHORT_BINUNICODE", "BINUNICODE8"}
    _PUTS = {"PUT", "BINPUT", "LONG_BINPUT"}
    _GETS = {"GET", "BINGET", "LONG_BINGET"}
    _EXTS = {"EXT1", "EXT2", "EXT4"}


    class _Lit(str):
        """A string the stream pushed itself, so STACK_GLOBAL can name it."""


    # (items popped below a MARK, pops down to a MARK?, items pushed) per opcode, from pickletools' own metadata.
    _EFFECT = {}
    for _op in pickletools.opcodes:
        _before = [s.name for s in _op.stack_before]
        if "mark" in _before:
            _EFFECT[_op.name] = (_before.index("mark"), True, len(_op.stack_after))
        else:
            _EFFECT[_op.name] = (len(_before), False, len(_op.stack_after))


    def _scan_stream(f, where, imports, errors):
        """Scan the pickle that starts at f's position. Returns False when the stream breaks."""
        stack, memo = [], {}
        try:
            for op, arg, pos in pickletools.genops(f):
                name = op.name
                if name in _STRINGS:
                    stack.append(_Lit(arg))
                    continue
                if name == "MARK":
                    stack.append(_MARK)
                    continue
                if name == "MEMOIZE" or name in _PUTS:
                    if not stack or stack[-1] is _MARK:
                        raise ValueError(name + " with nothing to store")
                    memo[len(memo) if name == "MEMOIZE" else arg] = stack[-1]
                    continue
                if name in _GETS:
                    if arg not in memo:
                        raise ValueError("memo slot %r read before it was stored" % (arg,))
                    stack.append(memo[arg])
                    continue
                if name == "DUP":
                    if not stack:
                        raise ValueError("DUP on an empty stack")
                    stack.append(stack[-1])
                    continue
                if name == "STACK_GLOBAL":
                    if len(stack) < 2:
                        raise ValueError("STACK_GLOBAL with fewer than two items")
                    qual, module = stack.pop(), stack.pop()
                    if type(module) is _Lit and type(qual) is _Lit:
                        imports.append((where, pos, str(module), str(qual)))
                    else:
                        errors.append((where, pos, "STACK_GLOBAL whose names are not literals in the stream"))
                    stack.append(None)
                    continue
                if name in ("GLOBAL", "INST"):
                    module, _, qual = arg.partition(" ")
                    imports.append((where, pos, module, qual))
                elif name in _EXTS:
                    errors.append((where, pos, "extension code %r resolves in the loading process, not the file" % (arg,)))
                before, to_mark, after = _EFFECT[name]
                if to_mark:
                    k = len(stack) - 1
                    while k >= 0 and stack[k] is not _MARK:
                        k -= 1
                    if k < 0:
                        raise ValueError(name + " without a MARK")
                    del stack[k:]
                if before:
                    if len(stack) < before or any(x is _MARK for x in stack[-before:]):
                        raise ValueError(name + " pops past the stack")
                    del stack[-before:]
                stack.extend([None] * after)
        except Exception as e:  # genops raises on unknown opcodes and truncation; argument decoding can fail too
            errors.append((where, f.tell(), "broken stream: %s" % (e,)))
            return False
        return True


    def _scan_pickles(data, where, imports, errors):
        if not data:
            errors.append((where, 0, "empty: no pickle stream"))
            return
        f = io.BytesIO(data)   # one buffer for the whole file: slicing off each stream would copy the rest every time
        while f.tell() < len(data):
            if not _scan_stream(f, where, imports, errors):
                return


    def scan(data, allowed):
        imports, errors = [], []
        if data[:4] == b"PK\x03\x04":
            try:
                with zipfile.ZipFile(io.BytesIO(data)) as z:
                    for info in z.infolist():
                        if info.filename.endswith(".pkl"):
                            _scan_pickles(z.read(info), info.filename, imports, errors)
            except Exception as e:
                errors.append(("", 0, "unreadable archive: %s" % (e,)))
        else:
            _scan_pickles(data, "", imports, errors)
        blocked = any((m, n) not in allowed for _, _, m, n in imports)
        return {"ok": not errors and not blocked, "imports": imports, "errors": errors}

## Question

<Callout kind="interview" label="Coding, 45 minutes">
Your model registry mirrors files from public hubs and accepts uploads from internal teams. Before any service may load a file, a scanner reads its pickles and decides whether every import is one a weights file needs. Write <code>scan</code> to the contract in the starter: an allowlist, not a denylist, and a file the scanner cannot read with certainty is never ok.
</Callout>

Hugging Face's Hub runs a pickle import scan on every upload with `pickletools.genops` and says plainly that it is not foolproof and that its lists of safe and unsafe imports are best-effort. Picklescan, the open-source scanner, works from a denylist of dangerous functions. In February 2025 ReversingLabs described two models on the Hub whose pickle stream broke shortly after the payload ran; on their broken test file, Picklescan reported the error and not the dangerous import before it.

Part of the policy is yours: which zip members to treat as pickles, what to report when a `STACK_GLOBAL` takes names the stream computed rather than wrote, whether one broken member stops the scan of the others, and how to word an error. The tests check invariants. The imports you report match, in order, the ones a real `Unpickler` asks `find_class` for on random objects at every protocol; removing any reported import from the allowlist makes the file fail; a broken, truncated or empty file never passes; and every offset points at an importing opcode.

One constraint is not in the visible tests. Some files in the registry are dataset shards written with one `pickle.dump` per record: 20,000 pickles back to back, about 11 MB. The scanner runs on every upload and has 3 seconds for such a shard in the browser.

## Answer

The reference solution simulates the unpickler's stack without running anything, reports every import it can name, and turns everything it cannot name into an error.

**Three opcodes import.** `GLOBAL` carries the module and name in its argument, and so does `INST`, the protocol-0 opcode that many scanners skip; the unpickler calls `find_class` for both. `STACK_GLOBAL` pops the name and the module from the stack. Those strings can be pushed anywhere earlier, stored in the memo with `MEMOIZE` or `BINPUT`, and fetched back with `BINGET` just before the import, so looking at the two opcodes before `STACK_GLOBAL` is not enough. The `EXT1`, `EXT2` and `EXT4` opcodes import through the copyreg extension registry of the loading process, which the file cannot show, so they are errors.

**A stack that only knows about strings.** The scanner tracks one fact per stack slot: whether it holds a string the stream pushed itself. Every other opcode's effect comes from `pickletools`' own `stack_before` and `stack_after` metadata, so the simulation covers all opcodes without a table written by hand.

```python
qual, module = stack.pop(), stack.pop()
if type(module) is _Lit and type(qual) is _Lit:     # both pushed as literals, possibly via the memo
    imports.append((where, pos, str(module), str(qual)))
else:                                               # computed names: the scanner cannot say what is imported
    errors.append((where, pos, "STACK_GLOBAL whose names are not literals in the stream"))
```

**Broken streams keep their findings.** `genops` raises on an unknown opcode or a truncated argument, and the stack simulation raises on an underflow or a memo read before a write. The scanner catches both, records the error with its offset, and keeps every import it found before the break. A nullifAI-style file therefore reports `os.system` and an error, and is not ok on either count.

**Every pickle in the file, in one buffer.** After `STOP`, `genops` leaves the file position on the next byte, so a single `io.BytesIO` scans pickle after pickle with offsets that are already positions in the file. A scanner that slices off each pickle and starts again copies the remainder of the file every time. At 20,000 records of about 570 bytes, that copies roughly 20,000 × 11 MB / 2 ≈ 110 GB; the reference scanned the shard in about 0.3 seconds in CPython 3.12, and the slicing version took about 24 seconds on the same machine (measured 2026-10-28).

**Zip archives.** `torch.save` writes an uncompressed ZIP64 archive whose object graph is `data.pkl`; the storages under `data/` are raw bytes. The reference scans every member whose name ends in `.pkl`. A stricter choice scans every member that begins with a pickle protocol byte, at the cost of false errors on raw storages that happen to start with `0x80`.

**The quantity to choose: a size limit.** Torch writes members uncompressed, so a member whose compressed size is smaller than its size did not come from `torch.save` unchanged. Reasonable limits are a cap on uncompressed size at the registry's largest legitimate shard, and refusing compressed `.pkl` members outright. A cap far above the largest model only protects the scanner from a zip bomb late.

**The decision with two defensible answers: how wide the allowlist is.**

- **A short global list** (the torch rebuild functions, storage types, `collections.OrderedDict`) passes plain state dicts and rejects everything else. Each entry is a function the unpickler will call with arguments the attacker chooses, so fewer entries means less code to trust. The cost lands on teams whose checkpoints pickle a config dataclass or an optimizer object: their uploads fail and someone has to convert or re-save them.
- **Per-publisher lists**, extended when a known team's files need a class, pass more files. The cost is that every entry is reviewed as code that runs on untrusted input, and a list that grows by request is a list nobody can audit. CVE-2025-32434 was a bypass of PyTorch's own restricted unpickler: an allowlist written by the framework's maintainers still had a path to code execution.

The conflict is coverage of legitimate files against what the scanner vouches for. The reference keeps the list short and routes rejected files to conversion: a worker with no network and no credentials loads the file once and writes safetensors, and services only ever load the safetensors output.

**Where this fails.** The scanner and the loader are two parsers of the same bytes, and any disagreement between them is an evasion. The nullifAI files were one such disagreement: the loader acted on opcodes that the scanner threw away with the error. Archives are another place to look, since Python's `zipfile` and the loader's own archive reader are separate implementations; an archive the two read differently is rejected, not scanned. And a pass means only that every import is on the list. The allowlisted functions still run with the file's arguments, which is why a pass is a filter in front of a safe format, not a reason to keep loading pickles.

## Follow-ups

- *What changes to run this in the registry's ingestion path?* Stream members from object storage into `genops` instead of reading files into memory, scan with the same zip reader the loaders use or reject archives that two readers parse differently, record the scanner and allowlist versions with each verdict, and rescan the registry whenever the allowlist changes.
- *Your allowlist rejects a fifth of internal uploads because trainers pickle their config dataclass with the weights. Product wants the dataclass added. What do you add, and what risk do you accept in writing?*
- *What would you need to see across the fleet before you retired the scanner and accepted only safetensors?*
- *A researcher sends a file that passes your scanner and still runs code under `torch.load`. Where do you look first: the scanner, the allowlist or the loader?*

## Common mistakes

- Working from a denylist of dangerous names. `os.system`, `posix.system`, `subprocess.Popen`, `builtins.eval` and every function that calls them are an open set.
- Resolving `STACK_GLOBAL` from the two opcodes before it, which misses names fetched from the memo.
- Skipping `INST`, which imports in protocol 0.
- Returning an error for a broken stream but dropping the imports found before the break.
- Allowing a wildcard such as `numpy.*` to match dotted names, which the unpickler resolves by walking attributes under protocol 4 and later.
- Scanning only the first pickle in a file.
