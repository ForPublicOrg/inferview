    import re

    _SLOT = re.compile(r":([a-z_][a-z0-9_]*)|\{([a-z_][a-z0-9_]*)\}")
    _KINDS = {"int", "str", "enum", "prefix", "int_list", "order"}


    def _is_int(v):
        return isinstance(v, int) and not isinstance(v, bool)


    class QueryTool:
        """Decisions: an out-of-range number is refused, not clamped, so the model learns the range instead of getting
        a different answer than it asked for; duplicate ids in a list are dropped, keeping the first occurrence."""

        def __init__(self, templates, execute, max_rows=100):
            self.execute, self.max_rows = execute, max_rows
            self.templates = {name: self._compile(name, t) for name, t in templates.items()}

        @staticmethod
        def _compile(name, t):
            """Split the SQL once into literal text and slots, and refuse templates the tool cannot keep safe."""
            sql, params = t["sql"], t["params"]
            pieces, pos, used = [], 0, set()
            for m in _SLOT.finditer(sql):
                pieces.append(("text", sql[pos:m.start()]))
                if m.group(1):
                    pieces.append(("value", m.group(1)))
                    used.add(m.group(1))
                else:
                    pieces.append(("order", m.group(2)))
                    used.add(m.group(2))
                pos = m.end()
            pieces.append(("text", sql[pos:]))
            if "tenant_id" not in used:
                raise ValueError(f"template {name} does not filter on :tenant_id")
            if "tenant_id" in params:
                raise ValueError(f"template {name} declares tenant_id as a model argument")
            for slot in used - {"tenant_id"}:
                if slot not in params or params[slot][0] not in _KINDS:
                    raise ValueError(f"template {name} uses {slot} without a known type")
            for p, spec in params.items():
                if p not in used:
                    raise ValueError(f"template {name} declares {p} but never uses it")
                if (spec[0] == "order") != (("order", p) in pieces):
                    raise ValueError(f"template {name}: order slots are {{name}}, value slots are :name")
            return pieces, params

        @staticmethod
        def _check(name, spec, v):
            kind = spec[0]
            if kind == "int":
                lo, hi = spec[1], spec[2]
                if not _is_int(v) or not lo <= v <= hi:
                    return f"{name} must be an integer from {lo} to {hi}"
            elif kind in ("str", "prefix"):
                if not isinstance(v, str) or len(v) > spec[1] or "\x00" in v:
                    return f"{name} must be text of at most {spec[1]} characters"
            elif kind == "enum":
                if not isinstance(v, str) or v not in spec[1]:
                    return f"{name} must be one of {', '.join(spec[1])}"
            elif kind == "order":
                if not isinstance(v, str) or v not in spec[1]:
                    return f"{name} must be one of {', '.join(spec[1])}"
            elif kind == "int_list":
                if not isinstance(v, list) or not 1 <= len(v) <= spec[1]:
                    return f"{name} must be a list of 1 to {spec[1]} integers"
                if not all(_is_int(x) for x in v):
                    return f"{name} must contain integers only"
            return None

        def _bind(self, pieces, params, tenant_id, args):
            sql, values = [], []
            for kind, x in pieces:
                if kind == "text":
                    sql.append(x)
                elif kind == "order":
                    sql.append(params[x][1][args[x]])  # the fragment comes from the template author, never the model
                elif x == "tenant_id":
                    sql.append("?")
                    values.append(tenant_id)
                elif params[x][0] == "int_list":
                    ids = list(dict.fromkeys(args[x]))
                    sql.append(", ".join("?" * len(ids)))
                    values.extend(ids)
                elif params[x][0] == "prefix":
                    escaped = args[x].replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")
                    sql.append("?")
                    values.append(escaped + "%")
                else:
                    sql.append("?")
                    values.append(args[x])
            return "".join(sql), values

        def call(self, session, name, args):
            try:
                if name not in self.templates:
                    return {"error": f"unknown query {name!r}; available: {', '.join(sorted(self.templates))}"}
                pieces, params = self.templates[name]
                if not isinstance(args, dict):
                    return {"error": "arguments must be an object"}
                extra = sorted(set(args) - set(params))
                missing = sorted(set(params) - set(args))
                if extra or missing:
                    return {"error": f"unexpected arguments {extra}, missing arguments {missing}"}
                for p, spec in params.items():
                    problem = self._check(p, spec, args[p])
                    if problem:
                        return {"error": problem}
                tenant_id = session["tenant_id"]
                if not _is_int(tenant_id):
                    return {"error": "no tenant for this session"}
                sql, values = self._bind(pieces, params, tenant_id, args)
                cursor = self.execute(sql, values)
                rows = cursor.fetchmany(self.max_rows + 1)
                return {"rows": [list(r) for r in rows[:self.max_rows]], "truncated": len(rows) > self.max_rows}
            except Exception as e:  # the model sees a short error, never a stack trace or the SQL text
                return {"error": f"query failed: {type(e).__name__}"}

## Question

<Callout kind="interview" label="Coding round, 50 minutes">
Our support agent answers questions about a customer's orders by querying the production database. The model chooses the query and its arguments, and the model reads ticket text that customers write. Build the tool so that nothing the model sends can change what SQL runs or whose rows come back.
</Callout>

Write `QueryTool(templates, execute, max_rows=100)` with a method `call(session, name, args)`.

**Templates are written by the team.** Each has `sql` and `params`. In the SQL, `:name` marks a value and `{name}` marks an ORDER BY slot. Each argument's spec is one of:

| spec | the model sends | the query gets |
|---|---|---|
| `("int", lo, hi)` | an integer from lo to hi | that integer |
| `("str", max_len)` | text of at most max_len characters | that text |
| `("enum", values)` | one of values | that string |
| `("prefix", max_len)` | text of at most max_len characters | a pattern matching values that start with exactly that text |
| `("int_list", max_items)` | 1 to max_items integers | one placeholder per integer, for `IN (:name)` |
| `("order", choices)` | a key of choices | the team's SQL fragment for that key, in `{name}` |

**Tenancy.** Every template filters on `:tenant_id`, which is filled from `session["tenant_id"]`, never from the model. The constructor raises `ValueError` for a template the tool cannot keep inside one tenant.

**Calls.** `call` sends one statement to `execute(sql, params)`, using `?` placeholders in order, and returns `{"rows": [[...], ...], "truncated": bool}`, with at most `max_rows` rows. Anything wrong with the call returns `{"error": str}` and runs nothing. The error goes to the model, which reads attacker-written text, so it must not echo SQL or driver messages. `call` never raises.

**Scale.** The orders table holds tens of millions of rows and some templates can match most of a tenant's. The cursor streams rows; the hidden tests count how many you pull.

**Your decisions, to state in a comment:** what an out-of-range number does (refuse or clamp), and what duplicate ids in a list do.

## Answer

The reference splits each template once, at construction, into literal text and slots, and refuses at that point any template that does not use `:tenant_id`, declares `tenant_id` as a model argument, uses a slot with no spec, or declares an argument it never uses. Mistakes in templates are the team's and fail at deploy, not in front of a customer.

**The SQL text depends only on the template, the ORDER BY key and the length of each list.** Values never reach the SQL string; they go into the parameter list. The hidden tests check this as an invariant: for a battery of injection strings the SQL sent to `execute` is byte-identical to the SQL for a harmless value, the number of `?` equals the number of parameters, and the first parameter is always the session's tenant. ORDER BY cannot take a parameter, so the model picks a key and the fragment comes from the template author. A model-supplied `tenant_id` is an error, because silently ignoring it hides an attack from the logs.

**Three type checks that parameterisation does not give you:**

- **Booleans are integers in Python.** `isinstance(True, int)` is true, so a naive check binds `True` as 1. The reference rejects `bool` explicitly, and rejects `"20"` and `20.0`.
- **LIKE wildcards survive binding.** A bound `%` is still a wildcard, so a prefix search for `%` returns every customer in the tenant. The reference escapes the escape character, `%` and `_`, then appends `%`, and the template says `ESCAPE '\'`. The clause matters for portability: PostgreSQL defaults to a backslash escape, while SQLite and the SQL standard have none without it.
- **Lists are validated whole before any placeholder is built.** Empty, too long, strings, booleans and nested lists are refused.

**My decisions.** Out-of-range numbers are refused, not clamped: a clamped `LIMIT` is a silent change to the model's question, and the error tells it what range to use. Duplicate ids are removed with order kept, which changes the placeholder count but not the meaning of `IN`.

**The bounded result.** `cursor.fetchmany(max_rows + 1)` pulls at most one row more than it returns, and the extra row is how it knows to set `truncated`. `fetchall()` followed by a slice pulls everything: on the test's one-million-row cursor that is a million tuples in memory to return a hundred, 99.6 MiB at peak under `tracemalloc`, measured with CPython 3.12.6 on the laptop that checked this problem on 2026-10-28, against 16.4 KiB for the reference on the same cursor.

**The decision with two defensible answers: templates or model-written SQL.**

- **Registered templates** (the reference). Nothing the model sends changes the statement. Cost: the agent answers only questions someone anticipated, and each new one is a code change and a review.
- **Model-written SELECT statements** run as a read-only role, with row-level security keyed on a tenant set per transaction, a statement timeout and a parser that allows one SELECT. Cost: the boundary is now the database's permissions. Every column the role can read is readable, a slow query is one prompt away, and an error in the row-level policy is a cross-tenant leak with no tool in front of it.

The conflict is flexibility against blast radius. For a support agent whose questions are mostly "status of order X", I would ship templates and log refused calls, which tell you which template to write next.

**Where the reference fails.** Tenancy is not authorisation. `customers_by_name` with prefix `""` returns the tenant's customers a hundred at a time, and nothing here asks whether this support agent may see all of them; the user's own entitlements belong in the template's WHERE clause, filled from the session like the tenant. The rows themselves are attacker-written text on the way back into the model's context, so a customer name such as "ignore previous instructions" arrives as a tool result. And the `:name` syntax collides with PostgreSQL's `::` casts: `created_at::date` in a template is refused as an undeclared slot called `date`, so authors write `CAST(created_at AS date)`.

## Follow-ups

- *What changes against a real PostgreSQL driver?* Psycopg's default client-side cursor receives the whole result at `execute`, so `fetchmany` no longer bounds memory: use a server-side cursor or put `LIMIT max_rows + 1` in the statement. Use the driver's own paramstyle, set a statement timeout, run as a read-only role, and set the tenant for row-level security in the same transaction as the query.

## Common mistakes

- Formatting values into the SQL string after "sanitising" them.
- Taking `tenant_id` from the model's arguments when present, or merging session and arguments into one dictionary.
- Using `isinstance(v, int)` and accepting booleans.
- Binding a LIKE prefix without escaping `%` and `_`.
- `fetchall()` then slicing.
- Returning driver errors to the model, which shows it the schema.
