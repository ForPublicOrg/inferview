    import html
    import re
    from urllib.parse import urlsplit

    LINK_SCHEMES = {"http", "https", "mailto"}
    _C0_AND_SPACE = "".join(chr(i) for i in range(0x21))
    _SPECIAL = re.compile(r"[`*\[!]")
    _HEADING = re.compile(r"(#{1,6})[ \t]+(.*)")


    def _esc(text):
        return html.escape(text, quote=True)


    def _as_browser_reads(url):
        """The URL string a browser would act on and its parts, or (None, None) when we refuse to guess."""
        url = url.strip(_C0_AND_SPACE)
        for ch in "\t\n\r":
            url = url.replace(ch, "")
        if "\\" in url:
            return None, None  # a browser reads a backslash as a slash in an http(s) URL; urlsplit does not
        try:
            parts = urlsplit(url)
            parts.port  # raises ValueError on a malformed port
        except ValueError:
            return None, None
        return url, parts


    def _link_href(url):
        url, p = _as_browser_reads(url)
        if p is None or p.scheme.lower() not in LINK_SCHEMES:
            return None
        if p.scheme.lower() != "mailto" and not p.hostname:
            return None
        return url


    def _image_src(url, hosts):
        url, p = _as_browser_reads(url)
        if p is None or p.scheme.lower() != "https":
            return None
        if p.username is not None or p.password is not None or p.port not in (None, 443):
            return None
        return url if p.hostname in hosts else None


    class _Finder:
        """str.find with a memo per needle, so a scan that only moves forward never rescans the same tail."""

        def __init__(self, s):
            self.s, self.memo = s, {}

        def find(self, needle, start):
            hit = self.memo.get(needle)
            if hit is not None:
                searched_from, pos = hit
                if start >= searched_from and (pos == -1 or pos >= start):
                    return pos
            pos = self.s.find(needle, start)
            self.memo[needle] = (start, pos)
            return pos


    def _inline(s, hosts):
        out, f, i, n = [], _Finder(s), 0, len(s)
        while i < n:
            m = _SPECIAL.search(s, i)
            if m is None:
                out.append(_esc(s[i:]))
                break
            if m.start() > i:
                out.append(_esc(s[i:m.start()]))
            i = m.start()
            c = s[i]
            if c == "`":
                end = f.find("`", i + 1)
                if end > i + 1:
                    out.append("<code>" + _esc(s[i + 1:end]) + "</code>")
                    i = end + 1
                    continue
            elif s.startswith("**", i):
                end = f.find("**", i + 2)
                if end > i + 2:
                    out.append("<strong>" + _esc(s[i + 2:end]) + "</strong>")
                    i = end + 2
                    continue
                out.append("**")
                i += 2
                continue
            elif c == "*":
                end = f.find("*", i + 1)
                if end > i + 1:
                    out.append("<em>" + _esc(s[i + 1:end]) + "</em>")
                    i = end + 1
                    continue
            elif c == "[" or s.startswith("![", i):
                image = c == "!"
                open_ = i + 1 if image else i
                close = f.find("]", open_ + 1)
                nested = f.find("[", open_ + 1)
                if close != -1 and (nested == -1 or nested > close) and s.startswith("(", close + 1):
                    end = f.find(")", close + 2)
                    if end != -1:
                        text, url = s[open_ + 1:close], s[close + 2:end]
                        if image:
                            src = _image_src(url, hosts)
                            # An image from any other host becomes its alt text: no request, and no link to click.
                            out.append(f'<img src="{_esc(src)}" alt="{_esc(text)}">' if src else _esc(text))
                        else:
                            href = _link_href(url)
                            # A refused link becomes its text, so a javascript: link reads as words.
                            out.append(f'<a href="{_esc(href)}" rel="nofollow noopener noreferrer">{_esc(text)}</a>' if href else _esc(text))
                        i = end + 1
                        continue
            out.append(_esc(c))
            i += 1
        return "".join(out)


    def render(md, image_hosts):
        hosts = {h.lower() for h in image_hosts}
        blocks, para = [], []

        def flush():
            if para:
                blocks.append("<p>" + _inline("\n".join(para), hosts) + "</p>")
                para.clear()

        for line in md.split("\n"):
            heading = _HEADING.fullmatch(line)
            if not line.strip():
                flush()
            elif heading:
                flush()
                level = len(heading.group(1))
                blocks.append(f"<h{level}>" + _inline(heading.group(2).strip(), hosts) + f"</h{level}>")
            else:
                para.append(line)
        flush()
        return "\n".join(blocks)

## Question

<Callout kind="interview" label="Coding round, 50 minutes">
Our support assistant's replies quote tickets and web pages that anyone can write, and the agent console renders each reply as HTML in a page that holds the agent's session. Write the renderer. It must survive an attacker who controls part of what the model reads.
</Callout>

Write `render(md, image_hosts)`, returning HTML for this Markdown subset:

- **Blocks.** A blank line separates paragraphs, each wrapped in `<p>`; lines inside a paragraph keep their newlines. A line that starts with one to six `#` and a space is a heading, `<h1>` to `<h6>`.
- **Inline.** `` `code` ``, `**strong**`, `*em*`, `[text](url)` and `![alt](url)`. Inline elements do not nest: the text inside them is plain text. Anything that does not form one of these is text.
- **Everything is text unless the subset says otherwise.** HTML in the input, including tags, entities and comments, appears to the reader as the characters typed.
- **Links.** An `<a href>` only when the URL, read the way a browser reads it, is absolute `http`, `https` or `mailto`.
- **Images.** An `<img src alt>` only when the URL, read the way a browser reads it, is `https` and its host is exactly one of `image_hosts`. Any other image must not cause a request.
- The only tags in the output are `p`, `h1` to `h6`, `a`, `img`, `code`, `strong` and `em`, and no attribute can run script.

**Your decisions, to state in a comment:** what a refused link becomes, and what a refused image becomes. Both are tested only for safety.

**Scale.** Replies are streamed and can run to a few hundred kilobytes, and the attacker controls their shape. The hidden tests time adversarial replies of 100,000 to 480,000 characters.

## Answer

The reference has three parts: escaping, a URL policy that reads URLs the way a browser does, and an inline scanner that never rescans the input.

**Escape everything, then emit only what you build.** Every piece of text, and every attribute value, goes through `html.escape(..., quote=True)`, which encodes `&`, `<`, `>`, `"` and `'`. Raw HTML in the reply therefore renders as its characters, and an attribute cannot be closed from inside a URL or an alt text. Escaping `&` is what defeats `jav&#x61;script:`: unescaped, the browser decodes the entity in the attribute and runs the URL; escaped, it reads the literal characters.

**Allowlist schemes on the URL the browser will act on.** Before deciding, the reference strips leading and trailing control characters and spaces and removes tabs and newlines, because the WHATWG URL parser does both, so `java\tscript:` and ` javascript:` are `javascript:` to a browser. It refuses any URL containing a backslash, because a browser treats `\` as `/` in an `http(s)` URL while `urllib.parse.urlsplit` does not: `https://attacker.example\@img.example.com/` has host `img.example.com` to Python and `attacker.example` to Chrome. Then it requires the scheme to be in the allowlist. A blocklist of `javascript:` misses `data:`, `vbscript:` and whatever the next browser accepts; LiteLLM's 2026 credential-exfiltration advisory names the same root cause in a different place, a request denylist that missed some parameters.

**Images get a stricter rule than links,** because an image is fetched with no click and its URL can carry anything the model was told to put there. The host must equal an allowed host exactly, with no user info and no non-default port, so `img.example.com@attacker.example`, `img.example.com.attacker.example` and `//attacker.example` all fail.

**My decisions.** A refused link becomes its text, so a `javascript:` link reads as words. A refused image becomes its alt text, with no link: turning it into a clickable link would hand the exfiltration URL to the reader, one click from firing.

**The decision with two defensible answers: which images may render at all.**

- **An allowlist of hosts you control** (the reference). Charts and screenshots keep working. Cost: every host on the list is now part of the security boundary. EchoLeak got through Microsoft 365 Copilot's image allowlist because `*.teams.microsoft.com` hosted an open redirect, and in ForcedLeak an allowlisted domain had expired and researchers registered it. An allowed host that serves user uploads, or redirects, is an exfiltration endpoint.
- **No images by URL.** The model can only reference attachments by id, which the server resolves. Cost: the assistant cannot show anything it did not attach, and support agents lose inline screenshots from linked tickets.

The conflict is rendering fidelity against exfiltration surface. For an agent console holding sessions and customer data I would ship the allowlist with exactly one host, our own attachment store, with no redirects and no user-chosen paths, and back it with a CSP whose `img-src` lists the same host, so a renderer bug still cannot load from elsewhere.

**Where the reference fails.** Links. A link to any `https` host renders, so an injected "your session expired, sign in again" link with data in the query string works with one click; Slack AI's case on Willison's tag was a link, not an image, and Slack called it intended behaviour. Showing the destination host beside every link, or confirming before leaving the console, costs the agents a click. And the parser is only safe if it is the only parser: if the client later runs the same reply through a full CommonMark library, reference-style syntax such as `![a][r]` with `[r]: https://...` comes back to life, which is exactly how EchoLeak escaped Copilot's filter.

**The performance constraint.** A regex renderer with `\[(.*?)\]\((.*?)\)` restarts its lazy scan at every `[`, so a reply of unclosed brackets is quadratic: measured with CPython 3.12.6 on the Windows laptop that checked this problem on 2026-10-28, 40,000 brackets took 11.3 s and 100,000 took 74.2 s, against 0.21 s for the reference on the same 100,000. Two and a half times the input took about six and a half times as long, which is the quadratic showing. The reference scans left to right and memoises each `str.find` per needle: once it knows the next `]` is at position j, or that there is none, every later start before j reuses the answer. Each character is examined a constant number of times.

## Follow-ups

- *What changes to run this in the real console?* A maintained CommonMark parser with raw HTML disabled, this URL policy in its link and image hooks, an HTML sanitiser on the output, the CSP `img-src` matching the allowlist, and one renderer shared by the streaming and final views. GitLab Duo's filter was bypassed through streaming, so a partial reply has to go through the same path as a complete one.
- *Product wants images from any `https` host back, because customers paste screenshot links. Which half of the risk do you accept in writing, and what do you add so you can see it being used?*

## Common mistakes

- Blocklisting `javascript:` instead of allowlisting schemes.
- Checking the URL with `urlsplit` alone, which disagrees with the browser on backslashes.
- Escaping text but writing URLs into `href` unescaped, which lets entities and quotes through.
- Turning a refused image into a link, which keeps the exfiltration one click away.
- Using lazy-quantifier regexes on attacker-shaped input.
