    from collections import OrderedDict


    class KVCache:
        def __init__(self, num_blocks, block_size=16):
            self.num_blocks = num_blocks
            self.block_size = block_size
            self.empty = list(range(num_blocks - 1, -1, -1))   # blocks holding nothing reusable
            self.evictable = OrderedDict()                      # cached full blocks nobody holds, least recently freed first
            self.cached = {}                                    # block hash -> block id
            self.bhash = [None] * num_blocks                    # hash of a full block's tokens and its whole prefix
            self.refs = [0] * num_blocks
            self.data = [[] for _ in range(num_blocks)]
            self.tables = {}
            self.lengths = {}
            self.order = {}                                     # running requests, oldest first

        def num_free(self):
            return len(self.empty) + len(self.evictable)

        def block_tokens(self, b):
            return list(self.data[b])

        def block_table(self, rid):
            return list(self.tables[rid])

        def length(self, rid):
            return self.lengths[rid]

        def _take(self):
            if self.empty:
                b = self.empty.pop()
            else:
                b, _ = self.evictable.popitem(last=False)       # evict only when no empty block is left
                del self.cached[self.bhash[b]]
                self.bhash[b] = None
            self.refs[b] = 1
            self.data[b] = []
            return b

        def _hold(self, b):
            if self.refs[b] == 0:
                del self.evictable[b]
            self.refs[b] += 1

        def _drop(self, b):
            self.refs[b] -= 1
            if self.refs[b] == 0:
                h = self.bhash[b]
                if h is not None and self.cached.get(h) == b:
                    self.evictable[b] = None
                else:
                    self.bhash[b] = None
                    self.empty.append(b)

        def _seal(self, table, i):
            """Block i of a table just became full: hash it with its parent and offer it to the prefix cache."""
            b = table[i]
            parent = self.bhash[table[i - 1]] if i else None
            h = hash((parent, tuple(self.data[b])))
            self.bhash[b] = h
            self.cached.setdefault(h, b)

        def add(self, rid, tokens):
            if rid in self.tables:
                raise ValueError(f"{rid} already present")
            bs = self.block_size
            hits, parent = [], None
            for i in range((len(tokens) - 1) // bs):            # never the whole prompt: one token must be computed
                chunk = tokens[i * bs:(i + 1) * bs]
                h = hash((parent, tuple(chunk)))
                b = self.cached.get(h)
                if b is None or self.data[b] != chunk:
                    break
                hits.append(b)
                parent = h
            need = -(-len(tokens) // bs) - len(hits)
            if need > self.num_free() - sum(1 for b in hits if self.refs[b] == 0):
                raise MemoryError("not enough free blocks")
            for b in hits:
                self._hold(b)
            table = list(hits)
            for i in range(len(hits), -(-len(tokens) // bs)):
                b = self._take()
                self.data[b] = list(tokens[i * bs:(i + 1) * bs])
                table.append(b)
                if len(self.data[b]) == bs:
                    self._seal(table, i)
            self.tables[rid] = table
            self.lengths[rid] = len(tokens)
            self.order[rid] = None
            return len(hits) * bs

        def append(self, rid, token):
            table, n, bs = self.tables[rid], self.lengths[rid], self.block_size
            if n % bs == 0:
                if not self.num_free():
                    raise MemoryError("no free block for the next token")
                table.append(self._take())
            elif self.refs[table[-1]] > 1:
                if not self.num_free():
                    raise MemoryError("no free block for copy on write")
                new = self._take()
                self.data[new] = list(self.data[table[-1]])
                self._drop(table[-1])
                table[-1] = new
            self.data[table[-1]].append(token)
            self.lengths[rid] = n + 1
            if len(self.data[table[-1]]) == bs:
                self._seal(table, len(table) - 1)

        def fork(self, parent, child):
            if child in self.tables:
                raise ValueError(f"{child} already present")
            for b in self.tables[parent]:
                self._hold(b)
            self.tables[child] = list(self.tables[parent])
            self.lengths[child] = self.lengths[parent]
            self.order[child] = None

        def free(self, rid):
            for b in reversed(self.tables.pop(rid)):            # the tail is least likely to be shared, so it goes first
                self._drop(b)
            del self.lengths[rid]
            del self.order[rid]

        def tokens(self, rid):
            return [t for b in self.tables[rid] for t in self.data[b]]

        def _released_if_freed(self, victims):
            drop = {}
            for rid in victims:
                for b in self.tables[rid]:
                    drop[b] = drop.get(b, 0) + 1
            return sum(1 for b, c in drop.items() if self.refs[b] == c)

        def preempt_for(self, n, protect=()):
            """Free running requests until n blocks are available. Victims: the most recently added first."""
            if self.num_free() >= n:
                return {}
            candidates = [r for r in reversed(self.order) if r not in protect]
            if self.num_free() + self._released_if_freed(candidates) < n:
                raise MemoryError("even preempting every other request would not free enough blocks")
            out = {}
            for rid in candidates:
                out[rid] = self.tokens(rid)
                self.free(rid)
                if self.num_free() >= n:
                    break
            return out

## Question

Write the KV block manager of a paged inference engine. GPU memory for keys and values is a pool of `num_blocks` blocks of `block_size` tokens. Each running request has a block table, the physical blocks holding its tokens in order. No tensors move here: `block_tokens(b)` returns the tokens stored in block `b` and stands in for its keys and values, and the tests rebuild every request's sequence from its table to check that nothing was overwritten.

The rules:

- `add(rid, tokens)` admits a request. Full blocks of its prompt that are already cached, whether a running request holds them or they were freed earlier, are reused rather than recomputed. A cached block matches only if every token before it matches too. At least one prompt token is always computed, because the engine needs logits for the next token. It returns the number of prompt tokens served from the cache. All or nothing: `MemoryError` if the prompt does not fit.
- `append(rid, token)` writes one generated token. A request never writes into a block another request also holds.
- `fork(parent, child)` starts a second sample sharing every block of the parent, copying nothing.
- `free(rid)` releases a request. Its full blocks stay cached until the memory is needed. `num_free()` counts them as free, and a cached block is evicted only when no empty block is left.
- `preempt_for(n, protect)` frees running requests until `n` blocks are free, never one named in `protect`, and returns each victim's tokens so the scheduler can recompute it. Preempt no more requests than needed. If `n` cannot be reached, raise `MemoryError` and preempt nobody.
- Any `MemoryError` leaves the state exactly as it was.

Two policies are yours: which cached block to evict, and which request to preempt. Say why you chose each.

The pool can be large. Admitting 100 prompts of 16,384 tokens that share an 8,192-token prefix into a 70,000-block pool, freeing them and admitting them again has to finish in under 3 seconds, including in the browser. Standard library only.

## Answer

The reference solution keeps, per physical block, a reference count, its tokens and, once full, a hash. Free blocks sit in two places: a stack of empty blocks, and an ordered map of cached blocks nobody holds, least recently freed first. A dictionary maps each hash to the one block registered under it.

**The hash.** A full block's key is `hash((parent_key, tuple(tokens)))`, with the parent block's key standing in for the whole prefix. This is the vLLM design, and it is what the performance test checks. A key built from the whole prefix tuple costs time proportional to the prefix for every block, so a 16,384-token prompt at block size 16 hashes about 8.4 million token positions (16 × (1 + 2 + ... + 1,024)) instead of 16,384. On a laptop the reference solution ran the timed test in under 0.1 seconds; the same code with whole-prefix keys took 14 seconds. On a hit the solution also compares the block's stored tokens with the prompt chunk, so a hash collision on that block costs a miss rather than wrong keys and values. Python's hash of a tuple of integers is not salted per process, so these keys are reproducible, but they are 64-bit and not meant to cross a process boundary.

**Admission.** `add` walks the prompt one full block at a time and stops at the first miss. It never matches the last block when the prompt is an exact multiple of the block size, so at least one token is computed. It counts the new blocks it needs against `num_free()`, minus the hits that are themselves sitting in the cached pool, before touching anything; that check is what makes it all or nothing. New full blocks are hashed and registered. If another block already holds that key, the new one stays unregistered and goes back to the empty stack when freed.

**Writes.** `append` takes a new block when the length is a multiple of the block size. Otherwise it writes into the last block, copying it first if its reference count is above one. That only happens after a fork: cached blocks are full, and full blocks are never written. A block that fills is hashed then, so generated tokens become cacheable too.

**Eviction.** Empty blocks go out first. A cached block is evicted only when none is left, least recently freed first. `free` releases a table from the tail, so a request's last blocks, which hash the most tokens and are the least likely to be shared, are evicted first. That is also the order vLLM uses. The alternative is to evict by how often a block was hit. It keeps a hot system prompt resident through a burst of one-off documents, but costs a counter per block and a heap instead of an ordered map. I would switch only if the hit-rate metric showed the system prompt being evicted.

**The preemption decision.** Two orders are defensible, and each has a cost:

- *The most recently admitted request first*, as vLLM's FCFS policy does. The oldest requests keep their progress, so every request finishes eventually, and the victim usually has the least computed. The cost: under sustained pressure the newest request is preempted again every time it comes back, and its time to first token has no bound.
- *The request whose release frees the most blocks*. Fewest preemptions per event. The cost: it picks long requests, which have the most to recompute because prefill is paid again in full, and a request near its last token can lose all of it.

The solution takes the first. It checks that freeing every unprotected request would reach `n`, so a failure preempts nobody, then frees in that order and stops as soon as `n` blocks are free. The returned tokens are the prompt plus everything generated, so the scheduler re-admits the victim as a longer prompt. Because a victim's full blocks stay cached, its recompute often hits most of them, which is part of why vLLM V1 defaults to recompute rather than swap.

**Where this design fails.** The prefix cache is shared by every request in the pool. In a multi-tenant engine, one tenant's request can hit blocks another tenant's prompt produced: a measurable timing difference and a data-boundary question. vLLM's answer is a cache salt mixed into the first block's hash, so only requests with the same salt share. The cost is the hit rate on prefixes common across tenants. The bookkeeping above does not change; the key does.

## Follow-ups

- *Security asks for per-tenant salts, and your measurements say that halves the hit rate on the shared system prompt. Which do you ship, and what risk do you accept in writing?*
- *What would you need to see in production to move from recompute to swap?* Preempted requests with long contexts whose recompute shows up in time to first token, with host memory bandwidth to spare. The PagedAttention paper found swapping more expensive than recompute at small block sizes.
- *What changes to run this inside a real engine?* The copy in copy on write becomes a block copy on the GPU, batched for the step. Block tables become a tensor the attention kernel reads. The hash has to be stable across processes if prefix caching spans workers or nodes. The scheduler owns this object from one thread and calls it once per step.

## Common mistakes

- Keying a block by the whole prefix, which is correct and quadratic in the prompt length.
- Matching a block on its own tokens alone, so two prompts with the same chunk after different prefixes share keys and values that are wrong for one of them.
- Serving an entire prompt from the cache, which leaves the engine with no logits for the first generated token.
- Returning a shared block to the free list when its first holder frees it.
- Checking for room after taking some blocks, so a failed `add` or `append` leaks them.
- Evicting cached blocks while empty ones remain, which throws away hits for nothing.
