    import time


    class StoreDown(Exception):
        """Raised by the store when it cannot be reached."""


    class Store:
        """In-memory stand-in for the shared store (Redis, or a database with conditional writes).
        get(key) returns (value, version); an absent key is (None, 0).
        cas(key, version, value) writes only if the key is still at that version, and returns whether it wrote.
        Every call is one round trip and is counted in self.ops. While self.down is True every call raises StoreDown."""

        def __init__(self):
            self.data, self.ops, self.down = {}, 0, False

        def get(self, key):
            self.ops += 1
            if self.down:
                raise StoreDown(key)
            return self.data.get(key, (None, 0))

        def cas(self, key, version, value):
            self.ops += 1
            if self.down:
                raise StoreDown(key)
            if self.data.get(key, (None, 0))[1] != version:
                return False
            self.data[key] = (value, version + 1)
            return True


    class TokenLimiter:
        """One gateway replica's view of a tenant's token bucket. Every replica serving the tenant builds its own
        TokenLimiter on the same store; replicas share nothing else."""

        def __init__(self, store, tenant, tokens_per_minute, burst=None, replicas=1, clock=time.monotonic):
            raise NotImplementedError

        def available(self):
            """The tenant's shared level after refill, or None while the store is down."""
            raise NotImplementedError

        def try_reserve(self, prompt_tokens, max_tokens):
            """Return the number of tokens reserved, or None to reject (the gateway answers 429)."""
            raise NotImplementedError

        def settle(self, reserved, used_tokens):
            """After the call: replace the reservation with what the provider reported."""
            raise NotImplementedError

    import time


    class StoreDown(Exception):
        """Raised by the store when it cannot be reached."""


    class Store:
        """In-memory stand-in for the shared store (Redis, or a database with conditional writes).
        get(key) returns (value, version); an absent key is (None, 0).
        cas(key, version, value) writes only if the key is still at that version, and returns whether it wrote.
        Every call is one round trip and is counted in self.ops. While self.down is True every call raises StoreDown."""

        def __init__(self):
            self.data, self.ops, self.down = {}, 0, False

        def get(self, key):
            self.ops += 1
            if self.down:
                raise StoreDown(key)
            return self.data.get(key, (None, 0))

        def cas(self, key, version, value):
            self.ops += 1
            if self.down:
                raise StoreDown(key)
            if self.data.get(key, (None, 0))[1] != version:
                return False
            self.data[key] = (value, version + 1)
            return True


    class TokenLimiter:
        MAX_ATTEMPTS = 8

        def __init__(self, store, tenant, tokens_per_minute, burst=None, replicas=1, clock=time.monotonic):
            self.store, self.key, self.clock = store, "tb:" + str(tenant), clock
            self.rate = tokens_per_minute / 60.0
            self.capacity = float(burst if burst is not None else tokens_per_minute)
            self.replicas = replicas
            self.pending = 0.0                 # change owed to the shared level: refunds positive, debts and outage spend negative
            self.seen = self.capacity          # last shared level this replica read
            self.local = None                  # [level, time] of this replica's share while the store is down

        def _level(self, value, now):
            if value is None:
                return self.capacity
            level, last = value
            return min(self.capacity, level + max(0.0, now - last) * self.rate)

        def _apply(self, decide):
            """Read, decide, conditional write; retry on a lost race. decide(level) -> (charge, result).
            Returns (written, result). Raises StoreDown."""
            for _ in range(self.MAX_ATTEMPTS):
                value, version = self.store.get(self.key)
                now = self.clock()
                level = min(self.capacity, self._level(value, now) + self.pending)
                charge, result = decide(level)
                if charge == 0 and self.pending == 0:
                    self.seen = level
                    return True, result
                if self.store.cas(self.key, version, (level - charge, now)):
                    self.pending, self.local, self.seen = 0.0, None, level - charge
                    return True, result
            return False, None

        def _share(self):
            now = self.clock()
            cap = self.capacity / self.replicas
            if self.local is None:
                self.local = [max(0.0, self.seen) / self.replicas, now]
            level, last = self.local
            self.local = [min(cap, level + (now - last) * self.rate / self.replicas), now]
            return self.local

        def available(self):
            try:
                return self._apply(lambda level: (0, level))[1]
            except StoreDown:
                return None

        def try_reserve(self, prompt_tokens, max_tokens):
            need = prompt_tokens + max_tokens
            if need > self.capacity:
                return None
            try:
                written, result = self._apply(lambda level: (need, need) if need <= level else (0, None))
                return result if written else None
            except StoreDown:
                share = self._share()
                if need > share[0]:
                    return None
                share[0] -= need
                self.pending -= need
                return need

        def settle(self, reserved, used_tokens):
            delta = reserved - used_tokens
            self.pending += delta
            if self.local is not None:
                self.local[0] = min(self.capacity / self.replicas, self.local[0] + delta)
            try:
                self._apply(lambda level: (0, None))
            except StoreDown:
                pass

## Question

<Callout kind="interview" label="Coding, 40 minutes">
"Our gateway runs several replicas. Write the per-tenant token limiter they share, and tell me what it does when its store is down."
</Callout>

Each tenant has a token bucket: `tokens_per_minute` refilled continuously, up to `burst` (default one minute's worth), starting full. Every gateway replica that serves the tenant builds its own `TokenLimiter` over the same `Store`, and the replicas share nothing else: no messages, no leader. The store offers only `get` and a conditional write, `cas`, the way a database with versioned rows or Redis under `WATCH` does. The clock is injected and is the same for every replica in the tests.

- `try_reserve(prompt_tokens, max_tokens)` returns the number of tokens it reserved, or `None` to reject. You choose the reservation size, between the prompt alone and the prompt plus `max_tokens`. A rejected request takes nothing.
- `settle(reserved, used_tokens)` replaces the reservation with what the provider reported. Usage above the reservation is charged even if it takes the level below zero.
- `available()` returns the shared level after refill, or `None` while the store is down.
- Two replicas can read the same version and both try to write. Neither may lose the other's charge.
- When the store raises `StoreDown`, the gateway still needs an answer for every request; no exception reaches the caller. Across all replicas, the tenant must not get more than its limit would have allowed, and usage during the outage is still owed when the store comes back. How a replica decides during the outage is yours to choose and defend; `replicas` tells you how many share the tenant.
- When no other replica is writing, `try_reserve` and `settle` each make at most two round trips to the store (every `get` and `cas` counts), and the state stored per tenant stays the same size however many requests the tenant has made.

## Answer

**State.** The store holds one value per tenant, `(level, time of last write)`. Refill is computed on read: `min(burst, level + elapsed × rate)`. Nothing runs in the background, and the value holds two numbers whatever the traffic, which is what the size test checks. A sliding log of recent requests answers the same question with state that grows with the request rate, and it fails that test.

**Atomic update without a lock.** Every change is read, decide, conditional write. `_apply` gets `(value, version)`, computes the refilled level, decides the charge, and writes with `cas(key, version, ...)`. If another replica wrote first, the version moved, the write is refused, and the loop starts again from the new level. A rejection that changes nothing skips the write, so it costs one round trip; an admitted request costs two. After eight lost races in a row the reference rejects the request rather than spinning, which is a choice: a very hot tenant sees some 429s under contention instead of unbounded latency. The visible race test stages the case a read-then-write limiter gets wrong: replica B reserves 3,000 between replica A's read and A's write, and A's write must not erase B's charge.

**How much to reserve.** The reference reserves the worst case, prompt plus `max_tokens`, as LiteLLM's budget reservation does for cost. Anthropic's own limiter does something different: it estimates input at the start and counts output only as it is generated, and `max_tokens` does not count at all. Both are defensible:

- *Worst case.* Requests in flight can never overdraw the bucket. The cost is admission: a tenant that sends 2,000-token prompts with `max_tokens` 4,096 and gets 300 tokens back holds 6,096 per request, so a 60,000 bucket admits 60,000 / 6,096 = 9 requests in flight where real usage of 2,300 each would allow 26.
- *Prompt only, or prompt plus an estimate such as the tenant's recent p95 output.* More requests are admitted, and settle charges the overshoot afterwards, so the level can go negative and the tenant waits. The cost: a burst of long answers overdraws the bucket before any limit applies, which matters when the bucket protects a provider quota rather than a fairness target.

Anything between the two is reasonable if settle charges the difference. Reserving less than the prompt is not: the prompt is known on arrival.

**Settle.** One delta, `reserved − used`, added to the level and capped at the burst. Usage above the reservation drives the level negative and the tenant pays it back through refill; clamping at zero would forgive tokens the provider billed.

**The store is down.** Two choices pass the tests, and they serve different owners.

- *Fail closed.* Reject everything until the store answers. The limit holds exactly; every tenant is down for the length of the outage, so the limiter's store becomes part of the product's availability. LiteLLM's fail-closed mode answers 503, not 429, so clients and load balancers can tell an outage from a limit.
- *A local share.* Each replica keeps admitting from its own bucket of `last seen level / replicas`, refilled at `rate / replicas`, and records what it admitted as a debt. When the store answers, the first successful write folds the debt, and any settlements made during the outage, into the shared level. Summed over replicas, admission during the outage is at most what the shared bucket would have allowed. LiteLLM's default is the risky version of this: each instance falls back to in-memory counters, and its docs say limits can then be exceeded across instances.

The reference takes the local share, because this bucket protects capacity and fairness, and an outage of a counter store should not be an outage of every tenant. For a dollar budget on a prepaid account the same reasoning points the other way.

**Where the local share fails.** It assumes load is spread evenly. A tenant whose traffic arrives over one long-lived connection lands on one replica and gets a quarter of its limit for the length of an outage on four replicas, while the other three shares go unused. It also trusts each replica's clock for refill: a replica whose clock runs ahead refills its share faster, and the level it writes after recovery is stamped with a time the others have not reached. The reference uses the injected clock everywhere; a real deployment takes the store's clock.

## Follow-ups

- *What changes to run this against Redis?* The read-decide-write loop becomes one Lua script run by `EVAL`, which is atomic on the server, so an admitted request is one round trip instead of two and there are no lost races to retry. Take the time from Redis `TIME` inside the script, not from the replica. Store reservations with an expiry so a replica that dies between reserve and settle does not hold tokens forever.
- *Pushback: finance says the local share lets a tenant overspend during an outage and wants fail closed; the product owner says a counter-store outage must not take every customer down. What do you ship, and what do you write down?*
- *Reversal: what would you need to see to switch from worst-case reservations to prompt-only reservations?* The distribution of reported output against `max_tokens` per tenant, and how often a burst of long outputs would have overdrawn the bucket by more than the provider's own limit allows.
- *How should a 429 compute `retry-after` when the level is negative?* The debt plus the next request's reservation, divided by the refill rate, rounded up.

## Common mistakes

- Reading the level, deciding, and writing unconditionally, so two replicas that read the same version both admit against it.
- Raising the store's exception to the caller, which turns a store outage into 500s with no decision.
- Falling back to the full limit on every replica during an outage, which multiplies the limit by the replica count.
- Dropping debts and refunds recorded while the store was down when it returns.
- Keeping a log of recent requests in the store, whose size grows with traffic and whose every update rewrites it.
- Clamping at zero in settle, which forgives tokens the provider billed.
