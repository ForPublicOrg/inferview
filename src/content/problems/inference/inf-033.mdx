    import heapq
    import itertools

    INF = float("inf")


    class _Grant:
        __slots__ = ("remaining", "expires_at", "seq", "queued")

        def __init__(self, remaining, expires_at, seq):
            self.remaining = remaining
            self.expires_at = expires_at
            self.seq = seq
            self.queued = True


    class _Account:
        __slots__ = ("balance", "held", "queue", "holds", "hold_expiries", "log")

        def __init__(self):
            self.balance = 0          # live credits, including held ones
            self.held = 0             # credits reserved by open holds
            self.queue = []           # heap of (expires_at, seq, grant): spend order and expiry order are the same
            self.holds = {}           # hold id -> [amount, deadline, open]
            self.hold_expiries = []   # heap of (deadline, hold id)
            self.log = []


    class Ledger:
        """Credits per account: grants with optional expiry, spent soonest-expiring first."""

        def __init__(self):
            self.accounts = {}
            self.last_at = None
            self.ids = itertools.count(1)
            self.charges = {}         # charge id -> [account, [(grant, units), ...] in spend order, refundable units]
            self.refund_keys = {}
            self.hold_owner = {}

        def _clock(self, account, at):
            if self.last_at is not None and at < self.last_at:
                raise ValueError("time went backwards")
            self.last_at = at
            acct = self.accounts.get(account)
            if acct is None:
                acct = self.accounts[account] = _Account()
            queue = acct.queue
            while queue and queue[0][0] <= at:
                expires_at, _, grant = heapq.heappop(queue)
                grant.queued = False
                if grant.remaining:
                    acct.balance -= grant.remaining
                    acct.log.append({"at": expires_at, "kind": "expire", "delta": -grant.remaining})
                    grant.remaining = 0
            heap = acct.hold_expiries
            while heap and heap[0][0] <= at:
                _, hold_id = heapq.heappop(heap)
                hold = acct.holds.get(hold_id)
                if hold is not None and hold[2]:
                    hold[2] = False
                    acct.held -= hold[0]
            return acct

        @staticmethod
        def _positive(amount):
            if not isinstance(amount, int) or isinstance(amount, bool) or amount <= 0:
                raise ValueError("amounts are positive integers")

        @staticmethod
        def _available(acct):
            return max(0, acct.balance - acct.held)

        def _spend(self, acct, amount):
            parts = []
            queue = acct.queue
            while amount:
                expires_at, seq, grant = queue[0]
                take = min(amount, grant.remaining)
                grant.remaining -= take
                amount -= take
                parts.append((grant, take))
                if not grant.remaining:
                    heapq.heappop(queue)
                    grant.queued = False
            return parts

        def grant(self, account, amount, at, expires_at=None):
            self._positive(amount)
            acct = self._clock(account, at)
            if expires_at is not None and expires_at <= at:
                raise ValueError("expires_at must be after at")
            g = _Grant(amount, INF if expires_at is None else expires_at, next(self.ids))
            heapq.heappush(acct.queue, (g.expires_at, g.seq, g))
            acct.balance += amount
            acct.log.append({"at": at, "kind": "grant", "delta": amount})
            return g.seq

        def balance(self, account, at):
            return self._clock(account, at).balance

        def available(self, account, at):
            return self._available(self._clock(account, at))

        def consume(self, account, amount, at):
            self._positive(amount)
            acct = self._clock(account, at)
            if amount > self._available(acct):
                return None
            charge_id = next(self.ids)
            self.charges[charge_id] = [account, self._spend(acct, amount), amount]
            acct.balance -= amount
            acct.log.append({"at": at, "kind": "consume", "delta": -amount, "charge": charge_id})
            return charge_id

        def hold(self, account, amount, at, ttl):
            self._positive(amount)
            if ttl <= 0:
                raise ValueError("ttl must be positive")
            acct = self._clock(account, at)
            if amount > self._available(acct):
                return None
            hold_id = next(self.ids)
            acct.holds[hold_id] = [amount, at + ttl, True]
            self.hold_owner[hold_id] = account
            acct.held += amount
            heapq.heappush(acct.hold_expiries, (at + ttl, hold_id))
            return hold_id

        def _close(self, hold_id, at):
            account = self.hold_owner[hold_id]        # KeyError for an unknown or closed hold
            acct = self._clock(account, at)
            del self.hold_owner[hold_id]
            hold = acct.holds.pop(hold_id)
            if hold[2]:
                acct.held -= hold[0]
            return account, acct

        def settle(self, hold_id, actual, at):
            if not isinstance(actual, int) or actual < 0:
                raise ValueError("actual is a non-negative integer")
            account, acct = self._close(hold_id, at)
            charged = min(actual, self._available(acct))
            parts = self._spend(acct, charged) if charged else []
            self.charges[hold_id] = [account, parts, charged]
            acct.balance -= charged
            acct.log.append({"at": at, "kind": "settle", "delta": -charged, "charge": hold_id})
            return charged

        def release(self, hold_id, at):
            self._close(hold_id, at)

        def refund(self, charge_id, amount, at, key):
            if key in self.refund_keys:
                return self.refund_keys[key]
            self._positive(amount)
            charge = self.charges[charge_id]            # KeyError for an unknown charge
            account, parts, refundable = charge
            if amount > refundable:
                raise ValueError("refund exceeds what is left of the charge")
            acct = self._clock(account, at)
            charge[2] -= amount
            left, restored = amount, 0
            while left:                                  # latest-expiring source first
                grant, units = parts[-1]
                back = min(left, units)
                if back == units:
                    parts.pop()
                else:
                    parts[-1] = (grant, units - back)
                left -= back
                if grant.expires_at > at:
                    grant.remaining += back
                    restored += back
                    if not grant.queued:
                        grant.queued = True
                        heapq.heappush(acct.queue, (grant.expires_at, grant.seq, grant))
            acct.balance += restored
            acct.log.append({"at": at, "kind": "refund", "delta": restored, "charge": charge_id})
            self.refund_keys[key] = restored
            return restored

        def audit(self, account, at):
            return list(self._clock(account, at).log)

      class _Model:
          """A slow, plain model of the rules: lists scanned on every call."""

          def __init__(self):
              self.grants = {}      # account -> list of [remaining, expires_at, seq]
              self.holds = {}       # hold id -> [account, amount, deadline]
              self.charges = {}     # charge id -> [account, [[grant, units], ...], refundable]
              self.keys = {}
              self.seq = 0
              self.expired = {}

          def tick(self, acct, now):
              for g in self.grants.setdefault(acct, []):
                  if g[1] is not None and g[1] <= now and g[0]:
                      self.expired[acct] = self.expired.get(acct, 0) + g[0]
                      g[0] = 0

          def live(self, acct, now):
              return [g for g in self.grants.get(acct, []) if g[0] and (g[1] is None or now < g[1])]

          def balance(self, acct, now):
              return sum(g[0] for g in self.live(acct, now))

          def held(self, acct, now, skip=None):
              return sum(h[1] for i, h in self.holds.items() if h[0] == acct and now < h[2] and i != skip)

          def available(self, acct, now, skip=None):
              return max(0, self.balance(acct, now) - self.held(acct, now, skip))

          def grant(self, acct, amount, now, exp=None):
              self.tick(acct, now)
              self.seq += 1
              self.grants.setdefault(acct, []).append([amount, exp, self.seq])

          def spend(self, acct, amount, now):
              parts = []
              order = sorted(self.live(acct, now), key=lambda g: (float("inf") if g[1] is None else g[1], g[2]))
              for g in order:
                  if not amount:
                      break
                  take = min(amount, g[0])
                  g[0] -= take
                  amount -= take
                  parts.append([g, take])
              assert amount == 0
              return parts

          def consume(self, acct, amount, now, cid):
              self.tick(acct, now)
              if amount > self.available(acct, now):
                  return False
              self.charges[cid] = [acct, self.spend(acct, amount, now), amount]
              return True

          def hold(self, acct, amount, now, ttl, hid):
              self.tick(acct, now)
              if amount > self.available(acct, now):
                  return False
              self.holds[hid] = [acct, amount, now + ttl]
              return True

          def settle(self, hid, actual, now):
              acct = self.holds[hid][0]
              self.tick(acct, now)
              charged = min(actual, self.available(acct, now, skip=hid))
              del self.holds[hid]
              self.charges[hid] = [acct, self.spend(acct, charged, now) if charged else [], charged]
              return charged

          def release(self, hid, now):
              self.tick(self.holds[hid][0], now)
              del self.holds[hid]

          def refund(self, cid, amount, now, key):
              if key in self.keys:
                  return self.keys[key]
              acct, parts, refundable = self.charges[cid]
              self.tick(acct, now)
              self.charges[cid][2] -= amount
              left, restored = amount, 0
              for p in sorted(parts, key=lambda p: (float("inf") if p[0][1] is None else p[0][1]), reverse=True):
                  back = min(left, p[1])
                  p[1] -= back
                  left -= back
                  if p[0][1] is None or now < p[0][1]:
                      p[0][0] += back
                      restored += back
              self.keys[key] = restored
              return restored

The tests open 20,000 holds and settle them, reading available after each, within 3 seconds."

## Question

<Callout kind="interview" label="Coding round">
"Build the credit ledger behind a model API: customers buy token credits, requests spend them. I'll give you the requirements in stages. Each stage has tests, and the next one opens when they pass."
</Callout>

Four stages grow one `Ledger` class. Every method takes the time `at` as an argument, and time never goes backwards across the whole ledger. Credits are integers. Stage n runs its visible tests and then hidden tests against your whole program, and each stage's brief appears only after the stage before it passes. Plan for about an hour.

The hidden tests check rules and invariants, not one canonical trace. These are yours to choose:

- The format of grant, charge and hold ids, and which of two grants with the same expiry is spent first.
- The structures that keep spending, expiry and hold timeouts inside the time budgets.
- What an audit entry carries beyond `at`, `kind` and `delta`.

Two quantities are yours to choose and defend in writing, because no test can pick them for you:

- Before stage 3: how large a hold is for a model request with a known prompt and a `max_tokens` cap, and how long the hold lives compared with the request timeout.
- Before stage 4: how long you keep a charge refundable, and what that costs in memory at 1,000 charges per second.

## Answer

The reference keeps, per account, a running `balance`, a running `held` total and one min-heap of live grants keyed by `(expires_at, sequence)`. One heap serves two jobs, because the spending order and the expiry order are the same order: every call first pops the grants whose expiry has passed (logging each unspent remainder as an `expire` entry at its expiry time), and `consume` then spends from the top of the same heap. Grants without an expiry sort last. A second heap holds hold deadlines, so a timed-out hold leaves `held` on the first call after its deadline. Each charge records which grants it drew from and how much, in spending order, so a refund can walk that list from the end. Every balance change appends to the account's log, which is why the log replays to the balance.

Each stage is built to break the design that only fitted the stage before:

- **Stage 1 breaks the summed balance.** Recomputing the balance by summing grants on every read is 20,000 reads over a list that grows to 20,000 grants, then 20,000 more over the spent list.
- **Stage 2 breaks the sort per spend.** Sorting live grants by expiry on every `consume` costs a sort of 20,000 grants, 10,000 times.
- **Stage 3 breaks the summed holds.** Adding up open holds on every `hold` and `available` call is quadratic in the number of requests in flight.
- **Stage 4 breaks the rescanned refund.** Scanning a charge's sources from the end on every refund, skipping sources already refunded, turns 10,000 one-credit refunds into about 50 million skipped entries.

A plain version that rescans every grant on each call did not finish stage 1 within the local harness's 60-second limit. The reference ran the four timed sections in 0.063 s, 0.084 s, 0.068 s and 0.014 s in CPython 3.12.6; Pyodide in the browser is slower, which is what the 3-second budgets leave room for.

### Three rules with two defensible answers

**What backs a hold.** The reference reserves an amount against the balance, not specific grants. It is simple and keeps `available` a subtraction. The cost: credits can expire under an open hold, and the settle then charges only what is still available, so the provider serves a request it cannot fully bill. The alternative pins the hold to grants and lets pinned credits outlive their expiry until the settle. That bills the request in full, at the cost of honouring credits after the date the customer was told they lapse, and a second data structure that must agree with the first.

**A request that cost more than its hold.** The reference charges up to what is available and absorbs the rest. The alternative records the shortfall as debt and lets the balance go negative. Absorbing loses revenue on every overshoot; debt keeps the revenue but puts collections in the request path, and a customer with debt has to be blocked by some other rule.

**Where a refund goes.** The reference returns credits to the grants they came from, latest-expiring source first, and credits whose grant has already expired are not restored. The alternative issues a fresh grant. A fresh grant is friendlier, and it is a leak: consume a credit the day before it expires, have the request refunded, and the credit comes back with a new expiry. Returning to the source grant closes that, at the cost of a support conversation every time a refund arrives after the source expired.

### The quantities you chose

**Hold size.** The worst case for a model request is the prompt's tokens at the input price plus `max_tokens` at the output price. Holding the worst case never undercharges but locks credits that are almost never spent, so a customer near the end of their balance sees requests refused while credits remain. Holding a high percentile of observed cost frees those credits and accepts an overshoot on the rare request above it, which the absorb rule turns into lost revenue. Either is reasonable if it is written down with the measured percentile; holding nothing, or holding a flat amount independent of `max_tokens`, is not.

**Hold lifetime.** The hold must outlive the request. A TTL shorter than the request timeout lets a long streaming request lose its hold mid-flight, and another request can then spend the same credits. Set the TTL to the request timeout plus the settle path's own worst latency, and alert on settles that arrive for timed-out holds.

**Refund window.** Every charge must be kept while it can be refunded. At 1,000 charges per second, 30 days is 1,000 × 86,400 × 30 = 2.592 billion charges. At an assumed 200 bytes per charge record that is about 518 GB, which does not belong in one process. A short window (a day or two, covering retries and failed requests) fits in memory; longer windows move charges to a store keyed by charge id.

### The constraint conflict

The round's clock against the business rules: a full ledger with pinned holds, debt and fresh-grant refunds does not fit an hour, and the stages reward the version whose rules are simple enough to state in one sentence each. In production the same conflict is revenue against availability: every rule that protects revenue (worst-case holds, debt) refuses or blocks a paying customer somewhere.

**How the recommended design fails.** The auto-released hold is the weak point. A request that outlives its hold is served against credits another request may already have spent, and the settle charges what is left, so a slow request at the end of a balance is under-billed with no error anywhere. The second weak point is memory: the reference keeps every charge, so a long-running process grows without bound until charges past the refund window are dropped or moved out.

## Follow-ups

- *What changes to run this as a real service?* The balance and the holds move into a store with atomic conditional updates per account (a row with a version, or a single-writer partition per account), because two API servers spending the last credits at once is the race the single process hides. Usage goes to billing from the log, keyed by charge id so a retry reports once; Stripe's meter events, for example, accept timestamps up to 35 days old but say an event accepted after an invoice finalises does not update that invoice, so the reporting job must run inside the billing period's own window. Time comes from the request's server-side timestamp, never from a client.
- *Pushback: finance says absorbing overshoot is lost revenue and wants debt. Engineering says debt needs collections. Which do you build, and what do you accept in writing?* Absorb, with the overshoot logged as its own entry kind and reported weekly; accept in writing the measured overshoot total, and agree the number at which debt becomes worth building.
- *Reversal: what would you need to see in production to move from amount holds to pinned holds?* Settles charging less than the actual cost because credits expired under a hold, at a rate finance considers material, or a customer contract promising that a started request is always billed from the credits it started with.
- *Events now arrive out of order: a refund can be recorded with an `at` earlier than the last consume. What changes?* The ledger becomes an event log sorted by event time with a replay from a checkpoint, and every balance read is "as of" a time. Expiry is the hard part: a late refund for a grant that has since expired changes whether that credit was live when it was spent.

## Common mistakes

- Spending in grant order instead of expiry order, which lets early-expiring credits lapse while later ones are spent.
- Letting `consume` spend credits that an open hold has reserved.
- Treating a refund retry as a second refund because it has no idempotency key.
- Restoring refunded credits as a new grant with a fresh expiry.
- Recomputing the balance from all grants on every read, or summing open holds on every call.
- Logging expiries at the time they were noticed instead of the time they happened, which breaks a replay of the log.
