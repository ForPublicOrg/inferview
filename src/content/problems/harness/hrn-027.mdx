    def decide(command, rules):
        """Return "deny", "ask" or "allow" for one shell command string, as a harness gate would.

        rules: {"deny": [...], "ask": [...], "allow": [...]}. Each rule is a list of words that
               match the start of a simple command: ["git", "push"] matches "git push origin main".
               A list can hold tens of thousands of rules.

        Precedence: deny beats ask beats allow, whatever the specificity of the rules. A simple
        command that no rule matches is "ask". A compound command takes the strictest decision
        of its parts.

        Split on && || ; | |& & and newlines only when the script is a chain of plain words:
        no substitution ($( ) or backticks), no $ expansion, no redirection (< >), no subshell
        or group (parentheses, braces), no glob (* ? [), no leading NAME=value assignment and
        no control flow (if, for, while, case and the rest). A script that cannot be split, or
        cannot be parsed (an unclosed quote, an operator with nothing after it), is never "allow".

        Quotes and backslashes are removed before matching, as the shell removes them.

        Deny and ask rules still fire through: a path to the program (/bin/rm), the wrappers
        timeout, nice, nohup, time, env and xargs, a script run by sh, bash or zsh with -c,
        and a command inside a script that cannot be split.

        Allow rules match only the program's bare name at the start of the command, after
        timeout, nice, nohup or time. They never match through env, a path, or a shell's -c.
        """
        raise NotImplementedError

    import re

    _SPECIAL = set("$`<>(){}*?[")
    _CONTROL = {"if", "then", "else", "elif", "fi", "for", "while", "until", "do", "done",
                "case", "esac", "select", "function", "!", "[[", "]]"}
    _ASSIGN = re.compile(r"[A-Za-z_][A-Za-z0-9_]*=")
    _SHELLS = {"sh", "bash", "zsh", "dash", "ksh"}
    _SCAN_SPLIT = re.compile(r"[\s;&|()<>{}$`]+")
    _STRICT = {"allow": 0, "ask": 1, "deny": 2}
    _MAX_DEPTH = 4


    def _base(word):
        return word.rsplit("/", 1)[-1]


    def _index(rules, key):
        # Rules indexed by first word, so a lookup costs one dict probe per word, not one pass over every rule.
        table = {}
        for r in rules.get(key, ()):
            if r:
                table.setdefault(_base(r[0]), []).append(tuple(r[1:]))
        return table


    def _lex(s):
        """Split a script into simple commands. Returns (parts, plain), or (None, False) when it cannot be parsed."""
        parts, words, cur = [], [], []
        has_word, plain, last_op = False, True, None
        i, n = 0, len(s)

        def flush():
            nonlocal cur, has_word
            if has_word:
                words.append("".join(cur))
            cur, has_word = [], False

        while i < n:
            c = s[i]
            if c == "'":
                j = s.find("'", i + 1)
                if j < 0:
                    return None, False
                cur.append(s[i + 1:j])
                has_word, i = True, j + 1
                continue
            if c == '"':
                i += 1
                while True:
                    if i >= n:
                        return None, False
                    c = s[i]
                    if c == '"':
                        break
                    if c == "\\" and i + 1 < n and s[i + 1] in '"\\$`\n':
                        if s[i + 1] != "\n":
                            cur.append(s[i + 1])
                        i += 2
                        continue
                    if c in "$`":
                        plain = False
                    cur.append(c)
                    i += 1
                has_word, i = True, i + 1
                continue
            if c == "\\":
                if i + 1 >= n:
                    return None, False
                if s[i + 1] != "\n":
                    cur.append(s[i + 1])
                    has_word = True
                i += 2
                continue
            if c in " \t\r":
                flush()
                i += 1
                continue
            two = s[i:i + 2]
            op = two if two in ("&&", "||", "|&") else (c if c in ";|&\n" else None)
            if op:
                flush()
                if words:
                    parts.append(words)
                    words = []
                elif op != "\n":
                    return None, False        # an operator with no command before it
                last_op = op
                i += len(op)
                continue
            if c in _SPECIAL:
                plain = False
            cur.append(c)
            has_word = True
            last_op = None
            i += 1
        flush()
        if words:
            parts.append(words)
        elif last_op in ("&&", "||", "|", "|&"):
            return None, False                # "npm test &&": nothing after the operator
        for p in parts:
            if p[0] in _CONTROL or _ASSIGN.match(p[0]):
                plain = False
        return parts, plain


    def _strip_for_allow(argv):
        # Wrappers that run their argument unchanged; env is not here, because an assignment can change what runs.
        while argv:
            head = argv[0]
            if head in ("nohup", "time"):
                argv = argv[1:]
            elif head == "timeout":
                rest = argv[1:]
                while rest and rest[0].startswith("-"):
                    rest = rest[1:]
                argv = rest[1:]
            elif head == "nice":
                rest = argv[1:]
                if rest[:1] == ["-n"]:
                    rest = rest[2:]
                elif rest and rest[0].startswith("-"):
                    rest = rest[1:]
                argv = rest
            else:
                break
        return argv


    def _matches_anywhere(words, table):
        # Deny and ask over-match on purpose: a rule fires wherever its words appear as a run, at any position.
        for k, w in enumerate(words):
            tails = table.get(_base(w))
            if tails:
                for t in tails:
                    if tuple(words[k + 1:k + 1 + len(t)]) == t:
                        return True
        return False


    def _matches_allow(argv, table):
        # Allow under-matches: exact first word (no path), after the wrappers above, rule as a prefix.
        if not argv:
            return False
        for t in table.get(argv[0], ()):
            if tuple(argv[1:1 + len(t)]) == t:
                return True
        return False


    def _scan(script, deny):
        words = _SCAN_SPLIT.split(script.replace("\\\n", "").replace("'", "").replace('"', "").replace("\\", ""))
        return "deny" if _matches_anywhere([w for w in words if w], deny) else "ask"


    def _shell_script(argv):
        if _base(argv[0]) not in _SHELLS:
            return None
        for k in range(1, len(argv)):
            a = argv[k]
            if a.startswith("-") and not a.startswith("--") and "c" in a[1:]:
                return argv[k + 1] if k + 1 < len(argv) else ""
        return None


    def _decide(script, tables, depth):
        deny, ask, allow = tables
        if depth > _MAX_DEPTH:
            return _scan(script, deny)
        parts, plain = _lex(script)
        if parts is None or not plain:
            return _scan(script, deny)
        if not parts:
            return "ask"
        worst = "allow"
        for words in parts:
            if _matches_anywhere(words, deny):
                return "deny"
            inner = _shell_script(words)
            if inner is not None:
                # A shell's -c script is checked for deny, but never allowed: the gate cannot vouch for it.
                d = "deny" if _decide(inner, tables, depth + 1) == "deny" else "ask"
            elif _matches_anywhere(words, ask):
                d = "ask"
            elif _matches_allow(_strip_for_allow(words), allow):
                d = "allow"
            else:
                d = "ask"
            if d == "deny":
                return "deny"
            if _STRICT[d] > _STRICT[worst]:
                worst = d
        return worst


    def decide(command, rules):
        tables = (_index(rules, "deny"), _index(rules, "ask"), _index(rules, "allow"))
        return _decide(command, tables, 0)

## Question

<Callout kind="interview" label="Coding, 45 minutes">
Every shell command your coding agent proposes goes through a permission gate before it runs. Security writes the rules as word prefixes in three lists: deny, ask and allow. Write <code>decide</code> to the contract in the starter. Deny beats ask beats allow, a compound command is judged by its strictest part, and a script the gate cannot read safely never reaches allow.
</Callout>

The two documented harnesses take different positions on the same input, as of 2026-10-28. Claude Code splits compound commands on `&&`, `||`, `;`, `|`, `|&`, `&` and newlines, requires every part to be allowed, and applies deny and ask rules inside subshells, substitutions and loop bodies. Codex splits a `bash -lc` script only when it is a linear chain of plain words; a script with redirection, substitution, variables, wildcards or control flow is evaluated as one command against the rules. The contract in the starter follows Codex on splitting and Claude Code on precedence and wrappers.

Part of the policy is yours: how far deny and ask rules over-match (anywhere in a command, or only where a program could start), how deep to follow a `bash -c` inside a `sh -c`, and whether a quoted metacharacter, as in `cat 'a > b'`, makes a script unsplittable. The tests check invariants: a chain decides as its strictest part, adding a deny rule never loosens a decision, an unsplittable script is never allowed, and a list of bypass attempts is denied.

One constraint is not in the visible tests. In this scenario the rule lists are generated from a monorepo's package scripts and run to tens of thousands of entries, and the gate runs on every shell call of every session. With 50,000 rules and a chain of 1,000 commands, one call has to finish well under a second.

## Answer

The reference solution has three parts: a lexer that decides whether a script can be split, two matchers that err in opposite directions, and an index that makes each call linear in the size of the command.

**The lexer.** One pass over the script tracks single quotes, double quotes, backslashes and line continuations, and cuts a simple command at each separator outside quotes. Outside single quotes, any of `` $ ` < > ( ) { } * ? [ `` marks the script as not plain; inside double quotes, `$` and the backtick still do, because the shell still expands them there. A leading `NAME=value` or a control keyword in the first position does the same. An unclosed quote, a trailing `&&` or two separators in a row make the script unparseable. Quote removal happens in the same pass, so `git 'push'`, `g''it push` and `r\m` reach the matcher as `git push` and `rm`.

**Deny and ask over-match; allow under-matches.** This asymmetry is the policy decision the problem turns on.

```python
def _matches_anywhere(words, table):
    for k, w in enumerate(words):                     # any position, first word compared by basename
        for tail in table.get(_base(w), ()):
            if tuple(words[k + 1:k + 1 + len(tail)]) == tail:
                return True
    return False

def _matches_allow(argv, table):                      # argv after timeout, nice, nohup, time only
    return any(tuple(argv[1:1 + len(t)]) == t for t in table.get(argv[0], ())) if argv else False
```

A deny rule that fires wherever its words appear catches `/bin/rm`, `timeout 5 rm`, `env FOO=1 rm`, `xargs rm` and `nohup rm` without a wrapper table for every program that runs its arguments. The cost is false denials: `grep -rn rm src/` is denied, and the agent has to rephrase or ask. An allow rule matches only the bare name in the first position. `./git status` is not allowed, because the agent can write a file named `git`; `env PATH=/tmp/bin git status` is not allowed, because the assignment changes which `git` runs. Claude Code documents the same asymmetry for leading assignments: an allow rule stops at an unknown assignment, a deny or ask rule matches past any.

**Scripts the gate will not split.** The reference removes quotes and backslashes, splits the raw text on whitespace and shell punctuation, and denies if any deny rule's words appear as a run; otherwise it asks. It never allows. `ls $(rm -rf build)`, `` ls `rm -rf build` `` and `cat <(curl ...)` are denied, and `git diff > review.patch` is asked. A `sh`, `bash` or `zsh` with `-c` is followed into its script up to four levels, then scanned; the result is deny or ask, never allow, because the gate would be vouching for a second parse.

**Performance.** Rules are indexed by their first word, so each word in the command costs one dictionary probe. Walking every rule for every part costs 1,000 × 50,000 = 5 × 10<sup>7</sup> comparisons per call at the hidden test's size, which takes seconds in CPython and longer in Pyodide. The reference still builds the index on every call, which is linear in the rules; a real gate builds it once when the rules load.

**The decision with two defensible answers: what to do with a script you will not split.**

- **Evaluate it as one command, as Codex does.** No prefix rule matches the whole script, so it goes to approval. The gate's parser stays small, and a construct it misreads costs a prompt, not a bypass. The cost lands on people: every `2>&1`, every `$(git rev-parse HEAD)` prompts, and Anthropic reports that users accept 93% of manual prompts, so more prompts buy less attention per prompt.
- **Parse deeper, as Claude Code does,** and apply deny and ask rules inside substitutions, subshells and loop bodies, prompting only when the parse fails. Fewer prompts, and a denied program inside `$( )` is denied rather than asked. The cost is a larger parser that must agree with the shell on every construct; each disagreement is a bypass.

The conflict is autonomy against what the gate can actually read. The reference takes the first position for allow and borrows the second for deny: it never allows what it did not split, and it scans unsplit text for denied words. I would keep that until traces show unsplittable scripts are a large share of prompts; the threshold is yours to set and defend, and 20% of all prompts is where I would start building the deeper parse.

**Where this fails.** A gate over argv is not a security boundary, and the reference is no exception. `git -C . push origin main` puts `-C .` between `git` and `push`, so the run `git push` never appears and the command is asked, not denied. `x=rm; $x -rf build` is asked. An allow rule for `npm test` runs whatever `package.json` says the test script is, and the agent can edit `package.json`. Claude Code's documentation says the same of its own rules: a deny rule covers the invocation the model usually writes, not the program. The sandbox's filesystem and network limits are the boundary; the gate decides when to bother a person.

## Common mistakes

- Splitting with a regex on `&&`, `;` and `|` before handling quotes, which cuts `echo 'a;b'` in half and misses the newline in `ls` followed by a newline and `rm -rf build`.
- Matching allow rules by basename, so `./git status` runs a file the agent wrote.
- Letting the most specific rule win, so an allow for `rm -rf build` beats a deny for `rm`.
- Returning allow for a command that matches no rule.
- Walking every rule for every part; it passes the examples and stalls on a generated rule list.
