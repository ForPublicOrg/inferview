    class TeacherError(Exception):
        pass


    class MockTeacher:
        """A deterministic stand-in for a teacher program with three predictors: route, search (which may run more
        than once in one trace, as a ReAct step does) and answer.

        script maps an example id to a list of outcomes, one per round, each "pass", "fail" or "error"; the last
        outcome repeats for later rounds and ids not in script always pass. teacher(example, round) returns
        {"prediction": {"answer": ...}, "trace": [(predictor_name, inputs, outputs), ...]} or raises TeacherError.
        A "fail" outcome answers "unknown". Like some clients, it reuses its output dicts between calls.
        Every call is appended to calls as (id, round)."""

        def __init__(self, script=None, searches=None):
            self.script = script or {}
            self.searches = searches or {}
            self.calls = []
            self._route = {}
            self._answer = {}

        def __call__(self, example, round=0):
            self.calls.append((example["id"], round))
            outcomes = self.script.get(example["id"], ["pass"])
            outcome = outcomes[min(round, len(outcomes) - 1)]
            if outcome == "error":
                raise TeacherError("tool timeout on " + example["id"])
            q = example["question"]
            self._route["queue"] = "billing" if "refund" in q else "general"
            trace = [("route", {"question": q}, self._route)]
            for i in range(self.searches.get(example["id"], 1)):
                trace.append(("search", {"query": q, "step": i}, {"passages": ["doc-%s-%d" % (example["id"], i)]}))
            self._answer["answer"] = example["answer"] if outcome == "pass" else "unknown"
            trace.append(("answer", {"question": q}, self._answer))
            return {"prediction": self._answer, "trace": trace}


    def exact(example, prediction, trace):
        return prediction["answer"] == example["answer"]


    def bootstrap_fewshot(teacher, trainset, metric, max_bootstrapped=4, max_rounds=1, exclude=(), threshold=None):
        """Collect few-shot demos from teacher traces that pass the metric.

        trainset: list of {"id", "question", "answer"}. metric(example, prediction, trace) returns a bool or a number;
        with threshold set, a trace passes when the score is at least threshold, otherwise when the score is truthy.
        exclude: ids that must never be sent to the teacher or become demos (the dev and held-out sets).
        Return {"demos": {predictor_name: [{"inputs": ..., "outputs": ...}, ...]}, "used": [ids of passing examples]}."""
        raise NotImplementedError

    import copy


    class TeacherError(Exception):
        pass


    class MockTeacher:
        """A deterministic stand-in for a teacher program with three predictors: route, search (which may run more
        than once in one trace, as a ReAct step does) and answer.

        script maps an example id to a list of outcomes, one per round, each "pass", "fail" or "error"; the last
        outcome repeats for later rounds and ids not in script always pass. teacher(example, round) returns
        {"prediction": {"answer": ...}, "trace": [(predictor_name, inputs, outputs), ...]} or raises TeacherError.
        A "fail" outcome answers "unknown". Like some clients, it reuses its output dicts between calls.
        Every call is appended to calls as (id, round)."""

        def __init__(self, script=None, searches=None):
            self.script = script or {}
            self.searches = searches or {}
            self.calls = []
            self._route = {}
            self._answer = {}

        def __call__(self, example, round=0):
            self.calls.append((example["id"], round))
            outcomes = self.script.get(example["id"], ["pass"])
            outcome = outcomes[min(round, len(outcomes) - 1)]
            if outcome == "error":
                raise TeacherError("tool timeout on " + example["id"])
            q = example["question"]
            self._route["queue"] = "billing" if "refund" in q else "general"
            trace = [("route", {"question": q}, self._route)]
            for i in range(self.searches.get(example["id"], 1)):
                trace.append(("search", {"query": q, "step": i}, {"passages": ["doc-%s-%d" % (example["id"], i)]}))
            self._answer["answer"] = example["answer"] if outcome == "pass" else "unknown"
            trace.append(("answer", {"question": q}, self._answer))
            return {"prediction": self._answer, "trace": trace}


    def exact(example, prediction, trace):
        return prediction["answer"] == example["answer"]


    def bootstrap_fewshot(teacher, trainset, metric, max_bootstrapped=4, max_rounds=1, exclude=(), threshold=None):
        excluded = set(exclude)
        demos, used, seen = {}, [], set()
        pending = []
        for example in trainset:
            key = " ".join(example["question"].lower().split())
            if example["id"] in excluded or key in seen:
                continue
            seen.add(key)
            pending.append(example)
        for rnd in range(max_rounds):
            failed = []
            for example in pending:
                if len(used) >= max_bootstrapped:
                    return {"demos": demos, "used": used}
                try:
                    result = teacher(example, rnd)
                except Exception:
                    failed.append(example)
                    continue
                score = metric(example, result["prediction"], result["trace"])
                if not (score >= threshold if threshold is not None else score):
                    failed.append(example)
                    continue
                used.append(example["id"])
                last = {}
                for name, inputs, outputs in result["trace"]:
                    last[name] = {"inputs": copy.deepcopy(inputs), "outputs": copy.deepcopy(outputs)}
                for name, demo in last.items():
                    demos.setdefault(name, []).append(demo)
            pending = failed
        return {"demos": demos, "used": used}

## Question

<Callout kind="interview" label="Code, 45 minutes">
"Write the bootstrapping step of a few-shot optimiser. A teacher program runs over the training set; keep a demonstration for each predictor only from runs that pass the metric, cap how many you keep, and do not spend teacher calls you do not need. The teacher in the tests is a mock. Tell me what changes when it is a real model."
</Callout>

`MockTeacher` in the starter is a deterministic three-predictor program (route, search, answer); a script decides which examples pass, fail or raise in each round, and `searches` makes the search predictor run more than once in one trace, as a ReAct step does. It reuses its output dicts between calls, the way some clients reuse response objects. `exact` is a metric that compares answers.

Write `bootstrap_fewshot(teacher, trainset, metric, max_bootstrapped, max_rounds, exclude, threshold)` returning `{"demos": {predictor: [{"inputs", "outputs"}, ...]}, "used": [ids]}`. The contract:

- A trace becomes demonstrations only if it passes the metric: the score is truthy, or at least `threshold` when one is given. A teacher that raises is a failed trace, not a crash.
- At most `max_bootstrapped` examples are used, and no predictor ends up with more than `max_bootstrapped` demonstrations. When a predictor appears several times in one trace, which occurrence you keep is your decision.
- Ids in `exclude` (the dev and held-out sets) never reach the teacher.
- Two training examples with the same question, ignoring case and whitespace, count once.
- With `max_rounds` above 1, failed examples are retried in the next round with the round number passed to the teacher, and only after the current round has finished without filling the cap.
- Teacher calls are billed. The tests run a 100,000-example training set and count the calls.

The visible tests cover the cap, failures and the call budget; the hidden tests check the rest of the contract.

## Answer

The reference solution does the cheap filtering first: one pass over the training set, without teacher calls, drops excluded ids and repeated questions, using a set for the exclusions so a long exclusion list costs one hash lookup per example. Then it runs rounds. Each round calls the teacher on the pending examples in order and returns as soon as `max_bootstrapped` have passed, so the 100,000-example test costs a handful of calls. An exception or a failing score puts the example on the next round's list.

For a passing trace it walks the steps and keeps one demonstration per predictor, the last occurrence, deep-copied. The copy is the part that is easy to miss: the mock reuses its output dicts, so a solution that stores `outputs` by reference ends with every answer demonstration showing whatever the teacher said last. Real clients do this less often than mocks, but a trace object that a framework mutates after the call has the same effect.

Keeping the last search step is a policy choice. The last step is the one that led to the answer; the first is the one that started the search; keeping all three would let one example fill a predictor's quota and push out other examples. The hidden tests accept any choice that keeps the cap and takes demonstrations only from passing traces.

**The decision, with two defensible answers.** Stop at the first `max_bootstrapped` passing examples in training-set order, as the reference does: the cost is a bias toward whatever sits at the front of the training set, so shuffle it with a recorded seed first. Or call the teacher on a larger sample and choose the demonstrations for coverage (one per category, say): the cost is more teacher calls, and a selection policy that now needs its own evaluation. With a cap of 4 and a training set of thousands, the first is the default and the second is worth its calls only when the held-out set shows a category with no demonstrations failing.

**The constraint conflict.** More rounds recover examples that failed by chance and spend more teacher calls; one round is cheap and drops every hard example, so the demonstrations come only from easy cases. That bias is the failure mode of the recommended answer: demos bootstrapped from passing traces over-represent what the teacher already does well, and the metric decides what "passing" means. A metric that passes a right answer for the wrong reason (a lucky guess, a leaked label) puts that reasoning into every future prompt.

## Follow-ups

- *Pushback: your demos come only from traces the teacher already gets right. Why would they teach the student anything, and what would you measure to find out?*
- *Reversal: what would make you drop bootstrapped demonstrations and use labelled examples from the training set instead?* A held-out result where the labelled examples do as well, at no teacher cost.
- *Operating: what changes to run this against a real model client?* Identical requests may come back from the client's cache, so a retry round has to change the request (a round-dependent temperature or a nonce) or it gets the same failure again; calls run in parallel under a rate limit, so the early stop has to cancel in-flight calls; and the teacher's model, version and the seed go into the registry entry with the demonstrations.
- *Two senior engineers disagree about whether an example whose question appears in the dev set under another id should be excluded. How do you decide without deciding by rank?*

## Common mistakes

- Running the teacher over the whole training set and filtering afterwards, which bills every call.
- Storing the teacher's output objects by reference.
- Counting the cap per predictor occurrence, so one ReAct trace with three search steps fills the search quota.
- Retrying a failure immediately in the same round, which changes nothing with a deterministic teacher and doubles the calls.
- Excluding dev ids after calling the teacher, which leaks nothing into the demos but still pays for the calls.
