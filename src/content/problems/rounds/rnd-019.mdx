        """A 2-D tensor over lists of rows, with reverse-mode autograd.

            raise NotImplementedError

            raise NotImplementedError

            raise NotImplementedError

            raise NotImplementedError

            """Sum of all elements, as a 1 x 1 tensor."""
            raise NotImplementedError

            """Only on a 1 x 1 tensor (ValueError otherwise): set .grad on every leaf (a tensor built

            raise NotImplementedError

    class Tensor:
        # A second backward() adds to leaf gradients, as PyTorch does: gradient accumulation over micro-batches needs it,
        # and the cost is that every training step must clear gradients first or it trains on stale sums.
        ACCUMULATES = True

        def __init__(self, data, requires_grad=False, _parents=(), _backward=None):
            self.data = [list(map(float, row)) for row in data]
            self.requires_grad = requires_grad
            self.grad = None
            self._parents = _parents
            self._backward = _backward      # maps this node's upstream gradient to one gradient per parent

        @property
        def shape(self):
            return (len(self.data), len(self.data[0]))

        def _result(self, data, parents, backward):
            rg = any(p.requires_grad for p in parents)
            return Tensor(data, rg, parents if rg else (), backward if rg else None)

        @staticmethod
        def _bshape(a, b):
            out = []
            for x, y in zip(a, b):
                if x != y and x != 1 and y != 1:
                    raise ValueError(f"cannot broadcast {a} with {b}")
                out.append(max(x, y))
            return tuple(out)

        @staticmethod
        def _reduce(grad, shape):
            # Undo broadcasting: sum the gradient over every axis the operand was stretched along.
            r, c = len(grad), len(grad[0])
            if shape[0] == 1 and r > 1:
                grad = [[sum(grad[i][j] for i in range(r)) for j in range(c)]]
            if shape[1] == 1 and c > 1:
                grad = [[sum(row)] for row in grad]
            return grad

        def _elementwise(self, other, f, df):
            R, C = self._bshape(self.shape, other.shape)
            a, b = self.data, other.data
            at = lambda m, i, j: m[i if len(m) > 1 else 0][j if len(m[0]) > 1 else 0]
            data = [[f(at(a, i, j), at(b, i, j)) for j in range(C)] for i in range(R)]

            def backward(g):
                ga = [[g[i][j] * df(at(a, i, j), at(b, i, j))[0] for j in range(C)] for i in range(R)]
                gb = [[g[i][j] * df(at(a, i, j), at(b, i, j))[1] for j in range(C)] for i in range(R)]
                return self._reduce(ga, self.shape), self._reduce(gb, other.shape)

            return self._result(data, (self, other), backward)

        def __add__(self, other):
            return self._elementwise(other, lambda x, y: x + y, lambda x, y: (1.0, 1.0))

        def __mul__(self, other):
            return self._elementwise(other, lambda x, y: x * y, lambda x, y: (y, x))

        def __matmul__(self, other):
            (n, m), (m2, p) = self.shape, other.shape
            if m != m2:
                raise ValueError(f"cannot multiply {self.shape} by {other.shape}")
            a, b = self.data, other.data
            bt = [list(col) for col in zip(*b)]
            data = [[sum(x * y for x, y in zip(row, col)) for col in bt] for row in a]

            def backward(g):
                ga = [[sum(x * y for x, y in zip(grow, brow)) for brow in b] for grow in g]           # g @ B^T
                gt = [list(col) for col in zip(*g)]
                gb = [[sum(a[i][k] * gt[j][i] for i in range(n)) for j in range(p)] for k in range(m)]  # A^T @ g
                return ga, gb

            return self._result(data, (self, other), backward)

        def relu(self):
            data = [[x if x > 0 else 0.0 for x in row] for row in self.data]

            def backward(g):
                return ([[gv if x > 0 else 0.0 for gv, x in zip(grow, row)] for grow, row in zip(g, self.data)],)

            return self._result(data, (self,), backward)

        def sum(self):
            r, c = self.shape

            def backward(g):
                return ([[g[0][0]] * c for _ in range(r)],)

            return self._result([[sum(sum(row) for row in self.data)]], (self,), backward)

        def backward(self):
            if self.shape != (1, 1):
                raise ValueError("backward() needs a 1 x 1 output")
            if not self.requires_grad:
                return
            # Iterative topological order: a recursive walk overflows the stack on a graph 20,000 nodes deep.
            order, visited, stack = [], set(), [(self, False)]
            while stack:
                node, done = stack.pop()
                if done:
                    order.append(node)
                    continue
                if id(node) in visited:
                    continue
                visited.add(id(node))
                stack.append((node, True))
                for p in node._parents:
                    if p.requires_grad and id(p) not in visited:
                        stack.append((p, False))
            grads = {id(self): [[1.0]]}
            for node in reversed(order):
                g = grads.pop(id(node), None)
                if g is None:
                    continue
                if not node._parents:
                    if node.grad is None or not Tensor.ACCUMULATES:
                        node.grad = g
                    else:
                        node.grad = [[x + y for x, y in zip(r1, r2)] for r1, r2 in zip(node.grad, g)]
                    continue
                for p, pg in zip(node._parents, node._backward(g)):
                    if not p.requires_grad:
                        continue
                    if id(p) in grads:
                        acc = grads[id(p)]
                        grads[id(p)] = [[x + y for x, y in zip(r1, r2)] for r1, r2 in zip(acc, pg)]
                    else:
                        grads[id(p)] = pg

## Question

<Callout kind="interview" label="Coding round">
"Implement reverse-mode autograd for 2-D tensors: add, multiply, matmul, ReLU and sum, and backward from a scalar. No autograd library; the gradients come from your own rules."
</Callout>

Fill in `Tensor`. `+` and `*` are elementwise and broadcast: on each axis the sizes match, or one of them is 1, so a `(1, m)` bias adds to every row of an `(n, m)` tensor and an `(n, 1)` column scales every column. `@` multiplies `(n, m)` by `(m, p)`. `sum()` returns a `1 x 1` tensor.

The rules your engine must keep:

- `backward()` runs only on a `1 x 1` tensor (`ValueError` otherwise) and sets `.grad` on every leaf that requires grad, as a list of rows shaped like that leaf's data. A leaf that does not require grad keeps `.grad` as `None`.
- A result requires grad when any input does.
- A tensor may be used many times. The visible tests build a chain 20,000 nodes deep, and the hidden tests build graphs where one leaf reaches the output along 2^60 distinct paths; `backward` must cost time in proportion to the nodes and edges, not the paths.

One decision is yours: what a second `backward()` on a new graph over the same leaves does to their `.grad`. Set `ACCUMULATES` and say why in the docstring; the hidden tests check that the engine does what the flag says.

## Answer

**One closure per op, one pass in topological order.** Each op records its parents and a function from the upstream gradient to one gradient per parent. `backward` orders the graph so every node comes after everything that consumes it, seeds the output with 1, and walks that order once. A node's incoming gradients are summed in a dictionary before it passes anything on. That summing is what makes the cost linear in edges: the 2^60-path graph has 61 nodes and 120 edges, and an engine that pushes gradients along each path instead never finishes.

**The order has a trap.** The reference builds a post-order with an explicit stack and marks a node visited when it is popped and expanded. Marking it when it is first pushed looks equivalent and is not: for `h + h * h`, the add's parents are `h` and `h * h`; if `h` is marked on push, the walk finishes `h * h` without descending into `h`, finishes `h` afterwards, and the reverse order then processes `h` before `h * h` has sent it a gradient. The visible tests pass with that bug; the hidden tests build exactly this shape. A recursive walk is correct but fails the 20,000-deep chain with `RecursionError`. Raising `sys.setrecursionlimit` makes that test pass and swaps the error for a depth at which the interpreter can crash, which the Python docs warn about; the iterative walk has no depth limit to choose.

**Broadcasting in reverse.** If an operand was stretched along an axis, its gradient is the upstream gradient summed along that axis: a `(1, m)` bias gets the column sums of an `(n, m)` gradient, an `(n, 1)` column gets the row sums, a `(1, 1)` scalar gets the total. For `C = A @ B`, the gradient of `A` is `g @ Bᵀ` (shape `n x m`) and of `B` is `Aᵀ @ g` (shape `m x p`). Every gradient has the shape of its tensor, which is the first assertion worth writing.

**The accumulate decision.** Accumulating is what PyTorch's `backward` does: its docs say gradients accumulate in the leaves and may need zeroing. It lets a training loop sum gradients over micro-batches without a second buffer and lets two losses from one forward pass call `backward` separately. Its cost is the bug it invites: a loop that forgets to clear gradients trains on a running sum, and nothing fails. Overwriting removes that bug; its cost is that micro-batch accumulation needs an explicit buffer and two-loss training must add the losses before one `backward`. The reference accumulates, because the training code it serves accumulates over micro-batches; a team whose loops never accumulate is better served by overwrite.

**The constraint the reference pays for.** Every closure holds references to its inputs' data, so the whole graph and every intermediate stays alive for as long as the output tensor is reachable. Memory against simplicity: freeing each node's saved data after its gradient is computed bounds memory to the live frontier, at the price of a graph that can be walked once. The reference does not free, so a training loop that appends each step's loss tensor to a list for logging keeps every step's graph alive and grows without bound; log `loss.data`, not `loss`.

When this problem was checked in CPython 3.12.6, the reference ran the visible 20,000-node chain, forward and backward, in 0.19 s.

## Follow-ups

- *What changes to run this on a real tensor library?* Storage moves to device arrays and each op's backward becomes a kernel; a `no_grad` mode skips recording parents so inference keeps no graph; saved inputs are freed after use, which is what PyTorch's default of not retaining the graph does; in-place ops need a version counter so a saved input that was overwritten raises instead of producing a wrong gradient.
- *Pushback: at your target batch size, the activations your closures save are twice the memory you have. Recomputing segments of the forward pass during backward fixes memory and costs compute. Which do you give up, and what do you write down?* Name the segments recomputed, the measured step-time increase, and the batch size you would have needed without it; the alternative is a smaller batch with a learning-rate change that needs its own eval.
- *Reversal: what would make you switch `ACCUMULATES` from accumulate to overwrite?* Evidence that the code base has loops calling `backward` more than once per step without clearing, and no loop that uses micro-batch accumulation.
- *Softmax followed by cross-entropy as two ops gives NaN on large logits. What do you change in the engine?* Add one fused op whose forward uses log-sum-exp and whose backward is softmax minus the one-hot label, divided by the batch size; the composed version takes the log of an exponential that already overflowed.

## Common mistakes

- Marking nodes visited when pushed rather than when expanded, which processes a node before all its consumers.
- Recursing through the graph, which stops at the recursion limit.
- Propagating per path instead of summing per node first.
- Returning an unreduced gradient for a broadcast operand, shaped like the output instead of the operand.
- Giving a leaf that does not require grad a zero gradient instead of `None`, which hides a frozen weight that was meant to train.
