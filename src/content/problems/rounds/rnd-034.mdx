      """The usage table. In production it lives in the warehouse and every scan is a full read billed by bytes, so code

      class ScanBudgetExceeded(RuntimeError):

      class Store:
          def __init__(self, records, max_scans=None):

              """Every record, oldest first. Each call is one full scan."""
              self.scans += 1

              return len(self._records)

  solution: |
    """Spend per tenant. Windows are half-open: [start, end) in Unix seconds, like every range in this package."""
    import heapq
    from bisect import bisect_left
    from decimal import Decimal

    from usage.prices import PER, price_for


    def record_cost(record):
        """Dollars for one call, unrounded. input_tokens includes cached_tokens (see records.py)."""
        p = price_for(record.model)
        uncached = record.input_tokens - record.cached_tokens
        return (
            uncached * p.input
            + record.cached_tokens * p.cached
            + record.output_tokens * p.output
        ) / PER


    def tenant_spend(store, tenant, start, end):
        """Dollars a tenant spent in [start, end)."""
        total = Decimal(0)
        for r in store.scan():
            if r.tenant == tenant and start <= r.ts < end:
                total += record_cost(r)
        return total


    def _top(spends, k):
        # Highest spend first; equal spends in tenant order, so the answer does not depend on scan order.
        return heapq.nsmallest(k, spends.items(), key=lambda item: (-item[1], item[0])) if k > 0 else []


    def top_tenants(store, start, end, k):
        """The k tenants with the highest spend in [start, end), highest first, as (tenant, dollars) pairs.
        Only tenants with at least one record in the window are candidates."""
        spends = {}
        for r in store.scan():
            if start <= r.ts < end:
                spends[r.tenant] = spends.get(r.tenant, Decimal(0)) + record_cost(r)
        return _top(spends, k)


    class SpendIndex:
        """Answers many spend and top-k questions about one store. Built from at most one scan.

        Per tenant: timestamps in order and prefix sums of cost, so a window is two binary searches and a subtraction.
        Decimal keeps the subtraction exact, so the answer equals tenant_spend to the last digit."""

        def __init__(self, store):
            self._ts = {}
            self._sums = {}
            for r in store.scan():  # oldest first, so each tenant's list stays sorted
                ts = self._ts.setdefault(r.tenant, [])
                sums = self._sums.setdefault(r.tenant, [Decimal(0)])
                ts.append(r.ts)
                sums.append(sums[-1] + record_cost(r))

        def _window(self, tenant, start, end):
            ts = self._ts.get(tenant)
            if not ts or start >= end:
                return 0, 0, None
            return bisect_left(ts, start), bisect_left(ts, end), self._sums[tenant]

        def spend(self, tenant, start, end):
            """Same answer as tenant_spend."""
            lo, hi, sums = self._window(tenant, start, end)
            return sums[hi] - sums[lo] if sums else Decimal(0)

        def top(self, start, end, k):
            """Same contract as top_tenants."""
            spends = {}
            for tenant in self._ts:
                lo, hi, sums = self._window(tenant, start, end)
                if sums and hi > lo:
                    spends[tenant] = sums[hi] - sums[lo]
            return _top(spends, k)

## Question

`usage/` turns the records an LLM gateway writes, one per model call, into dollars per tenant. `records.py` defines the record, `prices.py` the price table (the team's scenario prices, not a provider's), `store.py` the usage table and `invoice.py` the one place money is rounded. You edit `usage/billing.py`.

**Phase 1.** Finance says tenants with high prompt-cache hit rates are billed more than the provider charges the company for their calls. Find and fix the bug.

**Phase 2.** Implement `top_tenants(store, start, end, k)`: the k tenants with the highest spend in a half-open window, highest first. Decide how ties are ordered and say why.

**Phase 3.** The finance dashboard asks thousands of spend and top-k questions per page load. In production every `store.scan()` is a full read of a warehouse table, billed by bytes; the tests turn that into a rule by creating the store with `max_scans=1`. Implement `SpendIndex` so it answers every question from at most one scan. Choose the data structure and say what it costs in memory, and what happens to it as new records arrive.

## Answer

**Phase 1: cached tokens billed twice.** `records.py` says the gateway writes the OpenAI Responses usage shape, where `input_tokens` is every input token and `cached_tokens` is the part of it read from cache. `record_cost` prices all of `input_tokens` at the full rate and then prices `cached_tokens` again at the cached rate. For the visible record, 1,000,000 input tokens of which 600,000 cached on `model-s`, the shipped code computes 1.0 × $0.40 + 0.6 × $0.10 = $0.46; the correct cost is 0.4 × $0.40 + 0.6 × $0.10 = $0.16 + $0.06 = $0.22. The fix prices `input_tokens - cached_tokens` at the full rate. The bug is non-algorithmic, which is the class the phase is reported to use, and it is found by reading the record's contract in another file, not by reading `billing.py` harder.

The fix has its own failure mode, and it belongs in the answer. Anthropic's `input_tokens` excludes cache reads, which are reported separately. If the gateway ever logs Claude usage in its raw shape, the fixed code subtracts cached tokens from a number that never contained them, under-bills, and goes negative when cached exceeds input. Normalise at ingestion, in `parse_line`, per provider, and reject a record whose `cached_tokens` exceeds `input_tokens` instead of billing it.

**Phase 2: top-k.** One scan sums spend per tenant inside `[start, end)`, then `heapq.nsmallest(k, ..., key=(-spend, tenant))` returns the top k with ties broken by tenant name. Sorting everything is also fine at this size. The tie-break is the decision: break ties by name and the dashboard is stable across loads; leave them in dict order and two loads of the same page can disagree. The hidden tests accept any order among equal spends and check the invariants: descending order, exactly min(k, active tenants) entries, each value equal to the direct sum, and no tenant left out that spent more than the last one included.

**Phase 3: one scan, many questions.** The shipped `tenant_spend` scans the table per call, so the second question raises `ScanBudgetExceeded`; that is the larger tier. Two structures are defensible.

- **Per-tenant prefix sums.** In the one scan, append each record's timestamp and running total to its tenant's lists (the scan is in time order, so the lists stay sorted). A window is two `bisect_left` calls and a subtraction; top-k is that per tenant. Cost: memory proportional to the number of records, and a record that arrives late with an old timestamp has to be inserted in the middle, which shifts every later running total.
- **Per-tenant buckets** (hourly or daily totals). Memory proportional to tenants times buckets, independent of record count, and late records only touch one bucket. Cost: a window that does not start and end on a bucket edge needs the raw records at its edges, which the one-scan rule forbids, so the answers are exact only on bucket boundaries. You choose the bucket width; it is the precision the dashboard promises.

The reference solution uses prefix sums, because invoices need exact windows. Money stays `Decimal`, as `prices.py` requires, and that matters more here than anywhere else: a prefix-sum subtraction in `float` differs from the direct sum in the last digits, and an invoice rounded half-even to the cent can flip. With `Decimal` the subtraction is exact, so `SpendIndex.spend` equals `tenant_spend` to the last digit, which is what the hidden tests check on 5,000 random windows over 30,000 records.

The index is a snapshot. Records written after it is built are not in it, and nothing in the code says so. In production it needs a build time the dashboard shows, a rebuild trigger, and a watermark for late records: either rebuild the affected tenant's lists or keep a small unsorted tail of late records that every query scans.

On the assistant: phase 2 and phase 3 are the parts where generated code is plausible and wrong in the details (a closed window instead of half-open, a float accumulator, a tie order that depends on dict order). The invariant tests are the check; run them on whatever the assistant writes before reading it closely.

## Follow-ups

- *The table grows past what one process can hold. Which questions do you stop answering exactly, and who has to agree to that in writing?* Keep exact answers for invoice windows, which are month-aligned and can be precomputed; move dashboard windows to hourly buckets; finance signs off on the dashboard's precision.
- *What would you need to see to move this out of Python into a daily rollup table in the warehouse?* Scan cost of the rollup job against the dashboard's per-load cost, and how often windows that do not align to days are actually asked for.
- *Two engineers disagree: one wants prefix sums for exactness, the other buckets for memory. How do you decide without deciding by rank?* Log a week of dashboard queries and count how many windows fall on bucket edges; if nearly all do, buckets win.
- *How does this change in a real engine?* The gateway writes records from many hosts, so timestamps arrive out of order and some arrive twice; the index needs a dedupe key per call (the provider's response id) and a lateness bound after which a correction goes on the next invoice instead.

## Common mistakes

- Fixing phase 1 by dropping the cached term, which bills cached tokens at the full rate.
- Treating the window as closed, which counts a record at `end` in two adjacent windows.
- Building the index on `float` and comparing to the direct sum with a tolerance, which hides an invoice-level difference.
- Caching `tenant_spend` results per (tenant, start, end), which still rescans for every new window.
- Breaking top-k ties by dict order, so the dashboard reorders tenants between loads.
