      class SyncError(Exception):
          """Every error filesync raises on bad input: callers catch this one type."""
    filesync/hashing.py: |
      """All hashing in filesync goes through a Hasher, so tests and metrics can count it.

      Leaf and interior hashes use different one-byte prefixes, as in RFC 6962, so a leaf can never be passed off as an
      interior node. Digests are raw bytes inside the package and hex only at the edges (manifest.py)."""
      import hashlib


      class Hasher:
          def __init__(self):
              self.calls = 0

          def leaf(self, data):
              self.calls += 1
              return hashlib.sha256(b"\x00" + data).digest()

          def node(self, left, right):
              self.calls += 1
              return hashlib.sha256(b"\x01" + left + right).digest()
    filesync/manifest.py: |
      """A manifest is the sorted list of (path, content digest) for a workspace, committed to by one root hash."""
      import hashlib

      from filesync.errors import SyncError
      from filesync.hashing import Hasher
      from filesync.merkle import MerkleTree, verify


      def _leaf(path, content):
          return path.encode("utf-8") + b"\x00" + hashlib.sha256(content).digest()


      class Manifest:
          def __init__(self, files, hasher=None):
              """files maps a relative path to its content as bytes."""
              if not files:
                  raise SyncError("a manifest needs at least one file")
              self.hasher = hasher or Hasher()
              self.paths = sorted(files)
              self.tree = MerkleTree([_leaf(p, files[p]) for p in self.paths], self.hasher)

          def root_hex(self):
              return self.tree.root.hex()

          def prove(self, path):
              """Proof that path is in this manifest, for a remote that only knows root_hex()."""
              try:
                  i = self.paths.index(path)
              except ValueError:
                  raise SyncError(f"{path} is not in the manifest") from None
              return self.tree.proof(i)

          def replace(self, path, content):
              """Record a new version of one file."""
              try:
                  i = self.paths.index(path)
              except ValueError:
                  raise SyncError(f"{path} is not in the manifest") from None
              self.tree.update(i, _leaf(path, content))


      def check(path, content, proof, root_hex, hasher=None):
          """True when (path, content) is the file the manifest with root root_hex committed to."""
          return verify(hasher or Hasher(), _leaf(path, content), proof, bytes.fromhex(root_hex))

  solution: |
    """The hash tree behind Manifest. Follow the package conventions: hash only through the Hasher you are given,
    keep digests as bytes, and raise SyncError for bad input."""
    from filesync.errors import SyncError


    class MerkleTree:
        """A binary hash tree over a non-empty list of leaves (bytes), in order.

        levels[0] holds the leaf digests and levels[-1] the root. An odd node at the end of a level moves up unchanged
        instead of being paired with a copy of itself, so two different leaf lists cannot share a root, and the shape
        matches RFC 6962's split at the largest power of two."""

        def __init__(self, leaves, hasher):
            if not leaves:
                raise SyncError("a hash tree needs at least one leaf")
            self.hasher = hasher
            level = [hasher.leaf(x) for x in leaves]
            self.levels = [level]
            while len(level) > 1:
                nxt = [hasher.node(level[i], level[i + 1]) for i in range(0, len(level) - 1, 2)]
                if len(level) % 2:
                    nxt.append(level[-1])
                self.levels.append(nxt)
                level = nxt

        def __len__(self):
            return len(self.levels[0])

        @property
        def root(self):
            """The root digest (bytes)."""
            return self.levels[-1][0]

        def _check(self, index):
            if not isinstance(index, int) or isinstance(index, bool) or not 0 <= index < len(self):
                raise SyncError(f"leaf index {index!r} is outside 0..{len(self) - 1}")

        def proof(self, index):
            """Sibling digests from the leaf up to the root, as (side, digest) pairs: side "L" when the sibling sits
            to the left of the running hash, "R" when it sits to the right."""
            self._check(index)
            out = []
            for level in self.levels[:-1]:
                sib = index ^ 1
                if sib < len(level):
                    out.append(("L" if sib < index else "R", level[sib]))
                index //= 2
            return out

        def update(self, index, leaf):
            """Replace one leaf and bring the root up to date: one leaf hash and one node hash per level."""
            self._check(index)
            h = self.hasher.leaf(leaf)
            self.levels[0][index] = h
            for depth in range(len(self.levels) - 1):
                level = self.levels[depth]
                sib = index ^ 1
                if sib < len(level):
                    h = self.hasher.node(level[sib], h) if sib < index else self.hasher.node(h, level[sib])
                index //= 2
                self.levels[depth + 1][index] = h


    def verify(hasher, leaf, proof, root):
        """True when leaf, combined with proof, hashes to root."""
        h = hasher.leaf(leaf)
        for step in proof:
            if not (isinstance(step, tuple) and len(step) == 2 and isinstance(step[1], bytes)):
                return False
            side, sib = step
            if side == "L":
                h = hasher.node(sib, h)
            elif side == "R":
                h = hasher.node(h, sib)
            else:
                return False
        return h == root

## Question

<Callout kind="interview" label="The round, as a candidate report describes it">
"Can you implement this hash tree in the repo?" The repository is real, cloned to your machine, and you work inside it. A candidate report describes this round at Cursor; the repository below is original.
</Callout>

`filesync` compares file sets between a laptop and a remote workspace. `Manifest` in `manifest.py` already calls `MerkleTree` and `verify`; `merkle.py` is a stub with the interface `Manifest` expects. Read `hashing.py` and `errors.py` before writing anything: the package has conventions, and the tests hold you to them.

Implement `MerkleTree(leaves, hasher)` with `root`, `__len__`, `proof(index)` and `update(index, leaf)`, and the function `verify(hasher, leaf, proof, root)`. The workspace has tens of thousands of files and editors save often, so `update` must not rebuild the tree. Decide what happens when a level has an odd number of nodes, and say why.

## Hints

<details class="hint"><summary>Hint 1</summary><p>Ask where the leaf and interior hashing is supposed to happen in this package, and what `Hasher` counts.</p></details>
<details class="hint"><summary>Hint 2</summary><p>Ask whether two different file lists may ever produce the same root, and what the remote does when they do.</p></details>
<details class="hint"><summary>Hint 3</summary><p>Ask how many hash calls one save is allowed to cost on a workspace of 16,000 files.</p></details>

## Answer

**The conventions.** All hashing goes through the `Hasher` passed in, which prefixes leaves with `0x00` and interior nodes with `0x01`, as RFC 6962 does, and counts calls. Calling `hashlib` directly passes the visible tests and breaks two things: the domain separation that stops a leaf being passed off as an interior node, and the call count the package uses for metrics and the tests use for the update budget. Digests stay `bytes` inside the package and become hex only in `manifest.py`. Bad input raises `SyncError`, including a negative index, which Python lists would otherwise accept and quietly treat as counting from the end.

**The structure.** Keep every level: `levels[0]` holds the leaf digests, each next level pairs neighbours with `hasher.node`, and `levels[-1][0]` is the root. Building costs n leaf hashes and n - 1 node hashes, 2n - 1 in all. `proof(i)` walks up the levels and takes the sibling `i ^ 1` when it exists, tagged `"L"` or `"R"`. `update(i, leaf)` rehashes the leaf and then one node per level, so a save costs at most 1 + ceil(log2 n) hash calls; for 16,384 + 3 = 16,387 files that is 1 + 15 = 16, against 32,773 for a rebuild. Storing every level costs about 2n digests of 32 bytes, which is memory the reader trades for that update cost; the hidden tests count calls, so the rebuild fails them.

**The odd node.** Three choices, and only two hold up.

- *Duplicate the last node* and pair it with itself. Bitcoin did this, and its source documents the result as CVE-2012-2459: `[1,2,3,4,5,6]` and `[1,2,3,4,5,6,5,6]` produce the same root. Here that means a manifest with two extra files can match the root of one without them, and the remote would accept it. A hidden test checks exactly those two lists.
- *Promote the last node* to the next level unchanged. No hash is spent on it, and the shape it produces is the same as RFC 6962's split at the largest power of two below n.
- *Split as RFC 6962 does*, recursively. Same roots as promotion; recursion is the natural form if the tree is built top-down.

Promotion and the RFC split give the same answer, so the choice between them is about code shape. Duplication is the one that is wrong.

**What can still go wrong.** `verify` accepts any proof whose hashes chain to the root, and the proof format carries sides but not the leaf's index. A remote that needs to know *where* a file sits, not only that it is in the manifest, has to bind the index: check that the sequence of sides matches the binary form of the index for the tree's size, or include the index and tree size in the proof as RFC 6962's audit paths do. The reference solution does not, because `Manifest` puts the path inside every leaf, so the leaf itself says where it belongs.

## Follow-ups

- *A save touches 300 files at once (a branch switch). Do you call `update` 300 times or rebuild?* 300 updates cost about 300 × 16 = 4,800 calls on 16,387 files, a rebuild about 32,773; batched updates that rehash each touched parent once cost less than either. The crossover depends on how clustered the changed indices are.
- *Inserting a new file shifts every later index, so `update` does not cover it. What changes?* Sorted paths make an insert a rebuild from the insertion point; a tree keyed by path hash, or a Merkle search tree, keeps inserts local at the cost of a more complex structure.
- *How do two workspaces find which files differ without sending the whole manifest?* Exchange roots, then descend only into subtrees whose hashes differ; the cost is proportional to the number of changed files times the tree depth.
- *What changes when this runs against a real file watcher and a remote over the network?* Events arrive out of order and in bursts, so updates are applied per batch with the manifest version they apply to, and the remote compares roots only for the same version; the levels move to disk or are rebuilt at start-up, since holding about 2n digests per workspace is the memory cost.

## Common mistakes

- Hashing with `hashlib` directly, which drops the leaf and node prefixes.
- Duplicating the odd node, which lets two different file lists share a root.
- Rebuilding the whole tree in `update`.
- Accepting a negative index in `proof` or `update` instead of raising `SyncError`.
- Returning hex digests from `root`, which breaks `Manifest.root_hex()`.
