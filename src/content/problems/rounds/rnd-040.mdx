    import heapq
    import time
    from collections import OrderedDict


    class LRUCache:
        """The existing module, in production for a year: an LRU cache with no expiry."""

        def __init__(self, capacity):
            self.capacity = capacity
            self.data = OrderedDict()

        def get(self, key, default=None):
            if key not in self.data:
                return default
            self.data.move_to_end(key)
            return self.data[key]

        def put(self, key, value):
            self.data[key] = value
            self.data.move_to_end(key)
            if len(self.data) > self.capacity:
                self.data.popitem(last=False)

        def __len__(self):
            return len(self.data)


    class AssistantTTLCache(LRUCache):
        """The assistant's patch for "add a per-entry TTL to LRUCache, with an injectable clock". Kept as written."""

        def __init__(self, capacity, ttl, clock=time.monotonic):
            super().__init__(capacity)
            self.ttl, self.clock = ttl, clock
            self.expires = {}

        def _purge(self):
            now = self.clock()
            for key in list(self.expires):
                if now > self.expires[key]:
                    del self.expires[key]
                    del self.data[key]

        def get(self, key, default=None):
            self._purge()
            return super().get(key, default)

        def put(self, key, value, ttl=None):
            self._purge()
            super().put(key, value)
            self.expires[key] = self.clock() + (ttl if ttl is not None else self.ttl)


    class TTLCache:
        """LRU order in an OrderedDict, expiry order in a heap of (expires_at, seq, key).

        A heap entry is stale once its key was re-put or evicted; the seq stored beside the value tells
        live from stale. Every operation first pops what has expired, so expired entries never hold
        capacity. The heap is rebuilt from the live entries when stale ones outnumber them, which keeps
        memory within a constant multiple of capacity.
        """

        def __init__(self, capacity, ttl, clock=time.monotonic):
            self.capacity, self.ttl, self.clock = capacity, ttl, clock
            self.data = OrderedDict()   # key -> (value, expires_at, seq)
            self.heap = []
            self.seq = 0

        def _expire(self, now):
            heap, data = self.heap, self.data
            while heap and heap[0][0] <= now:
                _, seq, key = heapq.heappop(heap)
                entry = data.get(key)
                if entry is not None and entry[2] == seq:
                    del data[key]

        def _compact(self):
            if len(self.heap) > 2 * len(self.data) + 64:
                self.heap = [(exp, seq, key) for key, (_, exp, seq) in self.data.items()]
                heapq.heapify(self.heap)

        def get(self, key, default=None):
            self._expire(self.clock())
            entry = self.data.get(key)
            if entry is None:
                return default
            self.data.move_to_end(key)
            return entry[0]

        def put(self, key, value, ttl=None):
            now = self.clock()
            self._expire(now)
            self.seq += 1
            expires_at = now + (self.ttl if ttl is None else ttl)
            self.data[key] = (value, expires_at, self.seq)
            self.data.move_to_end(key)
            heapq.heappush(self.heap, (expires_at, self.seq, key))
            if len(self.data) > self.capacity:
                self.data.popitem(last=False)   # nothing has expired, so the least recently used live entry goes
            self._compact()

        def __len__(self):
            self._expire(self.clock())
            return len(self.data)


    def expiry_suite(make):
        now = [0.0]

        # The edge: an entry put at t with ttl d is live while clock < t + d and gone at t + d.
        c = make(4, 10, lambda: now[0])
        now[0] = 0.5
        c.put("a", 1)
        now[0] = 10.25
        assert c.get("a") == 1, "live just before t + ttl"
        now[0] = 10.5
        assert c.get("a") is None, "gone at exactly t + ttl"

        # A per-entry ttl overrides the default; len counts only live entries.
        now[0] = 0.0
        c = make(4, 10, lambda: now[0])
        c.put("short", 1, ttl=2)
        c.put("long", 2)
        now[0] = 2.0
        assert len(c) == 1 and c.get("short") is None and c.get("long") == 2

        # Reading does not extend the expiry.
        now[0] = 0.0
        c = make(4, 10, lambda: now[0])
        c.put("a", 1)
        for t in (3, 6, 9):
            now[0] = float(t)
            assert c.get("a") == 1
        now[0] = 10.0
        assert c.get("a") is None, "reads must not slide the expiry"

        # Writing again resets the expiry.
        now[0] = 0.0
        c = make(4, 10, lambda: now[0])
        c.put("a", 1)
        now[0] = 8.0
        c.put("a", 2)
        now[0] = 12.0
        assert c.get("a") == 2, "a re-put starts a new ttl"
        now[0] = 18.0
        assert c.get("a") is None

        # An expired entry is removed before a live one is evicted, even when the expired one was used more recently.
        now[0] = 0.0
        c = make(2, 100, lambda: now[0])
        c.put("old", 1, ttl=5)
        c.put("live", 2)
        now[0] = 1.0
        assert c.get("old") == 1
        now[0] = 6.0
        c.put("new", 3)
        assert c.get("live") == 2 and c.get("new") == 3, "an expired entry held capacity"

        # Capacity still holds with nothing expired.
        now[0] = 0.0
        c = make(2, 100, lambda: now[0])
        c.put("a", 1); c.put("b", 2); c.put("c", 3)
        assert len(c) == 2 and c.get("c") == 3

<Callout kind="interview" label="AI-assisted round, 45 minutes">
"Here is `cache.py`, an LRU cache that has been in production for a year. Add a per-entry TTL with the assistant. When you are done, I want tests that would catch an off-by-one in expiry." `LRUCache` in the editor is the existing module. `AssistantTTLCache` is the patch the assistant proposed; it passes the two tests the assistant wrote for it, which put a key, advance the clock past the TTL and check the key is gone.

**The rules for `TTLCache(capacity, ttl, clock=time.monotonic)`.**

- `put(key, value, ttl=None)` stores the value with an expiry of `clock() + ttl`, or the default `ttl`. Writing a key again replaces its value and its expiry and makes it the most recently used.
- `get(key, default=None)` returns the value while `clock()` is before the expiry and `default` from the expiry onwards. An entry put at t with ttl d is live for clock values below `t + d` and gone at `t + d`. A hit makes the key the most recently used and does not extend its expiry.
- `len(cache)` is the number of live entries.
- At most `capacity` live entries. When a put needs room, expired entries go first. If none has expired, evict one live entry other than the one just written: either the least recently used or the one that expires soonest. Choose, and say why.

**Write two things.**

1. `TTLCache`, starting from the patch or from scratch.
2. `expiry_suite(make)`. `make(capacity, ttl, clock)` returns an empty cache. Raise `AssertionError` when the cache breaks the rules above; return `None` when it keeps them. It runs against the assistant's patch, against correct caches built differently from yours (one of which makes the other eviction choice), and against caches that are wrong in ways an assistant plausibly writes. Use only `put`, `get` and `len`.

**Operating constraints.** The cache runs in a request path with tens of thousands of entries and mixed TTLs; the hidden tests time 150,000 operations against a 50,000-entry cache. The cache's own containers together hold at most `4 * capacity + 64` items whatever the sequence of operations, including a key rewritten 200,000 times.

<Fragment slot="short">
<p>The patch has four problems. The comparison `now > expires` keeps an entry readable at exactly its expiry, which is the off-by-one the interviewer asked about. The purge walks every key on every call, so each operation is linear in the cache size. A key evicted for capacity keeps its entry in `expires`, so when its time passes the purge deletes it from `data` again and raises `KeyError`, taking the cache down on an ordinary put. And `len` counts expired entries. The fix keeps LRU order in the `OrderedDict` and expiry order in a heap with lazy deletion, pops expired entries at the start of every operation, and compacts the heap when stale entries outnumber live ones. For eviction among live entries I keep LRU, because the cache sits in front of reads where recency predicts reuse; evicting the soonest-to-expire entry is the better choice when TTLs encode how much the caller still wants each entry.</p>
</Fragment>

### The suite

Each case targets a wrong cache an assistant plausibly writes, and none depends on which live entry a full cache evicts:

- **The edge, from a fractional start.** Put at 0.5 with a TTL of 10; live at 10.25, gone at 10.5. Kills `>` and kills truncation to whole seconds.
- **Per-entry TTL and `len`.** A short and a long entry; after the short one expires, `len` is 1.
- **Reads do not slide the expiry.** Read at 3, 6 and 9; gone at 10.
- **A rewrite resets it.** Put at 0, put again at 8; live at 12, gone at 18.
- **Expired entries never hold capacity.** Capacity 2: put an entry with a TTL of 5 and one with a TTL of 100, read the first at 1 so it is the most recently used, then put a third at 6. The long-lived entry must survive. A cache that expires entries only when they are read evicts it.
- **Capacity with nothing expired.** Three puts into capacity 2 leave two entries, including the newest.

The suite uses halves and quarters for clock values so that every comparison is exact in binary floating point.

### The fix and the decision

The reference stores `(value, expires_at, seq)` per key in the `OrderedDict` and pushes `(expires_at, seq, key)` onto a heap. A heap entry whose `seq` no longer matches the key's current entry is stale, left by a rewrite or an eviction, and is skipped when popped. That is the lazy deletion the `heapq` documentation describes. Every operation pops expired entries off the top first, so expired entries cost amortised O(log n) to remove, never a scan. When the heap holds more than `2 * len(data) + 64` entries it is rebuilt from the live entries, which bounds memory at roughly three times capacity.

The decision is the eviction order among live entries, and both choices are defensible. LRU (the reference) protects keys that are read often; its cost is that an entry with hours left can be evicted while one with seconds left survives. Soonest-expiry, Redis's `volatile-ttl`, loses the least remaining lifetime; its cost is that a hot key written with a short TTL is evicted first, and a full cache that sees no reads keeps evicting the newest short-lived writes. The quantity you choose is the compaction threshold. A factor of 2 over the live count keeps rebuilds amortised; a factor near 1 rebuilds on nearly every rewrite of a hot key, and no threshold at all lets one key rewritten in a loop grow the heap without bound, which the hidden memory test checks.

**The constraint conflict and what the reference gives up.** Reclaiming expired entries eagerly costs time on every call; reclaiming them lazily costs memory and capacity. The reference pays the time, bounded by the number of entries that actually expired. Its failure mode is a burst of expiries: if 40,000 entries were written in the same second with the same TTL, the first operation after that second pops all of them, and that one request pays a pause proportional to the burst. If tail latency matters more than reclaiming promptly, cap the pops per operation and let `len` and capacity checks tolerate a bounded number of expired entries, which the rules here do not allow.

## Follow-ups

- *The clock.* `time.monotonic` is right in one process. Expiry shared across processes needs wall time, and wall time can go backwards; decide whether an entry can outlive its TTL after a clock step.
- *Thread safety.* `get` mutates both the order and the heap, so it needs the same lock as `put`.
- *How would you have prompted the assistant?* State the boundary rule and the memory bound in the prompt, ask for the boundary tests first, and read the diff against the existing `popitem` call.
- *Run it in Redis.* Redis already stores an absolute expiry per key and expires keys both on access and by sampling keys with an expiry in the background, so the in-process heap disappears. `SET` clears a key's timeout while `INCR` keeps it, which is the rewrite rule above made explicit per command; check every write path. Since Redis 2.6 the expiry error is 0 to 1 milliseconds.

## Common mistakes

- Testing only "after the TTL has passed", which passes both sides of the off-by-one.
- Accepting the patch because each changed line reads correctly, without asking what the old module does to state the patch added.
- A heap with no compaction, so a key rewritten in a loop grows the heap forever.
- Purging on `get` only, so expired entries hold capacity and push out live ones.
- Writing a suite that asserts which live entry was evicted, which fails a correct cache that made the other choice.
