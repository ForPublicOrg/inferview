    import json


    class APIError(Exception):
        """An HTTP error from the model API: status code, retry-after in seconds (or None), and
        error.details.error_code from the body (or None)."""

        def __init__(self, status, retry_after=None, code=None):
            super().__init__(f"HTTP {status}")
            self.status, self.retry_after, self.code = status, retry_after, code


    class Crash(BaseException):
        """Process death in the tests. A BaseException, so `except Exception` cannot swallow it."""


    class Checkpoint:
        """Progress for one batch job. put() buffers a record in memory; flush() rewrites the whole checkpoint
        file, so one flush costs time proportional to every record stored so far. A crash loses unflushed records."""

        def __init__(self, durable=None, crash_after_puts=None):
            self.durable = dict(durable or {})
            self.pending = {}
            self.puts = self.flushes = 0
            self.crash_after_puts = crash_after_puts
            self.file_text = json.dumps(self.durable)

        def get(self, doc_id):
            return self.pending.get(doc_id, self.durable.get(doc_id))

        def put(self, doc_id, record):
            self.puts += 1
            if self.crash_after_puts is not None and self.puts > self.crash_after_puts:
                raise Crash()
            self.pending[doc_id] = record

        def flush(self):
            self.durable.update(self.pending)
            self.pending = {}
            self.file_text = json.dumps(self.durable, sort_keys=True)   # the whole file, every time
            self.flushes += 1

        def restart(self):
            """The checkpoint a new process sees after a crash: flushed records only."""
            return Checkpoint(self.durable)


    # How many records to write between flushes. Choose it and say why.
    FLUSH_EVERY = None

        error.details.error_code from the body (or None)."""

            self.pending = {}
            self.file_text = json.dumps(self.durable, sort_keys=True)   # the whole file, every time
            self.flushes += 1

        def restart(self):
            """The checkpoint a new process sees after a crash: flushed records only."""
            return Checkpoint(self.durable)


    # Records written between flushes. A crash repeats at most this many model calls; each flush rewrites the
    # whole checkpoint. 500 keeps a 50,000-document job to 100 flushes and the crash rework to 500 calls.
    FLUSH_EVERY = 500

    RETRY_STATUS = {408, 429, 500, 502, 503, 504, 529}
    STOP_STATUS = {401, 403}          # every later item fails the same way


    class _Abort(Exception):
        pass


    def _classify(exc):
        """'retry', 'stop' or 'fail' for one failed attempt."""
        if isinstance(exc, (TimeoutError, ConnectionError)):
            return "retry"
        if isinstance(exc, APIError):
            if exc.status == 429 and exc.code == "enforced_spend_limit_reached":
                return "stop"         # no retry-after; retrying fails until the spend cap resets
            if exc.status in STOP_STATUS:
                return "stop"
            if exc.status in RETRY_STATUS:
                return "retry"
        return "fail"                 # 400, 404, 413, a bug in our own parsing: retrying will not help


    def _one(doc, call, sleep, rng, max_retries, base, cap):
        for attempt in range(max_retries + 1):
            try:
                return {"status": "ok", "result": call(doc)}
            except Exception as exc:                     # Crash is a BaseException and passes through
                kind = _classify(exc)
                if kind == "stop":
                    raise _Abort(f"{type(exc).__name__}: {exc}")
                if kind == "fail":
                    return {"status": "failed", "retryable": False, "error": f"{type(exc).__name__}: {exc}"}
                if attempt == max_retries:
                    return {"status": "failed", "retryable": True, "error": f"gave up after {attempt + 1} attempts"}
                after = getattr(exc, "retry_after", None)
                sleep(after if after is not None else rng.uniform(0, min(cap, base * 2 ** attempt)))


    def run_batch(items, call, checkpoint, sleep, rng, max_retries=3, base=0.5, cap=8.0):
        summary = {"ok": 0, "failed": 0, "skipped": 0, "aborted": None}
        unflushed = 0
        try:
            for doc_id, doc in items:
                done = checkpoint.get(doc_id)
                if done is not None and (done["status"] == "ok" or not done.get("retryable")):
                    summary["skipped"] += 1
                    continue
                record = _one(doc, call, sleep, rng, max_retries, base, cap)
                checkpoint.put(doc_id, record)
                summary["ok" if record["status"] == "ok" else "failed"] += 1
                unflushed += 1
                if unflushed >= FLUSH_EVERY:
                    checkpoint.flush()
                    unflushed = 0
        except _Abort as stop:
            summary["aborted"] = str(stop)
        if unflushed or summary["aborted"]:
            checkpoint.flush()
        return summary

## Question

<Callout kind="interview" label="The prompt">
"We need to run a classification prompt over 50,000 documents. Write the runner. It will hit rate limits and timeouts, a few documents are malformed, and the process will be killed at some point. Rerunning it must not pay for finished work twice."
</Callout>

Write `run_batch(items, call, checkpoint, sleep, rng, ...)`. The arguments are:

- **`items`.** An iterable of `(doc_id, document)`, possibly a generator.
- **`call(document)`.** The model call. It raises `APIError` (with the HTTP status, `retry_after` in seconds when the response had a `retry-after` header, and the `error.details.error_code` from the body when present), `TimeoutError`, `ConnectionError`, or anything else.
- **`checkpoint`.** It stores one record per document. `put` buffers a record; `flush` rewrites the whole checkpoint file, so one flush costs time proportional to everything stored so far. A crash loses whatever was not flushed.

The runner has to meet five rules:

- Classify each failure with the Claude API's documented error behaviour.
- Retry what can succeed later, with capped exponential backoff and jitter, or exactly `retry_after` when the server gives it.
- Record what will not change on retry, and keep going.
- Stop the whole batch when every later document would fail the same way.
- On a rerun with the same checkpoint, never call the model again for a document that already has a final record.

`Crash` derives from `BaseException`, so it passes through `except Exception`.

Choose `FLUSH_EVERY`, the number of records between flushes, and defend it. The visible tests crash a run partway and check that the rerun repeats at most `FLUSH_EVERY` calls, then push 6,000 documents through within a 2-second budget.

## Answer

The reference classifies each failed attempt as retry, stop or fail:

- **Retry.** `TimeoutError`, `ConnectionError`, and `APIError` with status 408, 429, 500, 502, 503, 504 or 529. The errors page documents 529 as `overloaded_error` and says to retry a 500 with backoff.
- **Stop.** A 429 whose `error_code` is `enforced_spend_limit_reached`, and 401 or 403. The rate-limits page says the spend-cap 429 has no `retry-after` header and keeps failing until access resumes, including through the SDK's own retries. A rejected key fails identically for every later document.
- **Fail.** Everything else: a 400, 404 or 413, and bugs in our own code such as a `KeyError` while parsing the output. These are recorded and not retried.

`_one` loops over attempts and returns a record, which is never raised past the loop, so one document cannot sink the batch. It sleeps `retry_after` when given, otherwise `rng.uniform(0, min(cap, base * 2 ** attempt))`, which is the full-jitter scheme from Marc Brooker's AWS post. It never sleeps after the last attempt. A stop raises an internal exception that `run_batch` turns into `aborted`, after flushing what it has.

On a rerun, a document whose record is `ok`, or a failure marked not retryable, is skipped. A document that ran out of transient retries is tried again, because the overload may have passed.

The flush interval is the number this problem turns on. Each flush rewrites everything stored, so with n documents and interval F, total serialisation work is about n² ÷ (2F) records:

| Documents | FLUSH_EVERY | Records serialised |
|---|---|---|
| 50,000 | 1 | about 1.25 billion |
| 50,000 | 500 | about 2.5 million |

In this harness's local CPython 3.12.6, `FLUSH_EVERY = 1` took 8.3 seconds for the visible test's 6,000 documents, and the reference's 500 took 0.019 seconds. On the other side of the trade, a crash repeats at most F calls, and each one costs a model call. Reasonable values run from a few hundred to a few thousand: a crash then wastes minutes of spend at most, and checkpoint cost stays negligible. One is unreasonable (quadratic cost); so is anything near n (a crash late in the run repeats nearly everything).

The production answer removes the quadratic term. Append one JSON line per record to a log, flush and `fsync` every F records, and on restart read the log and keep the last record per id. The interval then trades only crash rework against fsync rate.

Retry layering is the other trap. The errors page says the official SDK already retries transient failures twice by default, honouring `retry-after`. The AWS Builders' Library shows what stacked retries do: with retries at each layer of a five-deep stack, load on the bottom layer rises 243 times. If this runner owns the policy, construct the client with `max_retries=0`.

**How the recommended answer fails.** Retrying exhausted transient failures on every rerun means a document that always times out costs four calls on each run, forever. Store an attempt count in the record and stop after a run limit you choose. The guarantee is also at-least-once: a crash between `call` and `flush` repeats up to F calls, and nothing in a model API makes a repeated classification free.

## Follow-ups

- *Is the provider's batch API a better fit?* For an offline job, usually. The Claude Message Batches docs give a 50 percent price reduction, a limit of 100,000 requests or 256 MB per batch, a 24-hour expiry, and results in any order matched by `custom_id`. Errored, canceled and expired requests are not billed, and the docs separate `invalid_request_error`, which must be fixed before resending, from server errors, which can be retried as they are. The cost is latency: the docs say most batches finish within an hour, but a batch that has not finished within 24 hours expires, and its unprocessed requests come back as `expired`.
- *Add concurrency for the real client.* Wrap `_one` in an `asyncio.Semaphore`, and put a token bucket in front of `call` sized from your input-tokens-per-minute limit. The rate-limits page says limits are enforced with a token bucket and that short bursts can trip them, so request count alone is the wrong budget.
- *Pushback: finance wants each document billed exactly once. What do you promise?* At-least-once with a bound: at most F repeated calls per crash, stated in writing with the measured crash rate. A model call has no idempotency key that would make a repeat free, so "exactly once" would be a false promise.
- *The spend cap stops the batch at document 31,000 on a Friday night. What does the on-call see, and what happens on Monday?* A run summary with `aborted` and the error code, every finished record flushed, and an alert. On Monday someone raises the cap or waits for the reset, and reruns with the same checkpoint, which skips the 31,000.

## Common mistakes

- Treating every 429 the same, which burns retries against the spend cap.
- Retrying a 400, or a bug in your own parsing code.
- Catching `BaseException`, or letting `Crash` be swallowed.
- Flushing after every record against a checkpoint that rewrites everything.
- Never flushing before an abort.
- Keeping the SDK's default retries underneath your own.
- Skipping documents on rerun by position instead of by id.
