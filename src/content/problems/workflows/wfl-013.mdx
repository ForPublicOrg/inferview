    class Crash(BaseException):
        """Process death in the tests. It derives from BaseException so `except Exception` cannot swallow it."""


    class PaymentUnavailable(Exception):
        """Transient: the provider did not take the request. Safe to retry."""


    class PaymentRejected(Exception):
        """Permanent: the provider refused this refund. Never retry it."""

        clock     a function returning the current time in seconds. Never read the wall clock.
        """

        def __init__(self, store, payments, clock):
            raise NotImplementedError

            raise NotImplementedError

            """Record that the order was delivered now. The event can arrive late, and more than once."""
            raise NotImplementedError

        def tick(self):
            """Called by a scheduler about once a second. Refund what is due."""
            raise NotImplementedError

    from collections.abc import MutableMapping

    class Store(MutableMapping):
        """The durable store: values are kept as JSON text, so a change counts only once it is written back."""
        def __init__(self):
            self.data = {}
        def __getitem__(self, key):
            return json.loads(self.data[key])
        def __setitem__(self, key, value):
            self.data[key] = json.dumps(value, allow_nan=False)
        def __delitem__(self, key):
            del self.data[key]
        def __iter__(self):
            return iter(list(self.data))
        def __len__(self):
            return len(self.data)

    class Payments:
        """Deduplicates by idempotency key. Unavailable while clock() < down_until."""
        def __init__(self, clock=None, down_until=None, reject=()):
            self.by_key, self.calls, self.count = {}, 0, {}
            self.clock, self.down_until, self.reject = clock, down_until, set(reject)
            self.die_after_next_refund = False
        def refund(self, order_id, amount_cents, idempotency_key):
            self.calls += 1
            if order_id in self.reject:
                raise PaymentRejected(order_id)
            if self.down_until is not None and self.clock() < self.down_until:
                raise PaymentUnavailable("503")
            if idempotency_key in self.by_key:
                assert self.by_key[idempotency_key] == (order_id, amount_cents), "key reused with different parameters"
                return {"replayed": True}
            self.by_key[idempotency_key] = (order_id, amount_cents)
            self.count[order_id] = self.count.get(order_id, 0) + 1
            if self.die_after_next_refund:
                self.die_after_next_refund = False
                raise Crash("the process died after the provider applied the refund")
            return {"id": "re_%d" % len(self.by_key)}
        def refunds(self, order_id):
            return self.count.get(order_id, 0)

    now = [0]
    clock = lambda: now[0]

    store, pay = Store(), Payments()
    wf = RefundWorkflow(store, pay, clock)
    wf.place("A", 1250, 100)

    now[0] = 101
    wf.tick()

    import heapq
    import json


    class Crash(BaseException):
        """Process death in the tests. It derives from BaseException so `except Exception` cannot swallow it."""


    class PaymentUnavailable(Exception):
        """Transient: the provider did not take the request. Safe to retry."""


    class PaymentRejected(Exception):
        """Permanent: the provider refused this refund. Never retry it."""


    class RefundWorkflow:
        RETRY_FIRST = 30   # seconds after the first transient failure
        RETRY_MAX = 900    # cap on the backoff

        def __init__(self, store, payments, clock):
            self.store, self.payments, self.clock = store, payments, clock
            # (act_after, order_id): an order is due once the clock is strictly past act_after.
            # The heap lives in memory only; the store is the truth, so a restart rebuilds it in one pass.
            self.due = []
            for key in self.store:
                if key.startswith("order:"):
                    o = self.store[key]
                    if o["state"] in ("pending", "refunding"):
                        self.due.append((o["next"], o["id"]))
            heapq.heapify(self.due)

        def place(self, order_id, amount_cents, deadline):
            key = "order:" + order_id
            if key in self.store:
                return  # a redelivered event must not reset an order that has moved on
            self.store[key] = {"id": order_id, "amount": amount_cents, "deadline": deadline,
                               "state": "pending", "attempts": 0, "next": deadline}
            heapq.heappush(self.due, (deadline, order_id))

        def delivered(self, order_id):
            key = "order:" + order_id
            o = self.store.get(key)
            # Policy: a late delivery still cancels the refund, but only until the refund intent is written.
            # After that the provider may already have paid, so the refund is finished instead.
            if o is None or o["state"] != "pending":
                return
            o["state"] = "delivered"
            self.store[key] = o

        def tick(self):
            now = self.clock()
            while self.due and self.due[0][0] < now:
                when, order_id = heapq.heappop(self.due)
                key = "order:" + order_id
                o = self.store.get(key)
                if o is None or o["state"] not in ("pending", "refunding") or o["next"] != when:
                    continue  # delivered, finished, or a stale heap entry
                if o["state"] == "pending":
                    o["state"] = "refunding"
                    self.store[key] = o  # the intent, written before the external call
                try:
                    # The key depends only on the order, so every attempt, before or after a crash, is the same request.
                    self.payments.refund(order_id, o["amount"], "refund:" + order_id)
                except PaymentUnavailable:
                    o["attempts"] += 1
                    o["next"] = now + min(self.RETRY_MAX, self.RETRY_FIRST * 2 ** (o["attempts"] - 1))
                    self.store[key] = o
                    heapq.heappush(self.due, (o["next"], order_id))
                    continue
                except PaymentRejected:
                    o["state"] = "review"
                    self.store[key] = o
                    continue
                o["state"] = "refunded"
                self.store[key] = o

        def status(self, order_id):
            o = self.store.get("order:" + order_id)
            if o is None:
                raise KeyError(order_id)
            return "pending" if o["state"] == "refunding" else o["state"]

## Question

<Callout kind="interview" label="Interview question">
Build a workflow that refunds an order when it is not delivered by its deadline. It must be deterministic and safe to retry against an external payment API, and the tests will kill it at every boundary.
</Callout>

Write `RefundWorkflow(store, payments, clock)` with `place(order_id, amount_cents, deadline)`, `delivered(order_id)`, `tick()` and `status(order_id)`.

- `store` is the only thing that survives a crash. It keeps values as JSON, so a value you read is a copy: change it and write it back.
- `payments.refund(order_id, amount_cents, idempotency_key)` deduplicates by key, like the Stripe API: a repeated key returns the first response instead of paying again. It raises `PaymentUnavailable` (transient) or `PaymentRejected` (permanent).
- `clock()` is the only source of time. Delivery at or before the deadline is on time.
- A crash is a `Crash` exception, which derives from `BaseException`, raised at any write to the store or just before or after the provider applies a refund. After a crash the test builds a new `RefundWorkflow` on the same store. Events (`place`, `delivered`) are redelivered at least once; the scheduler calls `tick()` again later.
- `tick()` runs about once a second against tens of thousands of open orders, so its cost must not grow with the number of orders that are not due.

The outcome the tests hold you to, at every crash point: an order delivered on time is never refunded; an order that missed its deadline is refunded exactly once or, if the provider rejects it, ends in `review` without another call; `status` says `refunded` if and only if the provider paid once; a provider outage delays refunds without making `tick()` raise or calling the provider on every tick.

Two decisions are yours. What a delivery that arrives after the deadline, but before the refund, does to the refund. And the retry schedule while the provider is down.

<details class="hint"><summary>Hint 3</summary><p>How many orders does a tick read when nothing is due? Which structure can you rebuild from the store after a restart?</p></details>

## Answer

The reference solution keeps one JSON record per order in the store and a heap of `(act_after, order_id)` in memory. The heap is rebuilt from the store in the constructor, so a restart costs one pass over the open orders, and each tick pops only what is due: idle ticks cost nothing, which is what the performance test checks with 20,000 orders and 6,000 ticks under a 3-second budget.

The crash safety comes from three choices. The idempotency key is `refund:<order_id>`: it depends only on the order, so a retry after a lost response, or after a restart, is the same request, and the provider returns the first refund. Before calling the provider, the order moves to an internal `refunding` state and that intent is written. From then on a late `delivered` event is ignored, because the provider may already have paid; without the intent record, a crash after the payment followed by a late delivery leaves an order marked delivered and refunded at the same time, which is the case the hidden crash sweep is built to find. And every change is written back before the next external call, so the record on disk is never behind what the provider has done by more than one step, and that step is always safe to repeat.

The policy choices in the reference: a late delivery cancels the refund only while no refund has been attempted, and transient failures back off from 30 seconds, doubling to a 15-minute cap, with no limit on attempts. Ignoring every late delivery is also accepted by the tests and is simpler, at the cost of refunding customers who did receive their order. A limit on attempts that moves the order to `review` is also reasonable; the tests only require that a one-hour outage costs no more than 120 calls for one order and that the refund happens after the provider recovers.

Where this design fails. The heap holds every open order in memory, so a store of millions of open orders makes restarts slow and memory heavy; a real system keeps a due-time index in the database and queries the next due slice. The provider's deduplication is the last line of defence, and Stripe's v1 API may drop keys once they are at least 24 hours old: an order stuck in `refunding` through a longer outage needs a reconciliation read (has this order been refunded?) before it retries, or the key no longer protects it.

## Follow-ups

- What changes to run this in a durable execution engine such as Temporal or DBOS? A timer per order replaces the heap, the provider call becomes a step or Activity with the order-derived key, the intent record becomes the engine's journal, and the clock comes from the engine's replay-safe API.
- The provider's dashboard shows a refund your store has no record of. Which crash point produces that, and which job finds it?
- Partial refunds are coming: an order can be refunded twice for different items. What happens to the key, and to the "refunded once" invariant?
- *Pushback.* Operations says cancelling a refund on a late delivery generates support tickets from customers who were told a refund was coming, and asks you to ignore late deliveries. Finance says that pays for food that arrived. Which policy ships, and what does the losing side get in writing?
- *Reversal.* What would you need to see in production to replace the in-memory heap with a due-time index in the database?
- Finance wants a refund issued within 60 seconds of the deadline at the 99th percentile. Which part of this design sets that bound, and what do you measure?

## Common mistakes

- Generating the idempotency key per attempt, or from a random id that is not written before the call.
- Writing the outcome only after the provider call, with nothing on disk to show a refund may be in progress.
- Mutating the dict returned by the store and never writing it back.
- Scanning every order on each tick.
- Retrying in a loop inside `tick()` while the provider is down, which hammers it and blocks the scheduler.
- Catching `Exception` around the provider call and treating every failure as transient, so a rejected refund is retried forever.
