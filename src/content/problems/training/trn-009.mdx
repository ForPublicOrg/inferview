    class Machine:
        def __init__(self, mem):
            self.mem = list(mem)
            self.reg = [0] * NREGS
            self.cycles = 0

        def _r(self, r):
            if type(r) is not int or not 0 <= r < NREGS:
                raise ValueError("bad register %r" % (r,))
            return r

        def _a(self, addr):
            if type(addr) is not int or not 0 <= addr < len(self.mem):
                raise ValueError("bad address %r" % (addr,))
            return addr

        def run(self, program):
            for bundle in program:
                used = {"load": 0, "alu": 0, "store": 0}
                writes, stores = {}, {}
                for op in bundle:
                    kind = op[0]
                    cls = "alu" if kind in ALU else kind
                    if cls not in used:
                        raise ValueError("unknown op %r" % (op,))
                    used[cls] += 1
                    if used[cls] > SLOTS[cls]:
                        raise ValueError("more than %d %s ops in one cycle" % (SLOTS[cls], cls))
                    if kind == "store":
                        _, addr, src = op
                        stores[self._a(addr)] = self.reg[self._r(src)]
                        continue
                    if kind == "load":
                        _, dst, addr = op
                        val = self.mem[self._a(addr)]
                    else:
                        _, dst, x, y = op
                        a = self.reg[self._r(x)]
                        if kind == "add":
                            val = a + self.reg[self._r(y)]
                        elif kind == "xor":
                            val = a ^ self.reg[self._r(y)]
                        elif kind == "addi":
                            val = a + y
                        elif kind == "shl":
                            val = a << y
                        else:
                            val = a >> y
                    if self._r(dst) in writes:
                        raise ValueError("two writes to r%d in one cycle" % dst)
                    writes[dst] = val & MASK
                for r, v in writes.items():
                    self.reg[r] = v
                for addr, v in stores.items():
                    self.mem[addr] = v
                self.cycles += 1
            return self


    def build_kernel(n, a_base, b_base, out_base):
        # Baseline: one op per cycle, correct and slow.
        program = []
        for i in range(n):
            program.append([("load", 0, a_base + i)])
            program.append([("load", 1, b_base + i)])
            program.append([("xor", 0, 0, 1)])
            program.append([("addi", 0, 0, C1)])
            program.append([("shl", 1, 0, 12)])
            program.append([("xor", 0, 0, 1)])
            program.append([("shr", 1, 0, 19)])
            program.append([("xor", 0, 0, 1)])
            program.append([("addi", 0, 0, C2)])
            program.append([("store", out_base + i, 0)])
        return program

    NREGS = 32
    SLOTS = {"load": 2, "alu": 4, "store": 1}
    ALU = ("add", "xor", "addi", "shl", "shr")
    C1, C2 = 0x7ED55D16, 0x165667B1


    def reference(a, b):
        x = (a ^ b) & MASK
        x = (x + C1) & MASK
        x ^= (x << 12) & MASK
        x ^= x >> 19
        return (x + C2) & MASK

                        elif kind == "xor":
                            val = a ^ self.reg[self._r(y)]
                        elif kind == "addi":
                            val = a + y
                        elif kind == "shl":
                            val = a << y
                        else:
                            val = a >> y

    def build_kernel(n, a_base, b_base, out_base):
        # List scheduler. Each item is a chain of steps on its own register pair.
        # Every cycle, in-flight items issue their next step oldest first while
        # slots remain; new items start with the load slots left over.
        def steps_for(i, p, q):
            return [
                ("load", [("load", p, a_base + i), ("load", q, b_base + i)]),
                ("alu", [("xor", p, p, q)]),
                ("alu", [("addi", p, p, C1)]),
                ("alu", [("shl", q, p, 12)]),
                ("alu", [("xor", p, p, q)]),
                ("alu", [("shr", q, p, 19)]),
                ("alu", [("xor", p, p, q)]),
                ("alu", [("addi", p, p, C2)]),
                ("store", [("store", out_base + i, p)]),
            ]

        free = list(range(NREGS // 2 - 1, -1, -1))
        inflight = []
        program = []
        nxt = 0
        while nxt < n or inflight:
            used = {"load": 0, "alu": 0, "store": 0}
            bundle = []
            for job in inflight:
                cls, ops = job["steps"][job["pos"]]
                if used[cls] + len(ops) <= SLOTS[cls]:
                    used[cls] += len(ops)
                    bundle.extend(ops)
                    job["pos"] += 1
            started = []
            while nxt < n and free and used["load"] + 2 <= SLOTS["load"]:
                pair = free.pop()
                steps = steps_for(nxt, 2 * pair, 2 * pair + 1)
                bundle.extend(steps[0][1])
                used["load"] += 2
                started.append({"steps": steps, "pos": 1, "pair": pair})
                nxt += 1
            program.append(bundle)
            remaining = []
            for job in inflight:
                if job["pos"] == len(job["steps"]):
                    free.append(job["pair"])
                else:
                    remaining.append(job)
            inflight = remaining + started
        return program

## Question

<Callout kind="interview" label="Interview question">
"Here is a simulated machine and a correct kernel for it. Make the kernel take fewer cycles. The tests are fixed."
</Callout>

The machine executes a program that is a list of bundles. Each bundle is one cycle and holds up to 2 `load`, 4 ALU (`add`, `xor`, `addi`, `shl`, `shr`) and 1 `store` operations. There are 32 registers, `r0` to `r31`, of 32 bits. Every operation in a bundle reads register values from before the bundle; writes land at the end of the cycle, and two writes to one register in a bundle are an error. Addresses are immediates.

The kernel computes `out[i] = reference(a[i], b[i])` for `n` items: an XOR, then a short add-shift-xor hash. `build_kernel(n, a_base, b_base, out_base)` returns the program. The starter's version issues one operation per cycle and takes 10 cycles per item.

Rewrite `build_kernel` so that every `n` runs in at most `2n + 32` cycles. How the work is scheduled and which registers hold what is your decision. The hidden tests run your program on their own copy of the machine with its own reference function, on sizes and memory layouts you do not see, and check that nothing outside the output range is written. Changing `Machine`, `SLOTS`, `NREGS` or `reference` in the editor changes nothing they check.

## Hints

A list scheduler that gives each in-flight item a register pair and issues ready steps oldest first reaches 453 cycles at n = 256 (bound 448) and 1,755 at n = 1,000 (bound 1,750).</p>

<Fragment slot="long">
<p><b>The bound.</b> The baseline uses one of seven slots per cycle. Full bundles would be limited by the ALU: 7n / 4 = 1.75n cycles, against n for loads and n for stores. The target of 2n + 32 leaves room for the pipeline to fill and drain and for an imperfect schedule; a schedule that does not overlap items cannot meet it, because one item alone needs 9 cycles.</p>
<p><b>Register pressure against latency.</b> Each item's ALU steps form a chain: every step reads the previous step's result, so one item issues at most one ALU operation per cycle. Keeping 4 ALU slots busy needs several items at different points in their chains. Each item needs two registers (the running value and a temporary), so the 32 registers cap the number in flight at 16. A design that unrolls every item onto fresh registers runs out at item 17; a design that keeps too few items in flight leaves ALU slots empty.</p>
<p><b>Two defensible designs.</b> A list scheduler, as in the reference solution, models each item as a sequence of steps, keeps a pool of free register pairs, issues each in-flight item's next step oldest first while slots remain, and starts new items with leftover load slots. It adapts to any change in the operation mix or slot limits, at the cost of a generator that is harder to read and whose output is hard to inspect by eye. A hand-written software pipeline, a fixed stagger of items with a prologue and epilogue, is easier to verify and gives a predictable cycle count, but it is tied to this chain length and these limits, and the remainder when n does not divide the stagger is where its bugs live. With a correctness gate that tests odd sizes, either passes; choose the scheduler if the machine or the kernel is expected to change.</p>
<p><b>Failure modes of the reference.</b> Oldest-first greedy scheduling is not optimal in general; with operations of different latencies or a chain that branches, it can leave slots idle that a lookahead scheduler would fill. The program is fully unrolled with immediate addresses, so its length grows with n; a machine with bounded instruction memory needs a loop with address registers, which this machine does not provide. And the 2n + 32 target is a test, not the bound: the reference is within 5 cycles of 1.75n, and a reader who stops at 2n has left 12% on the table.</p>
</Fragment>

The reference solution keeps the machine and reference from the starter and replaces `build_kernel`. Each item's work is a list of steps tagged with its slot class: one step of two loads, seven ALU steps and one store. Free register pairs sit on a stack. Every cycle, the in-flight items are visited oldest first and each issues its next step if its slot class has room; then new items start while load slots remain and a register pair is free. An item that has issued its store returns its pair at the end of the cycle, which is safe because the store read its register during that cycle and a new item's loads cannot land before the end of the next.

## Follow-ups

- Run the same kernel with a hash that has one more ALU step. What is the new bound, and does the register budget still allow enough items in flight?
- The take-home's README reports that first-day submissions below a threshold had all modified the tests. How would you design a grader for an internal optimisation contest so that this cannot pass?
- Your scheduler's output for n = 100,000 is 175,000 bundles long. What do you change if the machine has room for 4,096 bundles of program?

## Common mistakes

- Packing several operations of one item into a bundle. They read pre-bundle values, so the chain breaks and the output is wrong.
- Starting every item at once on fresh registers. It runs past `r31` long before n = 256.
- Freeing a register pair in the same cycle its item's last ALU step writes, before the store has read the result.
- Editing `SLOTS`, `NREGS` or `Machine` to make the visible tests pass. The hidden tests run their own machine.
- Tuning for n = 256 only: the hidden tests include n = 0, 1 and sizes that are not multiples of any stagger.
