                  resp = self._page(accounts, params)
                  if resp.status == 200:
                      pairs = [f"{a['id']}={self.token(a['id'])}" for a in resp.body["data"]]
                      resp.headers["x-account-tokens"] = ",".join(pairs)
                  return resp
              if len(parts) == 4 and parts[:2] == ["v1", "accounts"] and parts[3] == "charges":
                  account = parts[2]
                  if account not in self.charges:
                      return Response(404, {"error": "no_such_account"})
                  if headers.get("x-account-token") != self.token(account):
                      return Response(401, {"error": "invalid_token"})
                  if method == "GET":
                      return self._page(list(reversed(self.charges[account])), params)
                  if method == "POST":
                      return self._create(account, headers, body)
              return Response(404, {"error": "not_found"})

          def _create(self, account, headers, body):
              key = headers.get("idempotency-key")
              if not key:
                  return Response(400, {"error": "idempotency_key_required"})
              if key in self.saved:
                  saved_body, saved = self.saved[key]
                  if saved_body != body:
                      return Response(400, {"error": "idempotency_key_reused"})
                  return Response(saved.status, dict(saved.body))
              ok = (isinstance(body, dict) and type(body.get("amount")) is int and body["amount"] > 0
                    and isinstance(body.get("currency"), str) and isinstance(body.get("order_id"), str))
              if not ok:
                  return Response(400, {"error": "invalid_charge"})
              charge = self.add_charge(account, body["amount"], body["currency"], body["order_id"])
              resp = Response(200, charge)
              self.saved[key] = (body, resp)
              if body["order_id"] in self.lost:
                  self.lost.discard(body["order_id"])
                  raise TimeoutError("the charge was made; the response was lost")
              return resp

      def _parse_tokens(header):
          tokens = {}
          for pair in header.split(","):
              account, sep, token = pair.strip().partition("=")
              if sep:
                  tokens[account] = token
          return tokens


      def account_tokens(api):
          """{account_id: token}, collected from the x-account-tokens header of every accounts page."""
          tokens = {}
          list_all(api, "/v1/accounts", on_page=lambda resp: tokens.update(_parse_tokens(resp.headers.get("x-account-tokens", ""))))
          return tokens


      def fetch_charges(api, account, tokens):
          """One account's charges. On a 401, refresh every token once (tokens is updated in place) and list again
          from the first page; a second 401 raises."""
          path = f"/v1/accounts/{account}/charges"
          try:
              return list_all(api, path, headers={"x-account-token": tokens.get(account, "")})
          except APIError as e:
              if e.status != 401:
                  raise
          tokens.clear()
          tokens.update(account_tokens(api))
          return list_all(api, path, headers={"x-account-token": tokens.get(account, "")})


      def charges_by_account(api, accounts=None):
          """{account_id: charges, newest first} for the given accounts, or every account."""
          tokens = account_tokens(api)
          wanted = sorted(tokens) if accounts is None else list(accounts)
          return {account: fetch_charges(api, account, tokens) for account in wanted}

      RETRY_STATUS = {429, 500, 502, 503, 504}
      MAX_ATTEMPTS = 4


      def send(api, method, path, params=None, headers=None, body=None):
          """The only place that retries. A POST without an idempotency key is sent once."""
          keyed = method != "POST" or bool((headers or {}).get("idempotency-key"))
          attempts = MAX_ATTEMPTS if keyed else 1
          for attempt in range(attempts):
              last = attempt == attempts - 1
              delay = min(4.0, 0.25 * 2 ** attempt)
              try:
                  resp = api.request(method, path, params=params, headers=headers, body=body)
              except TimeoutError:
                  if last:
                      raise
                  api.sleep(delay)
                  continue
              if resp.status not in RETRY_STATUS or last:
                  return resp
              if resp.status == 429:
                  try:
                      delay = float(resp.headers["retry-after"])
                  except (KeyError, ValueError):
                      pass
              api.sleep(delay)


      def create_charge(api, entry, tokens):
          """One POST through send, which owns the retries; a 401 refreshes the tokens once."""
          account = entry["account"]
          body = {"amount": entry["amount"], "currency": entry["currency"], "order_id": entry["order_id"]}
          for refreshed in (False, True):
              headers = {"x-account-token": tokens.get(account, ""), "idempotency-key": idempotency_key(entry["order_id"])}
              resp = send(api, "POST", f"/v1/accounts/{account}/charges", headers=headers, body=body)
              if resp.status == 401 and not refreshed:
                  tokens.clear()
                  tokens.update(account_tokens(api))
                  continue
              if resp.status != 200:
                  raise APIError(resp.status, resp.body)
              return resp.body


      def reconcile(api, ledger):
          """{order_id: status}. matched, mismatch and created as before; unknown when the account could not be
          listed or a create got no answer (the charge may exist); error when the provider refused the create."""
          tokens = account_tokens(api)
          found, unlisted = {}, set()
          for account in sorted({e["account"] for e in ledger}):
              try:
                  charges = fetch_charges(api, account, tokens)
              except (APIError, TimeoutError):
                  unlisted.add(account)           # without the list we cannot know what exists: create nothing here
                  continue
              for charge in charges:
                  found.setdefault((account, charge["order_id"]), charge)
          status = {}
          for entry in ledger:
              order, key = entry["order_id"], (entry["account"], entry["order_id"])
              if entry["account"] in unlisted:
                  status[order] = "unknown"
                  continue
              charge = found.get(key)
              if charge is not None:
                  same = (charge["amount"], charge["currency"]) == (entry["amount"], entry["currency"])
                  status[order] = "matched" if same else "mismatch"
                  continue
              try:
                  found[key] = create_charge(api, entry, tokens)
                  status[order] = "created"
              except TimeoutError:
                  status[order] = "unknown"
              except APIError as e:
                  status[order] = "unknown" if e.status in RETRY_STATUS else "error"
          return status

An integration round hands you an unfamiliar API and a sequence of parts that unlock in order. A candidate report from a Stripe onsite describes one such exercise with five parts, where completing three is "generally enough". This lab has five parts against `PaymentsAPI`, a deterministic payments service that runs in your browser. Read its docstring first, the way you would read the reference of an API you have never used.

The service behaves like the real thing in the ways that cost integrations money:

- **Cursor pagination.** Lists page by cursor and change while you page.
- **Rotating tokens.** Account tokens arrive only in response headers and rotate without warning.
- **Lost responses.** A charge can be made and its response lost.
- **Failures.** Failures arrive as 429s with `retry-after`, as 5xx responses, and as timeouts.

The idempotency rules follow Stripe's published behaviour: a repeated key with the same body returns the saved response, and the same key with a different body is an error.

Every request goes through `send`. In step 4 you replace `send` with a version that retries, and everything written in steps 1 to 3 picks it up without changes. The tests assert behaviour: the requests made, the keys sent, the charges that exist afterwards and the time spent sleeping. They never check prose.

Your code is saved with your progress as you type. Passing a step's tests unlocks the next step and counts toward your streak. Python runs in your browser, with the standard library only.
