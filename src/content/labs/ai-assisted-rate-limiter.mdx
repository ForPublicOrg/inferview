      class StoreUnavailable(Exception):
          """The shared store did not answer in time."""


      class MemoryStore:
          """Stands in for the store every server shares. get(key) returns (version, state), with state None for a
          key never written; put(key, state) writes blindly; cas(key, version, state) writes only if the key is still
          at that version and returns whether it did. Each call is one round trip, counted in calls. While down is
          True, every call raises StoreUnavailable."""

          def __init__(self):
              self.data, self.calls, self.down = {}, 0, False

          def _trip(self):
              if self.down:
                  raise StoreUnavailable("store timed out")
              self.calls += 1

          def get(self, key):
              self._trip()
              return self.data.get(key, (0, None))

          def put(self, key, state):
              self._trip()
              version, _ = self.data.get(key, (0, None))
              self.data[key] = (version + 1, state)

          def cas(self, key, version, state):
              self._trip()
              current, _ = self.data.get(key, (0, None))
              if current != version:
                  return False
              self.data[key] = (current + 1, state)
              return True

          def run(rate, burst, times, key="k"):
              now = [0.0]
              limiter = make(rate, burst, lambda: now[0])
              out = []
              for t in times:
                  now[0] = t

          # Exactly burst at one instant, then nothing.
          assert run(1, 3, [0, 0, 0, 0]) == [True, True, True, False], "burst, then deny"
          # A token comes back at exactly 1 / rate, not later.
          assert run(2, 1, [0, 0.5]) == [True, True], "refill at exactly 1 / rate"
          # Fractional refills accumulate across calls: a client polling faster than the rate still gets the rate.
          polls = [i * 0.25 for i in range(41)]

      class PausingStore(MemoryStore):
          """A MemoryStore that can run other servers' requests in the gap between a read and what the reader does
          with it. after_get(fn) queues fn; the next get reads its value, then runs everything queued, in order,

              super().__init__()
              self.queued = []

          def after_get(self, fn):
              self.queued.append(fn)

          def get(self, key):
              value = MemoryStore.get(self, key)
              queued, self.queued = self.queued, []
              for fn in queued:
                  fn()
              return value

      class Limiter:
          """Read with get, write with cas on the version read; on a lost race, read again and decide again."""

          def __init__(self, store, rate, burst, clock, attempts=4):
              self.store, self.rate, self.burst, self.clock = store, rate, burst, clock
              self.attempts = attempts

          def _decide(self, state, now):
              tokens, last = state if state is not None else (float(self.burst), now)
              # A server whose clock is behind sees negative elapsed time: no refill, and last never moves back.
              tokens = min(self.burst, tokens + max(0.0, now - last) * self.rate)
              last = max(last, now)
              if tokens >= 1:
                  return (tokens - 1, last), True
              return (tokens, last), False

          def allow(self, key):
              for _ in range(self.attempts):
                  now = self.clock()
                  version, state = self.store.get(key)
                  new_state, ok = self._decide(state, now)
                  if self.store.cas(key, version, new_state):
                      return ok
              return False   # lost every race: deny rather than admit a request the store never counted

      from collections import OrderedDict


      class ResilientLimiter(Limiter):
          """Limiter plus a decision for when the store does not answer. In "local" mode each server keeps its own
          bucket per key with 1/servers of the rate and of the burst (at least one token), so the fleet admits about
          the global limit while every server sees an even share of the traffic."""

          def __init__(self, store, rate, burst, clock, servers=1, outage="local", max_local=10_000):
              super().__init__(store, rate, burst, clock)
              self.servers, self.outage, self.max_local = servers, outage, max_local
              self.local = OrderedDict()   # key -> (tokens, last), least recently used first

          def _local(self, key):
              now = self.clock()
              rate, burst = self.rate / self.servers, max(1.0, self.burst / self.servers)
              tokens, last = self.local.pop(key, (burst, now))
              tokens = min(burst, tokens + max(0.0, now - last) * rate)
              ok = tokens >= 1
              self.local[key] = (tokens - 1 if ok else tokens, max(last, now))
              if len(self.local) > self.max_local:
                  # Dropping a bucket forgets what that key spent; a returning key starts full again.
                  self.local.popitem(last=False)
              return ok

          def allow(self, key):
              try:
                  return Limiter.allow(self, key)
              except StoreUnavailable:
                  if self.outage == "closed":
                      return False
                  return self._local(key)


      def worst_case_admits(servers, rate, burst, seconds, outage):
          if outage == "closed":
              return 0
          per_server = max(1.0, burst / servers) + (rate / servers) * seconds
          return servers * per_server
