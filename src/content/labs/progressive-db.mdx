      class Table:
          """Rows keyed by a string id; each row is a dict over a fixed set of typed columns."""

          def __init__(self, columns, **options):
              for name, kind in columns.items():
                  if kind not in (int, float, str):
                      raise TypeError(f"column {name!r}: types are int, float and str")
              self.columns = dict(columns)
              self.rows = {}

          def _check(self, row):
              clean = {}
              for name, value in row.items():
                  if name not in self.columns:
                      raise ValueError(f"unknown column {name!r}")
                  kind = self.columns[name]
                  if value is not None:
                      if kind is float and type(value) is int:
                          value = float(value)
                      if type(value) is not kind:                 # rejects True for an int column
                          raise TypeError(f"column {name!r} holds {kind.__name__}")
                  clean[name] = value
              return clean

          # Storage hooks: later steps extend these two, so every write path keeps indexes and expiry in step.
          def _write(self, row_id, row):
              self.rows[row_id] = row

          def _erase(self, row_id):
              return self.rows.pop(row_id, None)

          def put(self, row_id, row):
              clean = self._check(row)                           # validate before touching anything
              previous = self._erase(row_id)
              self._write(row_id, clean)
              return None if previous is None else dict(previous)

          def get(self, row_id):
              row = self.rows.get(row_id)
              return None if row is None else dict(row)

          def delete(self, row_id):
              return self._erase(row_id) is not None

          def __len__(self):
              return len(self.rows)

      import bisect
      import operator

      _OPS = {"=": operator.eq, "<": operator.lt, "<=": operator.le, ">": operator.gt, ">=": operator.ge}


      class Table(Table):
          def __init__(self, columns, **options):
              super().__init__(columns, **options)
              self.indexes = {}          # column -> sorted list of (value, row_id); rows with None are left out

          def create_index(self, column):
              if column not in self.columns:
                  raise ValueError(f"unknown column {column!r}")
              self.indexes[column] = sorted((row[column], rid) for rid, row in self.rows.items() if row.get(column) is not None)

          def _write(self, row_id, row):
              super()._write(row_id, row)
              for column, index in self.indexes.items():
                  if row.get(column) is not None:
                      bisect.insort(index, (row[column], row_id))

          def _erase(self, row_id):
              old = super()._erase(row_id)
              if old is not None:
                  for column, index in self.indexes.items():
                      if old.get(column) is not None:
                          i = bisect.bisect_left(index, (old[column], row_id))
                          del index[i]
              return old

          def _candidates(self, where):
              """Row ids that may match: the narrowest index range over the conditions, else every row."""
              ranges = {}                      # indexed column -> [lo, hi) after every condition on it
              first = lambda entry: entry[0]
              for column, op, value in where:
                  index = self.indexes.get(column)
                  if index is None:
                      continue
                  lo, hi = ranges.get(column, (0, len(index)))
                  if op in ("=", ">="):
                      lo = max(lo, bisect.bisect_left(index, value, key=first))
                  elif op == ">":
                      lo = max(lo, bisect.bisect_right(index, value, key=first))
                  if op in ("=", "<="):
                      hi = min(hi, bisect.bisect_right(index, value, key=first))
                  elif op == "<":
                      hi = min(hi, bisect.bisect_left(index, value, key=first))
                  ranges[column] = (lo, hi)
              if not ranges:
                  return list(self.rows)
              column, (lo, hi) = min(ranges.items(), key=lambda item: item[1][1] - item[1][0])
              return [rid for _, rid in self.indexes[column][lo:hi]]

          def scan(self, where=(), order_by=None, limit=None):
              where = list(where)
              for column, op, _ in where:
                  if column not in self.columns or op not in _OPS:
                      raise ValueError(f"bad condition on {column!r} with {op!r}")
              out = []
              for rid in self._candidates(where):
                  row = self.rows[rid]
                  if all(row.get(c) is not None and _OPS[op](row[c], v) for c, op, v in where):
                      out.append((rid, row))
              if order_by is None:
                  out.sort(key=lambda item: item[0])
              else:
                  def key(item):
                      value = item[1].get(order_by)
                      return (value is None, 0 if value is None else value, item[0])
                  out.sort(key=key)
              if limit is not None:
                  out = out[:limit]
              return [(rid, dict(row)) for rid, row in out]

      import heapq
      import time


      class ManualClock:
          """A clock the tests move by hand: call it for the time, set .now to move it."""

          def __init__(self, now=0.0):
              self.now = now

          def __call__(self):
              return self.now


      class Table(Table):
          def __init__(self, columns, clock=time.monotonic, **options):
              super().__init__(columns, **options)
              self.clock = clock
              self.expires = {}      # row_id -> expiry time, for rows with a ttl
              self.heap = []         # (expiry, row_id); stale entries are skipped

          def _alive(self, row_id, now):
              exp = self.expires.get(row_id)
              return row_id in self.rows and (exp is None or now < exp)

          def _erase(self, row_id):
              self.expires.pop(row_id, None)
              return super()._erase(row_id)

          def put(self, row_id, row, ttl=None):
              if ttl is not None and ttl <= 0:
                  raise ValueError("ttl must be positive")
              now = self.clock()
              was_alive = self._alive(row_id, now)
              previous = super().put(row_id, row)          # validates, erases, writes (and indexes)
              if ttl is not None:
                  self.expires[row_id] = now + ttl
                  heapq.heappush(self.heap, (now + ttl, row_id))
              return previous if was_alive else None

          def get(self, row_id):
              return super().get(row_id) if self._alive(row_id, self.clock()) else None

          def delete(self, row_id):
              alive = self._alive(row_id, self.clock())
              super().delete(row_id)                       # an expired row is dropped either way
              return alive

          def purge(self):
              """Drop every expired row from storage and indexes; return how many went."""
              now, gone = self.clock(), 0
              while self.heap and self.heap[0][0] <= now:
                  exp, row_id = heapq.heappop(self.heap)
                  if self.expires.get(row_id) == exp:
                      self._erase(row_id)
                      gone += 1
              return gone

          def scan(self, where=(), order_by=None, limit=None):
              self.purge()                                 # before the limit is applied, never after
              return super().scan(where, order_by=order_by, limit=limit)

          def __len__(self):
              self.purge()
              return super().__len__()

          def stats(self):
              return {"rows": len(self.rows), "index_entries": sum(len(i) for i in self.indexes.values())}

      class Table(Table):
          def __init__(self, columns, **options):
              super().__init__(columns, **options)
              self.snapshots = {}

          def snapshot(self):
              """Save the live rows and each row's remaining ttl; return a token for restore."""
              self.purge()
              now = self.clock()
              saved = {}
              for row_id, row in self.rows.items():
                  exp = self.expires.get(row_id)
                  saved[row_id] = (dict(row), None if exp is None else exp - now)
              token = len(self.snapshots) + 1
              self.snapshots[token] = saved
              return token

          def restore(self, token):
              saved = self.snapshots[token]                 # KeyError for an unknown token
              now = self.clock()
              self.rows, self.expires, self.heap = {}, {}, []
              for row_id, (row, remaining) in saved.items():
                  self.rows[row_id] = dict(row)             # a copy: the snapshot must survive later writes
                  if remaining is not None:
                      self.expires[row_id] = now + remaining
                      self.heap.append((now + remaining, row_id))
              heapq.heapify(self.heap)
              for column in self.indexes:                   # index definitions are schema: rebuild each over the restored rows
                  self.create_index(column)

      class Table(Table):
          def __init__(self, columns, **options):
              super().__init__(columns, **options)
              self.undo = None       # row_id -> (stored row or None, expiry) before the transaction first touched it

          def _remember(self, row_id):
              if self.undo is not None and row_id not in self.undo:
                  row = self.rows.get(row_id)
                  self.undo[row_id] = (None if row is None else dict(row), self.expires.get(row_id))

          def _erase(self, row_id):
              self._remember(row_id)                       # every path that drops a row, purge included
              return super()._erase(row_id)

          def _write(self, row_id, row):
              self._remember(row_id)
              super()._write(row_id, row)

          def begin(self):
              if self.undo is not None:
                  raise RuntimeError("a transaction is already open")
              self.undo = {}

          def commit(self):
              if self.undo is None:
                  raise RuntimeError("no open transaction")
              self.undo = None

          def rollback(self):
              if self.undo is None:
                  raise RuntimeError("no open transaction")
              undo, self.undo = self.undo, None            # restoring must not log itself
              for row_id, (row, exp) in undo.items():
                  super()._erase(row_id)
                  if row is not None:
                      super()._write(row_id, row)
                      if exp is not None:
                          self.expires[row_id] = exp
                          heapq.heappush(self.heap, (exp, row_id))

          def snapshot(self):
              if self.undo is not None:
                  raise RuntimeError("snapshot inside a transaction would save uncommitted rows")
              return super().snapshot()

          def restore(self, token):
              if self.undo is not None:
                  raise RuntimeError("restore inside a transaction cannot be rolled back")
              super().restore(token)

This lab grows one table class over five steps. Each step adds a requirement that a design fitted only to the previous step tends to get wrong. Step n runs its tests against everything you have written so far, so the indexes from step 2 have to survive the expiry from step 3, the snapshots from step 4 and the rollbacks from step 5.

Each step subclasses the last (`class Table(Table)`), so a step can call what came before it through `super()`. The design choice that pays off in every later step comes in step 1: route every write and every removal through one method each. Indexes, expiry and the undo log then hook into two places instead of six.

Time is injected through `ManualClock`, so the expiry tests are deterministic. Three steps carry budgets that a full scan or a whole-table copy cannot meet:

- **Step 2:** 600 range scans over 50,000 rows.
- **Step 3:** a purge that must hand back memory from 20,000 expired rows.
- **Step 5:** 2,000 rollbacks over 50,000 rows.

The specifications are original. They are modelled on the progressive in-memory database rounds that candidates report from Ramp, Anthropic and Perplexity, where each round adds functionality to the same code.
