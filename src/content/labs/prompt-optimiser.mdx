    brief: |
      Every optimiser in this lab climbs whatever metric you give it, so the metric comes first. MockModel stands in for the model: it reads rules from the instruction, copies labels from demos that share a word with the ticket, falls back to keywords, and answers with a bare label only when the instruction asks for one word. Write metric(example, output), which returns a score in [0, 1] and a feedback string. A reply that is exactly the gold label scores 1.0. A reply that contains the right label in the wrong format earns partial credit, and how much is your call; its feedback says format. A wrong label, no label, or more than one label scores 0.0, and the feedback names what was expected and what came back.       def bootstrap(model, instruction, trainset, metric, max_demos=4):
          """Run the teacher (model with instruction and no demos) over trainset in order and keep (ticket, label)
          demos from replies that score 1.0. Stop calling the model once max_demos are kept. Skip a ticket already kept."""
          demos, seen = [], set()
          for example in trainset:
              if len(demos) >= max_demos:
                  break
              key = " ".join(example["ticket"].lower().split())
              if key in seen:
                  continue
              reply = model.complete(instruction, [], example["ticket"])
              if metric(example, reply)["score"] >= 1.0:
                  demos.append((example["ticket"], example["label"]))
                  seen.add(key)
          return demos

      import random


      def propose(base, failures, n, seed=0):
          """failures: (example, feedback) pairs from running base. Return up to n distinct instructions, each base plus
          fixes read from the feedback: "Answer with one word." when any feedback is about format, and one rule
          "Tickets that mention 'w' are label." mined from a failing ticket."""
          rng = random.Random(seed)
          fix = " Answer with one word." if any(f.startswith("format") for _, f in failures) else ""
          words = sorted({(w, ex["label"]) for ex, f in failures if f.startswith("wrong label")
                          for w in ex["ticket"].lower().split() if w not in STOPWORDS and w not in KEYWORDS})
          out = [base + fix] if fix else []
          rng.shuffle(words)
          for w, label in words:
              if len(out) >= n:
                  break
              cand = f"{base}{fix} Tickets that mention '{w}' are {label}."
              if cand not in out:
                  out.append(cand)
          return out[:n]


      def random_search(model, instructions, demo_sets, dev, metric, budget, seed=0):
          """Sample (instruction, demo set) pairs without repeats, score each on all of dev (mean metric score), and return
          {"instruction", "demos", "dev_score", "tried"} for the best. Never spend more than budget model calls."""
          rng = random.Random(seed)
          pairs = [(i, d) for i in range(len(instructions)) for d in range(len(demo_sets))]
          rng.shuffle(pairs)
          best, tried = None, 0
          for i, d in pairs[: budget // max(1, len(dev))]:
              scores = [metric(ex, model.complete(instructions[i], demo_sets[d], ex["ticket"]))["score"] for ex in dev]
              score = sum(scores) / len(scores)
              tried += 1
              if best is None or score > best["dev_score"]:
                  best = {"instruction": instructions[i], "demos": list(demo_sets[d]), "dev_score": score}
          best["tried"] = tried
          return best

      import math


      def heldout_check(model, baseline, candidate, heldout, metric, z=1.96):
          """baseline and candidate are {"instruction", "demos"}. Score both on every held-out example (paired) and accept
          the candidate only when the lower bound of the interval on the mean paired gain is above zero.
          Return {"accept", "baseline", "candidate", "gain", "low", "high"}."""
          diffs, b_total, c_total = [], 0.0, 0.0
          for ex in heldout:
              b = metric(ex, model.complete(baseline["instruction"], baseline["demos"], ex["ticket"]))["score"]
              c = metric(ex, model.complete(candidate["instruction"], candidate["demos"], ex["ticket"]))["score"]
              b_total += b
              c_total += c
              diffs.append(c - b)
          n = len(diffs)
          if n < 2:
              raise ValueError("need at least two held-out examples")
          mean = sum(diffs) / n
          sd = math.sqrt(sum((d - mean) ** 2 for d in diffs) / (n - 1))
          half = z * sd / math.sqrt(n)
          return {"accept": mean - half > 0, "baseline": b_total / n, "candidate": c_total / n,
                  "gain": mean, "low": mean - half, "high": mean + half}

      import copy
      import hashlib
      import json


      class PromptRegistry:
          """Versions are immutable and numbered from 1 per name; labels are pointers to versions."""

          def __init__(self):
              self._versions = {}
              self._labels = {}

          def create(self, name, prompt, config=None):
              versions = self._versions.setdefault(name, [])
              versions.append(copy.deepcopy({"version": len(versions) + 1, "prompt": prompt, "config": config or {}}))
              self._labels.setdefault(name, {})["latest"] = len(versions)
              return len(versions)

          def get(self, name, label="production", version=None):
              n = version if version is not None else self._labels.get(name, {}).get(label)
              if n is None:
                  raise KeyError(f"{name} has no version labelled {label}")
              return copy.deepcopy(self._versions[name][n - 1])

          def move_label(self, name, label, version):
              if label == "latest":
                  raise ValueError("latest is managed by the registry")
              if not 1 <= version <= len(self._versions.get(name, [])):
                  raise KeyError(f"{name} has no version {version}")
              self._labels.setdefault(name, {})[label] = version

          def labels(self, name):
              return dict(self._labels.get(name, {}))


      def fingerprint(examples):
          rows = sorted(json.dumps(e, sort_keys=True) for e in examples)
          return hashlib.sha256("\n".join(rows).encode()).hexdigest()[:12]


      def register_winner(registry, name, winner, check, dev, heldout):
          """New version with the scores and data fingerprints in its config; "staging" only when the check accepted it."""
          config = {
              "demos": [list(d) for d in winner["demos"]],
              "dev_score": winner["dev_score"],
              "heldout": {k: check[k] for k in ("accept", "baseline", "candidate", "gain", "low", "high")},
              "dev_set": fingerprint(dev),
              "heldout_set": fingerprint(heldout),
          }
          version = registry.create(name, winner["instruction"], config)
          if check["accept"]:
              registry.move_label(name, "staging", version)
          return version
