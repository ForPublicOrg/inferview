      import hashlib
      import re
      import unicodedata

      _WS = re.compile(r"\s+")


      def normalise(text):
          return _WS.sub(" ", unicodedata.normalize("NFKC", text).casefold()).strip()


      def _order(record):
          return (record["ts"], record["id"])


      def exact_dedup(records):
          first = {}
          for r in sorted(records, key=_order):
              h = hashlib.sha256(normalise(r["text"]).encode()).hexdigest()
              first.setdefault(h, r["id"])
          keep_ids = set(first.values())
          kept, dropped = [], []
          for r in records:
              if r["id"] in keep_ids:
                  kept.append(r)
              else:
                  h = hashlib.sha256(normalise(r["text"]).encode()).hexdigest()
                  dropped.append((r["id"], "exact", first[h]))
          return kept, dropped
  - title: Near-dedup with MinHash and banding
    brief: |
      Paraphrased tickets and retried messages share most of their character 5-grams without being identical. Write the shingler, a MinHash signature, and `near_dedup`, which finds candidates through band buckets and confirms each with the exact Jaccard similarity of the shingle sets. With 16 bands of 4 rows, a pair at similarity s becomes a candidate with probability 1 - (1 - s^4)^16: 0.988 at s = 0.7 and 0.9998 at s = 0.8, so the hidden tests check that no two kept records reach max(threshold, 0.85) and that every dropped record really is within the threshold of the record it lost to. Character shingles are the choice here because support messages are a sentence or two, too short for word 5-grams.

      import random

      _P = (1 << 61) - 1


      def shingles(text, k=5):
          t = normalise(text)
          if len(t) <= k:
              return {t}
          return {t[i:i + k] for i in range(len(t) - k + 1)}


      def jaccard(a, b):
          return len(a & b) / len(a | b) if a or b else 1.0


      def _perms(num_perm, seed):
          rng = random.Random(seed)
          return [(rng.randrange(1, _P), rng.randrange(0, _P)) for _ in range(num_perm)]


      def minhash(shingle_set, num_perm=64, seed=1):
          xs = [int.from_bytes(hashlib.blake2b(s.encode(), digest_size=8).digest(), "big") for s in shingle_set]
          return [min((a * x + b) % _P for x in xs) for a, b in _perms(num_perm, seed)]


      def near_dedup(records, threshold=0.8, bands=16, rows=4):
          sh = {r["id"]: shingles(r["text"]) for r in records}
          buckets = {}
          kept_ids, dropped = set(), []
          for r in sorted(records, key=_order):
              sig = minhash(sh[r["id"]], num_perm=bands * rows)
              keys = [(b, tuple(sig[b * rows:(b + 1) * rows])) for b in range(bands)]
              candidates = set()
              for key in keys:
                  candidates.update(buckets.get(key, ()))
              best = None
              for c in sorted(candidates):
                  s = jaccard(sh[r["id"]], sh[c])
                  if s >= threshold and (best is None or s > best[0]):
                      best = (s, c)
              if best:
                  dropped.append((r["id"], "near", best[1]))
                  continue
              kept_ids.add(r["id"])
              for key in keys:
                  buckets.setdefault(key, []).append(r["id"])
          return [r for r in records if r["id"] in kept_ids], dropped

      _EMAIL = re.compile(r"[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,}")
      _CARD = re.compile(r"(?<!\d)(?:\d[ -]?){12,18}\d(?!\d)")
      _PHONE = re.compile(r"(?<![\d+])\+?(?:\d{1,3}[ .-]?)?\(?\d{3}\)?[ .-]?\d{3}[ .-]?\d{4}(?!\d)")
      _PRIORITY = {"CARD": 0, "EMAIL": 1, "PHONE": 2}


      def luhn_ok(digits):
          total = 0
          for i, ch in enumerate(reversed(digits)):
              d = int(ch)
              if i % 2:
                  d = d * 2 - 9 if d > 4 else d * 2
              total += d
          return total % 10 == 0


      def detect(text):
          found = [(m.start(), m.end(), "EMAIL") for m in _EMAIL.finditer(text)]
          for m in _CARD.finditer(text):
              digits = re.sub(r"\D", "", m.group())
              if 13 <= len(digits) <= 19 and luhn_ok(digits):
                  found.append((m.start(), m.end(), "CARD"))
          found += [(m.start(), m.end(), "PHONE") for m in _PHONE.finditer(text)]
          found.sort(key=lambda f: (-(f[1] - f[0]), _PRIORITY[f[2]], f[0]))
          chosen = []
          for f in found:
              if all(f[1] <= c[0] or f[0] >= c[1] for c in chosen):
                  chosen.append(f)
          return sorted(chosen)


      def redact(text):
          out, last = [], 0
          for start, end, kind in detect(text):
              out.append(text[last:start])
              out.append("<%s>" % kind)
              last = end
          out.append(text[last:])
          return "".join(out)

    solution: |
      class GateFailed(Exception):
          def __init__(self, report):
              super().__init__("gates failed: " + ", ".join(report["failures"]))
              self.report = report


      def run_batch(records, gates):
          n = len(records)
          scrubbed, pii = [], 0
          for r in records:
              if detect(r["text"]):
                  pii += 1
              scrubbed.append({**r, "text": redact(r["text"])})
          after_exact, exact = exact_dedup(scrubbed)
          kept, near = near_dedup(after_exact)
          report = {"input": n, "pii_records": pii, "exact_dropped": len(exact),
                    "near_dropped": len(near), "kept": len(kept), "failures": []}
          rates = {
              "max_pii_rate": pii / n if n else 0.0,
              "max_exact_rate": len(exact) / n if n else 0.0,
              "max_near_rate": len(near) / len(after_exact) if after_exact else 0.0,
          }
          for name, value in rates.items():
              if name in gates and value > gates[name]:
                  report["failures"].append(name)
          if "min_kept" in gates and len(kept) < gates["min_kept"]:
              report["failures"].append("min_kept")
          report["failures"].sort()
          if report["failures"]:
              raise GateFailed(report)
          return kept, report

      def side(user, eval_fraction, key):
          h = hashlib.sha256((key + "\x00" + user).encode()).digest()
          return "eval" if int.from_bytes(h[:8], "big") / 2 ** 64 < eval_fraction else "train"


      def split(records, eval_fraction, key, threshold=0.8, bands=16, rows=4):
          train = [r for r in records if side(r["user"], eval_fraction, key) == "train"]
          evals = [r for r in records if side(r["user"], eval_fraction, key) == "eval"]
          sh = {r["id"]: shingles(r["text"]) for r in records}
          index = {}
          for r in train:
              sig = minhash(sh[r["id"]], num_perm=bands * rows)
              for b in range(bands):
                  index.setdefault((b, tuple(sig[b * rows:(b + 1) * rows])), []).append(r["id"])
          leaks = []
          for r in evals:
              sig = minhash(sh[r["id"]], num_perm=bands * rows)
              candidates = set()
              for b in range(bands):
                  candidates.update(index.get((b, tuple(sig[b * rows:(b + 1) * rows])), ()))
              for c in sorted(candidates):
                  s = jaccard(sh[r["id"]], sh[c])
                  if s >= threshold:
                      leaks.append((r["id"], c, s))
          leaked = {e for e, _, _ in leaks}
          return train, [r for r in evals if r["id"] not in leaked], leaks

This lab builds the pipeline from the lesson on [curating eval and SFT data from production](/learn/llmops/data-curation/curation-pipeline) as one program. Each step adds functions to the same file and runs its tests against everything written so far, so the gates in step 4 call the deduplication from steps 1 and 2 and the redaction from step 3.

Records are dictionaries with `id`, `user`, `ts` and `text`, the shape of a logged user turn after the logging service has replaced the account id with a pseudonym. The test corpora are generated in the browser with planted duplicates, paraphrases and personal data; no real log text is involved.

The detectors in step 3 are a floor, not a scrubber you can ship. Presidio, which adds named-entity recognition and context rules to the same regex and checksum approach, states that it cannot guarantee finding all sensitive information, and names, addresses and free-text account details pass every regex here.
