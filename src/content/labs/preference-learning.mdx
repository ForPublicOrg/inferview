      import math


      def softplus(z):
          """log(1 + e^z) without overflow for large |z|."""
          return max(z, 0.0) + math.log1p(math.exp(-abs(z)))


      def sigmoid(z):
          if z >= 0:
              return 1.0 / (1.0 + math.exp(-z))
          e = math.exp(z)
          return e / (1.0 + e)


      def dot(u, v):
          return sum(a * b for a, b in zip(u, v))


      def rm_loss(w, pairs):
          """Mean Bradley-Terry loss: -log sigmoid(r(chosen) - r(rejected)) with r(x) = w . x."""
          return sum(softplus(-(dot(w, c) - dot(w, r))) for c, r in pairs) / len(pairs)


      def fit_reward(pairs, dim, lr=0.5, steps=300):
          diffs = [[a - b for a, b in zip(c, r)] for c, r in pairs]
          w = [0.0] * dim
          n = len(diffs)
          for _ in range(steps):
              g = [0.0] * dim
              for d in diffs:
                  coef = -sigmoid(-dot(w, d)) / n
                  for k in range(dim):
                      g[k] += coef * d[k]
              w = [wk - lr * gk for wk, gk in zip(w, g)]
          return w
  - title: The DPO loss
    brief: |
      DPO replaces the reward model with the policy's own log-ratio against a frozen reference: the margin for a pair is beta times the chosen log-ratio minus the rejected log-ratio, and the loss is softplus of minus the margin. Inputs are summed sequence log-probs, as a trainer would compute them in one forward pass over each response. When the policy equals the reference every margin is 0 and the loss is log 2 at any beta; the tests also feed margins of plus and minus 100,000, which a naive log(sigmoid(m)) turns into nan or inf.

          Returns (loss, margins) where margins[i] = beta * ((pol_c - ref_c) - (pol_r - ref_r)) for pair i.
          """

      def dpo_loss(pol_c, pol_r, ref_c, ref_r, beta):
          """Mean DPO loss over a batch, from summed sequence log-probs under the policy and the frozen reference.

          Returns (loss, margins) where margins[i] = beta * ((pol_c - ref_c) - (pol_r - ref_r)) for pair i.
          """
          margins = [beta * ((pc - rc) - (pr - rr)) for pc, pr, rc, rr in zip(pol_c, pol_r, ref_c, ref_r)]
          return sum(softplus(-m) for m in margins) / len(margins), margins
  - title: The DPO gradient by hand
    brief: |
      The policy is now log-linear over a small candidate set: log pi(y) = theta . phi(y) minus logsumexp over the candidates, and the reference is the same family at theta_ref. Derive the gradient of the mean DPO loss with respect to theta and write it out; do not differentiate numerically. Chosen and rejected share a candidate set, so the logsumexp term's gradient appears in both log-probs and cancels in their difference. The tests compare your gradient to central differences.

          m = max(xs)
          lse = m + math.log(sum(math.exp(x - m) for x in xs))

          """A log-linear policy over a fixed candidate set: log pi(y) = theta . phi(y) - logsumexp."""
          return log_softmax([dot(theta, phi) for phi in cands])

          An example is {"cands": [phi, ...], "chosen": i, "rejected": j}.

    solution: |
      def log_softmax(xs):
          m = max(xs)
          lse = m + math.log(sum(math.exp(x - m) for x in xs))
          return [x - lse for x in xs]


      def logprobs(theta, cands):
          """A log-linear policy over a fixed candidate set: log pi(y) = theta . phi(y) - logsumexp."""
          return log_softmax([dot(theta, phi) for phi in cands])


      def dpo_grad(theta, theta_ref, examples, beta):
          """Mean DPO loss over examples and its gradient with respect to theta.

          An example is {"cands": [phi, ...], "chosen": i, "rejected": j}.
          """
          loss, g = 0.0, [0.0] * len(theta)
          n = len(examples)
          for ex in examples:
              cands, i, j = ex["cands"], ex["chosen"], ex["rejected"]
              lp, lq = logprobs(theta, cands), logprobs(theta_ref, cands)
              m = beta * ((lp[i] - lq[i]) - (lp[j] - lq[j]))
              loss += softplus(-m) / n
              # d log pi(i) - d log pi(j) = phi_i - phi_j: the expectation terms cancel within one candidate set.
              coef = -beta * sigmoid(-m) / n
              for k in range(len(theta)):
                  g[k] += coef * (cands[i][k] - cands[j][k])
          return loss, g
  - title: Beta sweep under noisy labels
    brief: |
      Train from theta_ref with your gradient and sweep beta. The dataset repeats one comparison three times one way and once the other way, which is what tied or noisy labels look like to the loss. At the optimum the implicit margin, beta times the log-ratio difference, equals the log-odds of the label split, so a smaller beta buys the same margin with a larger move away from the reference. The loss's curvature in theta scales with beta squared, so pick a step size per beta that converges in the step budget; a fixed one either crawls at beta 0.05 or overshoots at beta 1.

    solution: |
      def kl(theta, theta_ref, cands):
          """KL(pi_theta || pi_ref) over one candidate set, in nats."""
          lp, lq = logprobs(theta, cands), logprobs(theta_ref, cands)
          return sum(math.exp(a) * (a - b) for a, b in zip(lp, lq))


      def train_dpo(theta_ref, examples, beta, steps=400, lr=1.0):
          # The loss's curvature in theta scales with beta squared, so a fixed lr either crawls at small beta
          # or overshoots at large beta; scaling by 1 / beta^2 gives every beta the same number of effective steps.
          theta = list(theta_ref)
          step = lr / (beta * beta)
          for _ in range(steps):
              _, g = dpo_grad(theta, theta_ref, examples, beta)
              theta = [t - step * gk for t, gk in zip(theta, g)]
          return theta


      def beta_sweep(theta_ref, examples, betas, steps=400, lr=1.0):
          """For each beta: train, then report the mean KL from the reference and the pairwise accuracy on examples."""
          out = []
          for beta in betas:
              theta = train_dpo(theta_ref, examples, beta, steps, lr)
              kls = [kl(theta, theta_ref, ex["cands"]) for ex in examples]
              acc = 0
              for ex in examples:
                  lp = logprobs(theta, ex["cands"])
                  acc += lp[ex["chosen"]] > lp[ex["rejected"]]
              out.append({"beta": beta, "theta": theta, "kl": sum(kls) / len(kls), "acc": acc / len(examples)})
          return out
  - title: The NLL term and the falling chosen response
    brief: |
      Llama 3 adds a negative log-likelihood term on the chosen response with coefficient 0.2 to keep its log-probability from falling. Add that term and its gradient.     solution: |
      def dpo_nll_grad(theta, theta_ref, examples, beta, alpha):
          """DPO loss plus alpha times the mean negative log-likelihood of the chosen response, and its gradient."""
          loss, g = dpo_grad(theta, theta_ref, examples, beta)
          if alpha == 0:
              return loss, g
          n = len(examples)
          for ex in examples:
              cands, i = ex["cands"], ex["chosen"]
              lp = logprobs(theta, cands)
              loss += alpha * -lp[i] / n
              probs = [math.exp(x) for x in lp]
              for k in range(len(theta)):
                  expected = sum(p * phi[k] for p, phi in zip(probs, cands))
                  g[k] += alpha * -(cands[i][k] - expected) / n
          return loss, g


      def train(theta_ref, examples, beta, alpha=0.0, steps=400, lr=1.0):
          theta = list(theta_ref)
          step = lr / (beta * beta)
          for _ in range(steps):
              _, g = dpo_nll_grad(theta, theta_ref, examples, beta, alpha)
              theta = [t - step * gk for t, gk in zip(theta, g)]
          return theta

The lesson [DPO in practice](/learn/training/post-training/dpo-practice) states the DPO loss and its gradient; this lab makes you derive both and then shows two behaviours that matter when you run DPO on real data. Everything runs on feature vectors and a log-linear policy over a handful of candidate responses, so each quantity is a few lines of arithmetic you can check by hand. No model is trained in the browser.

Step 1 is the reward model DPO removes. Steps 2 and 3 build the DPO loss on summed log-probs and then on a policy whose gradient you derive and check against central differences. Step 4 trains that policy on labels that disagree 3 to 1 and shows that beta sets how far the policy moves to express the same preference. Step 5 builds a case where the chosen response loses probability while every margin grows, and adds the NLL term Llama 3 used against it.
