  - title: The loop, its turn cap and the final-turn rule
    brief: |
      The starter is the whole skeleton: a scripted MockModel that records every request, helpers for replies, calls and tools, request building, tool execution, and one do-nothing hook for each later step. You write two functions. invoke(state, c) runs one call and returns its output as a string; an unknown tool or an exception comes back as a string starting with "error:". run_agent(model, task, tools, max_turns=8, **opts) is the loop: state = new_state(task, tools, opts); for each turn, call maybe_compact(state) and apply_config(state, turn), and on the last allowed turn first append {"role": "user", "content": FINAL_NOTE}; send build_request(state, final); record state["turns"]; append the reply as {"role": "assistant", "content": text, "tool_calls": calls}; if charge(state, reply) is true, stop with "budget"; a reply with no calls ends the run with whatever on_final returns, unless it returns "continue"; on the final turn calls are not run and the status is "max_turns"; otherwise extend the messages with run_tools(state, calls). Return result(state, status, text). The final-turn rule exists because a capped run that just stops leaves no answer: the Claude Agent SDK counts the final text-only response as a turn, and a run that hits max_turns without room for it ends with no response.

              self.requests = []

              if not self.replies:

          return {"text": text, "tool_calls": list(calls), "usage": {"input_tokens": tokens_in, "output_tokens": tokens_out}}


      def call(id, name, /, **args):
          return {"id": id, "name": name, "args": args}


      def tool(fn, description="", read_only=False, shell=False, duration=0.0):
          return {"fn": fn, "description": description, "read_only": read_only, "shell": shell, "duration": duration}

          state = {"opts": opts, "tools": dict(tools), "store": opts.get("store", {}), "messages": [],
                   "turns": 0, "calls": 0, "clock": 0.0, "spent": 0.0, "compactions": 0}

              m = setup(state)
              if m:
                  state["messages"].append(m)
          state["messages"].append({"role": "user", "content": task})
          return state


      def result(state, status, text):
          return {"status": status, "text": text, "turns": state["turns"], "messages": state["messages"],
                  "clock": state["clock"], "spent": state["spent"], "store": state["store"]}


      def tool_list(state):
          return [{"name": n, "description": t.get("description", "")} for n, t in state["tools"].items()]


      def build_request(state, final):
          return {"system": SYSTEM, "tools": tool_list(state), "messages": list(state["messages"]),
                  "tool_choice": "none" if final else "auto"}


      def execute_call(state, c):
          refusal = check_permission(state, c)
          if refusal is not None:
              return refusal
          return fit_output(state, c["id"], invoke(state, c))


      def run_tools(state, calls):
          out = []
          for c in calls:
              state["clock"] += state["tools"].get(c["name"], {}).get("duration", 0.0)
              out.append({"role": "tool", "id": c["id"], "content": execute_call(state, c)})

      def memory_message(state): return None
      def skills_setup(state): return None
      def maybe_compact(state): pass

      def check_permission(state, c): return None
      def fit_output(state, call_id, text): return text
      def charge(state, r): return False
      def on_final(state, r): return "done"


      def invoke(state, c):
          spec = state["tools"].get(c["name"])
          if spec is None:
              return f"error: unknown tool {c['name']}"
          try:
              return str(spec["fn"](**c["args"]))
          except Exception as e:
              return f"error: {type(e).__name__}: {e}"


      def run_agent(model, task, tools, max_turns=8, **opts):
          state = new_state(task, tools, opts)
          for turn in range(1, max_turns + 1):
              maybe_compact(state)
              apply_config(state, turn)
              final = turn == max_turns
              if final:
                  state["messages"].append({"role": "user", "content": FINAL_NOTE})
              r = model(build_request(state, final))
              state["turns"] = turn
              state["messages"].append({"role": "assistant", "content": r["text"], "tool_calls": r["tool_calls"]})
              if charge(state, r):
                  return result(state, "budget", r["text"])
              if not r["tool_calls"]:
                  verdict = on_final(state, r)
                  if verdict != "continue":
                      return result(state, verdict, r["text"])
                  continue
              if final:
                  return result(state, "max_turns", r["text"])
              state["messages"].extend(run_tools(state, r["tool_calls"]))
          return result(state, "max_turns", "")

      def fit_output(state, call_id, text):
          opts = state["opts"]
          budget = opts.get("output_budget", 30000)
          if len(text) <= budget:
              return text
          preview = min(opts.get("output_preview", 2000), budget)
          path, n = f"tool-output/{call_id}.txt", 1
          while path in state["store"]:
              n += 1
              path = f"tool-output/{call_id}-{n}.txt"
          state["store"][path] = text
          return text[:preview] + f"\n[output truncated: {len(text)} characters in total, saved at {path}]"
  - title: A deny, ask and allow gate for shell commands
    brief: |
      Write decide(command, rules) and check_permission(state, c). Rules are token prefixes under "deny", "ask" and "allow"; deny beats ask beats allow, and a command no rule matches is "ask". Split a script only when it is a linear chain of plain words joined by &&, ||, ; or |, and take the most restrictive decision across the parts. Anything else, such as a redirect, $( ), backticks, a variable, a glob, control flow, a newline, a background &, a quoting error or an empty part, is one invocation ["bash", "-lc", script] that a narrow allow rule cannot match. Codex's rules documentation lists the first five of these; the rest follow from splitting only what is provably a plain chain. check_permission gates only tools marked shell=True: allow runs; ask calls opts["approve"](command) and runs only on True; deny never runs, whatever an approver says. A refusal goes back to the model as the tool's result. The hidden tests try bypasses.
    starter: |
      def decide(command, rules):
          """Return "deny", "ask" or "allow" for one shell command string. rules maps "deny", "ask" and "allow" to
          lists of token prefixes such as ["git", "status"]."""
          raise NotImplementedError


      def check_permission(state, c):
          """Gate calls to tools marked shell=True, using decide(c["args"]["command"], opts["rules"]). Return None to
          let the call run, or a refusal string that goes back to the model instead of the output."""
          raise NotImplementedError

      def split_chain(script):
          """Split a linear chain of plain words into commands, or return None when the script is anything else."""
          commands, words, word, quote, quoted, i = [], [], "", None, False, 0
          while i < len(script):
              ch = script[i]
              if quote == "'":
                  if ch == "'":
                      quote = None
                  else:
                      word += ch
              elif quote == '"':
                  if ch == '"':
                      quote = None
                  elif ch in '$`\\':
                      return None  # substitution and escapes still work inside double quotes
                  else:
                      word += ch
              elif ch in "'\"":
                  quote, quoted = ch, True
              elif ch in " \t":
                  if word or quoted:
                      words.append(word)
                  word, quoted = "", False
              elif script.startswith(("&&", "||"), i) or ch in ";|":
                  if word or quoted:
                      words.append(word)
                  word, quoted = "", False
                  if not words:
                      return None  # an operator with no command before it

                  return None  # redirection, substitution, globs, background jobs, newlines and the rest
              else:
                  word += ch
              i += 1

              return None
          if word or quoted:
              words.append(word)
          if not words:
              return None  # empty script, or a trailing operator such as "npm test &&"

              if words[0] in CONTROL or re.match(r"^[A-Za-z_][A-Za-z0-9_]*=", words[0]):
                  return None  # control flow, or an environment assignment before the command
          return commands

    solution: |
      import re


      def charge(state, r):
          opts, usage = state["opts"], r.get("usage") or {}
          state["spent"] += usage.get("input_tokens", 0) * opts.get("price_in", 0.0) + usage.get("output_tokens", 0) * opts.get("price_out", 0.0)
          cap = opts.get("max_budget")
          return cap is not None and state["spent"] >= cap


      SKILL_NAME = re.compile(r"^(?!-)(?!.*--)[a-z0-9-]{1,64}(?<!-)$")


      def parse_skill(folder, text):
          """Return (name, description, body), or None when the frontmatter breaks the Agent Skills rules."""
          lines = text.split("\n")
          if not lines or lines[0].strip() != "---" or "---" not in [l.strip() for l in lines[1:]]:
              return None
          end = 1 + [l.strip() for l in lines[1:]].index("---")
          meta = {}
          for line in lines[1:end]:
              key, sep, value = line.partition(":")
              if sep:
                  meta[key.strip()] = value.strip().strip("\"'")
          name, description = meta.get("name", ""), meta.get("description", "")
          if not SKILL_NAME.match(name) or name != folder or not 1 <= len(description) <= 1024:
              return None
          return name, description, "\n".join(lines[end + 1:]).strip()


      def skills_setup(state):
          skills = {}
          for folder, text in (state["opts"].get("skills") or {}).items():
              parsed = parse_skill(folder, text)
              if parsed:
                  skills[parsed[0]] = parsed
          if not skills:
              return None
          sent = {}

          def load_skill(name):
              if name not in skills:
                  return f"error: no skill named {name}; available: {', '.join(sorted(skills))}"
              marker = f"<skill name=\"{name}\">"
              # Loaded means the body is still in the history: compaction may have summarised it away.
              if sent.get(name) == state["turns"] or any(marker in str(m.get("content", "")) for m in state["messages"]):
                  return f"skill {name} is already loaded in this conversation"
              sent[name] = state["turns"]
              return f"{marker}\n{skills[name][2]}\n</skill>"

          state["tools"]["load_skill"] = tool(load_skill, "Load a skill's full instructions by name", read_only=True)
          budget = state["opts"].get("skills_index_chars", 8000)
          header = "Skills you can load with load_skill (name: description):"
          names = sorted(skills)
          room = max(0, (budget - len(header)) // len(names) - 1)
          entries = []
          for n in names:
              entry = f"{n}: {skills[n][1]}"
              if len(entry) > room:
                  entry = entry[:max(len(n), room - 3)] + "..."  # shorten descriptions first, keep every name
              entries.append(entry)
          text = header + "".join("\n" + e for e in entries)
          while len(text) > budget and entries:
              entries.pop()  # still too long: drop skills from the end
              text = header + "".join("\n" + e for e in entries)
          return {"role": "user", "content": text}
