          s = str(text).strip()

              try:

              except ValueError:
                  pass
          raise ValueError(f"not a date: {text!r}")

      from collections import defaultdict


      def match_exact(receipts, txns, window_days):
          by_amount = defaultdict(list)
          for t in txns:
              by_amount[t["cents"]].append((t["posted"], t["id"]))
          wanted = defaultdict(list)
          for r in receipts:
              if r["currency"] == "USD":
                  wanted[r["cents"]].append((r["date"], r["id"]))
          span = timedelta(days=window_days)
          out = {}
          for cents, rs in wanted.items():
              ts = sorted(by_amount.get(cents, ()))
              p = 0
              # Every window has the same length, so taking receipts by date and giving each the earliest unused
              # transaction inside its window is maximum. A transaction posted before this receipt's date is also before
              # every later receipt's date, so the pointer never moves back.
              for day, rid in sorted(rs):
                  while p < len(ts) and ts[p][0] < day:
                      p += 1
                  if p < len(ts) and ts[p][0] <= day + span:
                      out[rid] = ts[p][1]
                      p += 1
          return out
  - title: Measure precision and recall of the matcher
    brief: "Nobody labels a year of receipts, so a matcher's numbers come from a labelled sample. score counts only the receipts in gold. A receipt matched to the wrong transaction counts twice, as a false positive and a false negative, and a receipt matched when gold says it has no transaction (a personal-card purchase, a second upload) is a false positive. wilson gives the interval to report next to each rate, because the first question about a rate from a 300-receipt sample is how much 300 receipts can say. label_sample draws the sample from the ids reproducibly, whatever order they arrive in. On generated books of 3,000 receipts (seed 4) the reference match_exact scores precision 0.975 and recall 0.684 against the full gold."

      import math


      def score(pred, gold):
          tp = fp = fn = 0
          for rid, truth in gold.items():
              guess = pred.get(rid)
              if guess is not None and guess == truth:
                  tp += 1
                  continue
              if guess is not None:
                  fp += 1          # matched to the wrong transaction, or matched when nothing should be
              if truth is not None:
                  fn += 1          # a wrong match also misses the right one
          return {"tp": tp, "fp": fp, "fn": fn,
                  "precision": tp / (tp + fp) if tp + fp else None,
                  "recall": tp / (tp + fn) if tp + fn else None}


      def wilson(k, n, z=1.96):
          if n == 0:
              return (0.0, 1.0)
          p = k / n
          centre = (p + z * z / (2 * n)) / (1 + z * z / n)
          half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / (1 + z * z / n)
          # The bounds are exactly 0 and 1 at k = 0 and k = n; rounding would otherwise leave them a hair inside.
          low = 0.0 if k == 0 else max(0.0, centre - half)
          high = 1.0 if k == n else min(1.0, centre + half)
          return (low, high)


      def label_sample(receipt_ids, size, seed=0):
          return random.Random(seed).sample(sorted(receipt_ids), size)

      import bisect
      from functools import lru_cache

      _NOISE = {"sq", "tst", "inc", "com", "help", "mktp", "us", "the"}
      TIP_MAX = 0.30        # a tip can push the charge up to 30% over the receipt total
      FX_FEE_MAX = 0.04     # a foreign charge can land up to 4% over the converted total


      @lru_cache(maxsize=None)
      def _tokens(text):
          words = re.findall(r"[a-z0-9]+", text.lower().replace("'", ""))
          return frozenset(w for w in words if not any(ch.isdigit() for ch in w) and w not in _NOISE and len(w) > 1)


      def merchant_similar(receipt_merchant, descriptor):
          mine = _tokens(receipt_merchant)
          return bool(mine) and len(mine & _tokens(descriptor)) * 2 >= len(mine)


      def _bounds(receipt, fx):
          c = receipt["cents"]
          if receipt["currency"] == "USD":
              return c, int(c * (1 + TIP_MAX))
          if receipt["currency"] == "EUR":
              return int(c * fx * 0.995), int(c * fx * (1 + FX_FEE_MAX)) + 1
          return None


      def amount_fits(receipt, txn, fx=FX_USD_PER_EUR):
          b = _bounds(receipt, fx)
          return b is not None and b[0] <= txn["cents"] <= b[1]


      def match_fuzzy(receipts, txns, window_days, fx=FX_USD_PER_EUR):
          by_day = defaultdict(list)
          for t in txns:
              by_day[t["posted"]].append((t["cents"], t["id"], t))
          for v in by_day.values():
              v.sort(key=lambda x: (x[0], x[1]))
          edges = []
          for r in receipts:
              b = _bounds(r, fx)
              if b is None:
                  continue
              for k in range(window_days + 1):
                  day = by_day.get(r["date"] + timedelta(days=k))
                  if not day:
                      continue
                  i = bisect.bisect_left(day, (b[0],))
                  while i < len(day) and day[i][0] <= b[1]:
                      t = day[i][2]
                      if merchant_similar(r["merchant"], t["descriptor"]):
                          exact = r["currency"] == "USD" and t["cents"] == r["cents"]
                          edges.append((0 if exact else 1, k, abs(t["cents"] - r["cents"]), r["id"], t["id"]))
                      i += 1
          # Most certain pairs first: exact amounts, then the shortest posting delay, then the smallest gap.
          edges.sort()
          out, used = {}, set()
          for _, _, _, rid, tid in edges:
              if rid not in out and tid not in used:
                  out[rid] = tid
                  used.add(tid)
          return out

      REASONS = ("missed", "already_matched", "merchant_mismatch", "amount_mismatch", "no_transaction")


      def explain_unmatched(receipts, txns, matches, window_days, fx=FX_USD_PER_EUR):
          by_day = defaultdict(list)
          for t in txns:
              by_day[t["posted"]].append(t)
          taken = set(matches.values())
          out = {}
          for r in receipts:
              if r["id"] in matches:
                  continue
              on_amount = on_merchant = both_free = both_taken = False
              for k in range(window_days + 1):
                  for t in by_day.get(r["date"] + timedelta(days=k), ()):

                      m = merchant_similar(r["merchant"], t["descriptor"])
                      on_amount |= a
                      on_merchant |= m
                      if a and m:
                          if t["id"] in taken:
                              both_taken = True
                          else:
                              both_free = True
              if both_free:
                  out[r["id"]] = "missed"
              elif both_taken:
                  out[r["id"]] = "already_matched"
              elif on_amount:
                  out[r["id"]] = "merchant_mismatch"
              elif on_merchant:
                  out[r["id"]] = "amount_mismatch"
              else:
                  out[r["id"]] = "no_transaction"
          return out


      def summary(explanations):
          counts = Counter(explanations.values())
          return sorted(counts.items(), key=lambda kv: (-kv[1], REASONS.index(kv[0])))
