      import math
      import random


      def log_softmax(logits):
          m = max(logits)
          lse = m + math.log(sum(math.exp(z - m) for z in logits))
          return [z - lse for z in logits]


      def softmax(logits):
          return [math.exp(v) for v in log_softmax(logits)]


      def sample(logits, rng):
          u = rng.random()
          acc = 0.0
          probs = softmax(logits)
          for i, p in enumerate(probs):
              acc += p
              if u < acc:
                  return i
          return len(probs) - 1

      ACTIONS = [f"answer: {n}" for n in range(K)] + ["answer: 3 or 5", "answer: 0 1 2 3 4 5 6 7", "I am not sure"]

  - title: Group-relative advantages
    brief: |
      GRPO replaces the value model with a baseline computed from the group itself: sample G responses to one prompt, score each, subtract the group mean and divide by the group's standard deviation.     solution: |
      def group_advantages(rewards, scale=True, eps=1e-6):
          n = len(rewards)
          mean = sum(rewards) / n
          centred = [r - mean for r in rewards]
          if not scale:
              return centred
          std = math.sqrt(sum(c * c for c in centred) / n)
          return [c / (std + eps) for c in centred]

      def clipped_grad(logits, old_logits, actions, advs, eps_low=0.2, eps_high=0.28):
          p = softmax(logits)
          lp = log_softmax(logits)
          lo = log_softmax(old_logits)
          g = [0.0] * len(logits)
          n = len(actions)
          for a, adv in zip(actions, advs):
              ratio = math.exp(lp[a] - lo[a])
              if adv == 0 or (adv > 0 and ratio > 1 + eps_high) or (adv < 0 and ratio < 1 - eps_low):
                  continue
              w = adv * ratio / n
              for k in range(len(g)):
                  g[k] -= w * p[k]
              g[a] += w
          return g


      def rollout(logits, prompt, G, rng):
          acts = [sample(logits, rng) for _ in range(G)]
          return acts, [verify(prompt, ACTIONS[a]) for a in acts]


      def train(prompts, steps=40, G=8, lr=1.0, inner=4, seed=0, eps_low=0.2, eps_high=0.28):
          rng = random.Random(seed)
          policy = {pr: [0.0] * len(ACTIONS) for pr in prompts}
          history = []
          for _ in range(steps):
              batch = []
              total = 0.0
              for pr in prompts:
                  acts, rewards = rollout(policy[pr], pr, G, rng)
                  total += sum(rewards) / G
                  batch.append((pr, list(policy[pr]), acts, group_advantages(rewards)))
              history.append(total / len(prompts))
              for _ in range(inner):
                  for pr, old, acts, advs in batch:
                      g = clipped_grad(policy[pr], old, acts, advs, eps_low, eps_high)
                      policy[pr] = [z + lr * gk for z, gk in zip(policy[pr], g)]
          return policy, history

  - title: KL term on and off
    brief: |
      GRPO as published adds beta times an estimate of KL from the policy to a frozen reference to the loss, per sampled token, with the estimator r - log r - 1 where r is the reference probability over the policy probability; it is never negative and its expectation under the policy is the exact KL. DAPO drops the term for long reasoning and TRL's default is now beta 0.     solution: |
      def kl_k3(logp, logp_ref):
          x = logp_ref - logp
          return math.exp(x) - x - 1.0


      def kl_exact(logits, ref_logits):
          lp, lr_ = log_softmax(logits), log_softmax(ref_logits)
          return sum(math.exp(a) * (a - b) for a, b in zip(lp, lr_))


      def kl_grad(logits, ref_logits, actions):
          lp, lr_ = log_softmax(logits), log_softmax(ref_logits)
          p = softmax(logits)
          g = [0.0] * len(logits)
          for a in actions:
              w = (1.0 - math.exp(lr_[a] - lp[a])) / len(actions)
              for k in range(len(g)):
                  g[k] -= w * p[k]
              g[a] += w
          return g


      def train_kl(prompts, ref, beta, steps=40, G=8, lr=1.0, inner=4, seed=0):
          rng = random.Random(seed)
          policy = {pr: list(ref[pr]) for pr in prompts}
          history = []
          for _ in range(steps):
              batch = []
              total = 0.0
              for pr in prompts:
                  acts, rewards = rollout(policy[pr], pr, G, rng)
                  total += sum(rewards) / G
                  batch.append((pr, list(policy[pr]), acts, group_advantages(rewards)))
              history.append(total / len(prompts))
              for _ in range(inner):
                  for pr, old, acts, advs in batch:
                      g = clipped_grad(policy[pr], old, acts, advs)
                      if beta:
                          kg = kl_grad(policy[pr], ref[pr], acts)
                          g = [a - beta * b for a, b in zip(g, kg)]
                      policy[pr] = [z + lr * gk for z, gk in zip(policy[pr], g)]
          return policy, history

    solution: |
      DEMO_REWARDS = [1.0, 0.0, 0.0]
      DEMO_LENGTHS = [40, 40, 400]


      def expected_grad(logits, rewards, lengths, mode, l_max):
          p = softmax(logits)
          mean_r = sum(pi * ri for pi, ri in zip(p, rewards))
          mean_len = sum(pi * li for pi, li in zip(p, lengths))
          g = [0.0] * len(logits)
          for i, (pi, ri, li) in enumerate(zip(p, rewards, lengths)):
              c = {"seq": 1.0 / li, "token": 1.0 / mean_len, "constant": 1.0 / l_max}[mode]
              w = pi * (ri - mean_r) * c
              for k in range(len(g)):
                  g[k] -= w * p[k]
              g[i] += w
          return g


      def length_demo(mode, steps=300, lr=200.0, l_max=400):
          z = [0.0, 0.0, 0.0]
          for _ in range(steps):
              g = expected_grad(z, DEMO_REWARDS, DEMO_LENGTHS, mode, l_max)
              z = [a + lr * b for a, b in zip(z, g)]
          return softmax(z)
