    solution: |
      import re


      def words(text):
          return re.findall(r"[a-z0-9]+", text.lower())


      def _blocks(lines):
          # A block is a paragraph or a whole table; a table is never split.
          blocks, cur, table = [], [], False
          for line in lines + [""]:
              is_t = line.lstrip().startswith("|")
              if cur and (not line.strip() or is_t != table):
                  blocks.append(("table" if table else "prose", "\n".join(cur)))
                  cur = []
              if line.strip():
                  if not cur:
                      table = is_t
                  cur.append(line)
          return blocks


      def chunk_markdown(doc_id, text, max_words=40):
          sections, path, body = [], [], []
          for line in text.splitlines():
              m = re.match(r"^(#{1,6})\s+(.*\S)\s*$", line)
              if m:
                  sections.append((" > ".join(h for _, h in path), body))
                  level = len(m.group(1))
                  path = [p for p in path if p[0] < level] + [(level, m.group(2))]
                  body = []
              else:
                  body.append(line)
          sections.append((" > ".join(h for _, h in path), body))
          chunks = []
          for heading, lines in sections:
              pieces = []
              for kind, block in _blocks(lines):
                  toks = block.split()
                  if kind == "prose" and len(toks) > max_words:
                      pieces += [(len(toks[i:i + max_words]), " ".join(toks[i:i + max_words])) for i in range(0, len(toks), max_words)]
                  else:
                      pieces.append((len(toks), block))
              cur, size = [], 0
              for n, piece in pieces:
                  if cur and size + n > max_words:
                      chunks.append((heading, "\n\n".join(cur)))
                      cur, size = [], 0
                  cur.append(piece)
                  size += n
              if cur:
                  chunks.append((heading, "\n\n".join(cur)))
          return [{"id": f"{doc_id}#{i}", "doc": doc_id, "heading": h, "text": t} for i, (h, t) in enumerate(chunks)]

    solution: |
      import math
      from collections import Counter


      class BM25:
          def __init__(self, chunks, k1=1.2, b=0.75):
              self.ids = [c["id"] for c in chunks]
              tfs = [Counter(words(c["heading"] + " " + c["text"])) for c in chunks]
              lengths = [sum(t.values()) for t in tfs]
              avg = sum(lengths) / len(lengths) if lengths else 0.0
              norm = [k1 * (1 - b + b * n / avg) if avg else k1 for n in lengths]
              self.postings = {}
              for i, tf in enumerate(tfs):
                  for term, c in tf.items():
                      self.postings.setdefault(term, []).append((i, c * (k1 + 1) / (c + norm[i])))
              n = len(chunks)
              self.idf = {t: math.log(1 + (n - len(p) + 0.5) / (len(p) + 0.5)) for t, p in self.postings.items()}

          def search(self, query, k):
              acc = {}
              for term in dict.fromkeys(words(query)):
                  for i, part in self.postings.get(term, []):
                      acc[i] = acc.get(i, 0.0) + self.idf[term] * part
              best = sorted((i for i in acc if acc[i] > 0), key=lambda i: (-acc[i], i))[:k]
              return [self.ids[i] for i in best]

  - title: Reciprocal rank fusion
    brief: |
      Write rrf. Each id scores the sum of 1 / (k + rank) over the rankings that contain it, with k = 60, the constant Cormack, Clarke and Buettcher fixed in a pilot and found near-optimal but not critical. Fusion uses ranks only, so BM25 scores and cosine similarities never have to share a scale. The order of the input rankings must not change the result, so the tie-break has to depend on the ids and their ranks, not on which list came first. What a list that repeats an id contributes is yours to decide; state it in a comment.
    starter: |
      def rrf(rankings, k=60, top_n=10):
          """Reciprocal rank fusion: each id scores the sum of 1 / (k + rank) over the rankings it appears in, rank from 1.

          rankings is a list of id lists, best first. Return up to top_n ids, best fused score first.
          """
          raise NotImplementedError
    tests: |
      bm25_ids = ["sec#0", "exp#1", "lv#0"]
      dense_ids = ["lv#0", "lv#1", "sec#0"]
      fused = rrf([bm25_ids, dense_ids])
      assert fused[:2] == ["sec#0", "lv#0"] or fused[:2] == ["lv#0", "sec#0"], "ids found by both retrievers come first"
      assert set(fused) == {"sec#0", "exp#1", "lv#0", "lv#1"}, "every id from either list is kept"
      assert rrf([["a", "b"], ["b", "c"]], k=60)[0] == "b", "rank 2 in both lists (2/62) beats rank 1 in one list (1/61)"
      assert rrf([["a", "b", "c", "d"]], top_n=2) == ["a", "b"], "top_n caps the result"

    solution: |
      def rrf(rankings, k=60, top_n=10):
          score, best = {}, {}
          for ranking in rankings:
              seen = set()
              for rank, cid in enumerate(ranking, start=1):
                  if cid in seen:
                      continue  # a list that repeats an id counts its first rank only
                  seen.add(cid)
                  score[cid] = score.get(cid, 0.0) + 1.0 / (k + rank)
                  best[cid] = min(best.get(cid, rank), rank)
          # Policy: ties go to the id with the better single rank, then to the id itself, so the order is stable.
          return sorted(score, key=lambda c: (-score[c], best[c], c))[:top_n]
  - title: Rerank, then cut to a budget
    brief: |
      ScriptedReranker stands in for a cross-encoder. Like a real one, it reads the query and the passage through one shared window, so a long passage is scored on its beginning. Write rerank, which scores each candidate once with the question as the query, sorts best first with stable ties, drops scores below min_score and cuts to top_n. Then write build_context, which picks the chunks for the prompt in reranked order within a word budget. Whether to stop at the first chunk that does not fit or to skip it and try the next is your policy; the tests accept both and check that the best chunk is never displaced and that the budget holds.
    starter: |
      class ScriptedReranker:
          """A deterministic stand-in for a cross-encoder.

          It reads the query and the passage through one shared window of `window` words, query first, so a passage longer
          than the room left is scored on its beginning only. The score is the share of distinct query words found in the
          visible part of the passage. Every call is logged in self.calls as (query, passage).
          """

          def __init__(self, window=64):
              self.window = window
              self.calls = []

          def score(self, query, passage):
              self.calls.append((query, passage))
              q = words(query)
              visible = words(passage)[: max(0, self.window - len(q))]
              return len(set(q) & set(visible)) / len(set(q)) if q else 0.0


      def passage(chunk):
          return chunk["heading"] + "\n" + chunk["text"]


      def rerank(query, ids, chunks_by_id, reranker, top_n=4, min_score=0.0):
          """Score each candidate's passage(chunk) once. Return up to top_n (id, score) pairs, best first, dropping scores
          below min_score; candidates with equal scores keep their incoming order."""
          raise NotImplementedError


      def build_context(reranked, chunks_by_id, budget_words):
          """Choose the chunk ids that go into the prompt from the reranked pairs, best first, with the total word count
          of their text at most budget_words."""
          raise NotImplementedError

      class ScriptedReranker:
          """A deterministic stand-in for a cross-encoder.

          It reads the query and the passage through one shared window of `window` words, query first, so a passage longer
          than the room left is scored on its beginning only. The score is the share of distinct query words found in the
          visible part of the passage. Every call is logged in self.calls as (query, passage).
          """

          def __init__(self, window=64):
              self.window = window
              self.calls = []

          def score(self, query, passage):
              self.calls.append((query, passage))
              q = words(query)
              visible = words(passage)[: max(0, self.window - len(q))]
              return len(set(q) & set(visible)) / len(set(q)) if q else 0.0


      def passage(chunk):
          return chunk["heading"] + "\n" + chunk["text"]


      def rerank(query, ids, chunks_by_id, reranker, top_n=4, min_score=0.0):
          scored = [(cid, reranker.score(query, passage(chunks_by_id[cid]))) for cid in ids]
          kept = [p for p in scored if p[1] >= min_score]
          kept.sort(key=lambda p: -p[1])  # sort is stable, so equal scores keep their incoming order
          return kept[:top_n]


      def build_context(reranked, chunks_by_id, budget_words):
          # Policy: skip a chunk that does not fit and keep trying smaller ones further down.
          chosen, used = [], 0
          for cid, _ in reranked:
              n = len(chunks_by_id[cid]["text"].split())
              if used + n <= budget_words:
                  chosen.append(cid)
                  used += n
          return chosen

This lab builds the pipeline the module debugs, one stage per step, in one file. Each step adds functions and runs its tests against everything you have written so far, so step 7 calls the chunker you wrote in step 1. The steps' tests feed each stage fixed inputs where they can, so a failing test in step 5 points at your reranking code and not at your BM25.

There is no model in the lab. Three stand-ins in the starters replace the parts that would need one: `embed()` maps words onto fixed concept vectors, `ScriptedReranker` scores the share of query words it can see through a window it shares with the query, and `ScriptedModel` answers with sentences from its context and cites them. All three are deterministic, and the tests assert behaviour (calls made, budgets respected, citations checked) rather than exact output where a policy is yours to choose.

Steps 1 to 4 build retrieval: structure-aware chunks, BM25, the dense stand-in and reciprocal rank fusion. Steps 5 and 6 are the stages between retrieval and the answer, where recall@k cannot see: reranking, the context budget and the citation check. Step 7 runs everything and breaks one stage at a time; the test passes only when your trace names the stage that was broken.

Your code is saved with your progress as you type. Passing a step's tests unlocks the next step and counts toward your streak. Python runs in your browser with the standard library only.
