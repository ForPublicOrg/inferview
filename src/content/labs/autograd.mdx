    solution: |
      class Value:
          def __init__(self, data, _parents=()):
              self.data = float(data)
              self.grad = 0.0
              self._parents = _parents
              self._backward = lambda: None

          def __add__(self, other):
              other = other if isinstance(other, Value) else Value(other)
              out = Value(self.data + other.data, (self, other))

              def _backward():
                  self.grad += out.grad
                  other.grad += out.grad

              out._backward = _backward
              return out

          def __mul__(self, other):
              other = other if isinstance(other, Value) else Value(other)
              out = Value(self.data * other.data, (self, other))

              def _backward():
                  self.grad += other.data * out.grad
                  other.grad += self.data * out.grad

              out._backward = _backward
              return out

          def relu(self):
              out = Value(self.data if self.data > 0 else 0.0, (self,))

              def _backward():
                  self.grad += (1.0 if self.data > 0 else 0.0) * out.grad

              out._backward = _backward
              return out

          def _topo(self):
              # Post-order with an explicit stack; a node is expanded once, on its first pop.
              order, seen, stack = [], set(), [(self, False)]
              while stack:
                  node, done = stack.pop()
                  if done:
                      order.append(node)
                  elif id(node) not in seen:
                      seen.add(id(node))
                      stack.append((node, True))
                      stack.extend((p, False) for p in node._parents if id(p) not in seen)
              return order

          def backward(self):
              order = self._topo()
              for node in order:
                  node.grad = 0.0
              self.grad = 1.0
              for node in reversed(order):
                  node._backward()

    solution: |
      def zeros(r, c):
          return [[0.0] * c for _ in range(r)]


      def _broadcast(sa, sb):
          out = []
          for x, y in zip(sa, sb):
              if x != y and 1 not in (x, y):
                  raise ValueError(f"shapes {sa} and {sb} do not broadcast")
              out.append(max(x, y))
          return tuple(out)


      def _at(m, i, j):
          # Read m as if broadcast: an axis of size 1 repeats its only entry.
          return m[i if len(m) > 1 else 0][j if len(m[0]) > 1 else 0]


      def _accumulate(grad, g, R, C):
          # Add an R x C upstream gradient into a possibly smaller grad, summing over the axes that were broadcast.
          for i in range(R):
              gi = grad[i if len(grad) > 1 else 0]
              for j in range(C):
                  gi[j if len(gi) > 1 else 0] += g[i][j]


      class Tensor:
          def __init__(self, data, _parents=()):
              self.data = [[float(v) for v in row] for row in data]
              self.grad = zeros(*self.shape)
              self._parents = _parents
              self._backward = lambda: None

          @property
          def shape(self):
              return (len(self.data), len(self.data[0]))

          def __add__(self, other):
              R, C = _broadcast(self.shape, other.shape)
              out = Tensor([[_at(self.data, i, j) + _at(other.data, i, j) for j in range(C)] for i in range(R)], (self, other))

              def _backward():
                  _accumulate(self.grad, out.grad, R, C)
                  _accumulate(other.grad, out.grad, R, C)

              out._backward = _backward
              return out

          def __mul__(self, other):
              R, C = _broadcast(self.shape, other.shape)
              a, b = self.data, other.data
              out = Tensor([[_at(a, i, j) * _at(b, i, j) for j in range(C)] for i in range(R)], (self, other))

              def _backward():
                  g = out.grad
                  _accumulate(self.grad, [[g[i][j] * _at(b, i, j) for j in range(C)] for i in range(R)], R, C)
                  _accumulate(other.grad, [[g[i][j] * _at(a, i, j) for j in range(C)] for i in range(R)], R, C)

              out._backward = _backward
              return out

          def __matmul__(self, other):
              (n, m), (m2, p) = self.shape, other.shape
              if m != m2:
                  raise ValueError(f"cannot multiply {self.shape} by {other.shape}")
              a, b = self.data, other.data
              bt = list(zip(*b))
              out = Tensor([[sum(x * y for x, y in zip(row, col)) for col in bt] for row in a], (self, other))

              def _backward():
                  g = out.grad
                  for i in range(n):
                      for k in range(m):
                          self.grad[i][k] += sum(g[i][j] * b[k][j] for j in range(p))       # g @ B^T
                  for k in range(m):
                      for j in range(p):
                          other.grad[k][j] += sum(a[i][k] * g[i][j] for i in range(n))      # A^T @ g

              out._backward = _backward
              return out

          def relu(self):
              out = Tensor([[v if v > 0 else 0.0 for v in row] for row in self.data], (self,))

              def _backward():
                  for i, row in enumerate(self.data):
                      for j, v in enumerate(row):
                          if v > 0:
                              self.grad[i][j] += out.grad[i][j]

              out._backward = _backward
              return out

          def sum(self):
              out = Tensor([[sum(sum(row) for row in self.data)]], (self,))

              def _backward():
                  for row in self.grad:
                      for j in range(len(row)):
                          row[j] += out.grad[0][0]

              out._backward = _backward
              return out

          def _topo(self):
              order, seen, stack = [], set(), [(self, False)]
              while stack:
                  node, done = stack.pop()
                  if done:
                      order.append(node)
                  elif id(node) not in seen:
                      seen.add(id(node))
                      stack.append((node, True))
                      stack.extend((p, False) for p in node._parents if id(p) not in seen)
              return order

          def backward(self):
              if self.shape != (1, 1):
                  raise ValueError("backward() starts from a 1 x 1 tensor")
              order = self._topo()
              for node in order:
                  node.grad = zeros(*node.shape)
              self.grad = [[1.0]]
              for node in reversed(order):
                  node._backward()

    solution: |
      import math


      def cross_entropy(logits, labels):
          N = len(logits.data)
          probs, total = [], 0.0
          for row, y in zip(logits.data, labels):
              m = max(row)
              lse = m + math.log(sum(math.exp(v - m) for v in row))     # log-sum-exp, shifted by the row max
              total += lse - row[y]
              probs.append([math.exp(v - lse) for v in row])
          out = Tensor([[total / N]], (logits,))

          def _backward():
              g = out.grad[0][0] / N
              for i, (p, y) in enumerate(zip(probs, labels)):
                  for j, pj in enumerate(p):
                      logits.grad[i][j] += g * (pj - (1.0 if j == y else 0.0))

          out._backward = _backward
          return out

      def grad_check(build, tensors, eps=1e-6):
          build().backward()
          analytic = [[list(row) for row in t.grad] for t in tensors]
          worst = 0.0
          for t, ga in zip(tensors, analytic):
              for i, row in enumerate(t.data):
                  for j, old in enumerate(row):
                      row[j] = old + eps
                      up = build().data[0][0]
                      row[j] = old - eps
                      dn = build().data[0][0]
                      row[j] = old
                      num = (up - dn) / (2 * eps)
                      # Relative to the larger magnitude, floored so two near-zero gradients do not divide by zero.
                      worst = max(worst, abs(num - ga[i][j]) / max(abs(num), abs(ga[i][j]), 1e-8))
          return worst
  - title: Prefix-product backward without division
    brief: |
      A cumprod op: a running product along each row. Its backward pass may not divide by the inputs, because they can be zero. With P[k-1] the product before position k and S[k] = g[k] + a[k+1] S[k+1] built from the right, the gradient at k is P[k-1] times S[k]: two passes, O(n), no special case for zeros.     solution: |
      def cumprod(t):
          rows = []
          for row in t.data:
              p, out_row = 1.0, []
              for v in row:
                  p *= v
                  out_row.append(p)
              rows.append(out_row)
          out = Tensor(rows, (t,))

          def _backward():
              for a, g, grad in zip(t.data, out.grad, t.grad):
                  n = len(a)
                  # s[k] = sum over i >= k of g[i] * a[k+1]...a[i], built right to left; then multiply by a[0]...a[k-1].
                  s = [0.0] * n
                  acc = 0.0
                  for k in range(n - 1, -1, -1):
                      acc = g[k] + (a[k + 1] * acc if k + 1 < n else 0.0)
                      s[k] = acc
                  prefix = 1.0
                  for k in range(n):
                      grad[k] += prefix * s[k]
                      prefix *= a[k]

          out._backward = _backward
          return out


      Tensor.cumprod = cumprod
