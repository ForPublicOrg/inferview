      import math


      def flip_gate(baseline, candidate, alpha=0.05, must_pass=()):
          if set(baseline) != set(candidate):
              raise ValueError("baseline and candidate must score the same items")
          regressed = sorted(k for k in baseline if baseline[k] and not candidate[k])
          improved = sorted(k for k in baseline if candidate[k] and not baseline[k])
          b, c = len(regressed), len(improved)
          p_value = _sign_test(b, c)
          blocked_items = sorted(set(regressed) & set(must_pass))
          if blocked_items or (b > c and p_value < alpha):
              decision = "block"
          else:
              decision = "ship"
          return {"regressed": regressed, "improved": improved, "p_value": p_value,
                  "decision": decision, "must_pass_regressed": blocked_items}


      def _sign_test(b, c):
          n, k = b + c, min(b, c)
          if n == 0:
              return 1.0
          # log of C(n, k) / 2^n, then walk down to 0 with the term ratio, which never builds a big integer
          log_term = math.lgamma(n + 1) - math.lgamma(k + 1) - math.lgamma(n - k + 1) - n * math.log(2)
          term = math.exp(log_term)
          total = term
          for j in range(k, 0, -1):
              term *= j / (n - j + 1)
              total += term
              if term < total * 1e-17:
                  break
          return min(1.0, 2 * total)

      import math
      from statistics import NormalDist


      def sample_size(p0, mde, alpha=0.05, power=0.8, share=0.5, deff=1.0):
          p1 = p0 + mde
          if not (0 < p0 < 1 and 0 < p1 < 1) or mde == 0:
              raise ValueError("p0 and p0 + mde must lie strictly between 0 and 1, and mde must be non-zero")
          if not 0 < share < 1 or deff < 1:
              raise ValueError("share must lie in (0, 1) and deff must be at least 1")
          z = NormalDist().inv_cdf(1 - alpha / 2) + NormalDist().inv_cdf(power)
          total = z * z * (p0 * (1 - p0) / (1 - share) + p1 * (1 - p1) / share) / mde ** 2 * deff
          control = math.ceil(total * (1 - share))
          treatment = math.ceil(total * share)
          return {"control": control, "treatment": treatment, "total": control + treatment}


      def duration_days(total, daily, whole_weeks=True):
          week = [daily] * 7 if isinstance(daily, (int, float)) else list(daily)
          if len(week) != 7 or sum(week) <= 0 or any(v < 0 for v in week):
              raise ValueError("daily is one positive volume or seven non-negative weekday volumes")
          full_weeks, rest = divmod(total, sum(week))
          if rest == 0 and full_weeks > 0:
              full_weeks, rest = full_weeks - 1, sum(week)
          days = int(full_weeks) * 7
          seen = 0
          for v in week:
              if seen >= rest:
                  break
              seen += v
              days += 1
          if whole_weeks:
              days = -(-days // 7) * 7
          return days

      import math


      def srm_check(counts, shares, threshold):
          if len(counts) != len(shares) or len(counts) < 2:
              raise ValueError("one count and one planned share per arm, at least two arms")
          if abs(sum(shares) - 1) > 1e-9 or any(s <= 0 for s in shares):
              raise ValueError("planned shares must be positive and sum to 1")
          n = sum(counts)
          if n <= 0 or any(c < 0 for c in counts):
              raise ValueError("counts must be non-negative with a positive total")
          chi2 = sum((c - n * s) ** 2 / (n * s) for c, s in zip(counts, shares))
          p_value = _upper_gamma_q((len(counts) - 1) / 2, chi2 / 2)
          return {"chi2": chi2, "p_value": p_value, "mismatch": p_value < threshold}


      def _upper_gamma_q(a, x):
          """Regularised upper incomplete gamma Q(a, x): series below a + 1, continued fraction above."""
          if x <= 0:
              return 1.0
          log_front = -x + a * math.log(x) - math.lgamma(a)
          if x < a + 1:
              term = total = 1 / a
              ap = a
              for _ in range(10000):
                  ap += 1
                  term *= x / ap
                  total += term
                  if abs(term) < abs(total) * 1e-15:
                      break
              return max(0.0, 1 - total * math.exp(log_front))
          tiny = 1e-300
          b = x + 1 - a
          c = 1 / tiny
          d = 1 / b
          h = d
          for i in range(1, 10000):
              an = -i * (i - a)
              b += 2
              d = an * d + b
              d = tiny if abs(d) < tiny else d
              c = b + an / c
              c = tiny if abs(c) < tiny else c
              d = 1 / d
              delta = d * c
              h *= delta
              if abs(delta - 1) < 1e-15:
                  break
          return math.exp(log_front) * h
  - title: Clustered standard errors for multi-turn sessions
    brief: |
      When users are assigned and conversations are measured, conversations from one user are correlated and the naive standard error is too small. Miller's Eq. 4 adds the within-cluster cross terms of the residuals to the naive variance; the two together equal the squared residual total of each cluster, summed over clusters and divided by n squared, which is one pass over the rows.       import math
      from collections import defaultdict
      from statistics import NormalDist


      def clustered_se(rows):
          """rows: iterable of (cluster_id, value). Miller (2024) Eq. 4: the naive variance plus the within-cluster cross terms,
          which together equal the sum over clusters of the squared summed residuals, divided by n squared."""
          rows = list(rows)
          n = len(rows)
          if n < 2:
              raise ValueError("need at least two rows")
          mean = sum(v for _, v in rows) / n
          sums = defaultdict(float)
          squares = 0.0
          for cluster, v in rows:
              e = v - mean
              sums[cluster] += e
              squares += e * e
          se_naive = math.sqrt(squares / n / n)
          se_clustered = math.sqrt(sum(s * s for s in sums.values()) / n / n)
          return {"n": n, "clusters": len(sums), "mean": mean, "se_naive": se_naive, "se_clustered": se_clustered}


      def diff_clustered(control_rows, treatment_rows, level=0.95):
          c = clustered_se(control_rows)
          t = clustered_se(treatment_rows)
          diff = t["mean"] - c["mean"]
          se = math.sqrt(c["se_clustered"] ** 2 + t["se_clustered"] ** 2)
          z = NormalDist().inv_cdf(0.5 + level / 2)
          return {"diff": diff, "se": se, "low": diff - z * se, "high": diff + z * se}

      def decide(primary, guardrails, srm, mde):
          """primary: {"diff", "low", "high"}, higher is better. guardrails: [{"name", "diff", "low", "high", "max_harm",
          "higher_is_better"}]. srm: the srm_check result. mde: the smallest lift that pays for the change."""
          if srm["mismatch"]:
              return {"decision": "invalid", "reasons": [f"sample ratio mismatch (p = {srm['p_value']:.2g}); outcomes not read"],
                      "primary": primary, "guardrails": guardrails, "mde": mde}
          reasons, harmed, unproven = [], [], []
          for g in guardrails:
              sign = 1 if g["higher_is_better"] else -1
              worst = g["low"] if g["higher_is_better"] else g["high"]
              if sign * g["diff"] < -g["max_harm"]:
                  harmed.append(g["name"])
                  reasons.append(f"guardrail {g['name']} breached: estimate {g['diff']:+.4g} beyond allowed harm {g['max_harm']:.4g}")
              elif sign * worst < -g["max_harm"]:
                  unproven.append(g["name"])
                  reasons.append(f"guardrail {g['name']} not cleared: interval bound {worst:+.4g} beyond allowed harm {g['max_harm']:.4g}")
          if harmed:
              decision = "no-ship"
          elif primary["high"] < 0:
              decision = "no-ship"
              reasons.append(f"primary metric worse: interval {primary['low']:+.4g} to {primary['high']:+.4g}")
          elif primary["high"] < mde:
              decision = "no-ship"
              reasons.append(f"primary interval {primary['low']:+.4g} to {primary['high']:+.4g} excludes the minimum effect {mde:+.4g}")
          elif primary["low"] <= 0 or unproven:
              decision = "inconclusive"
              if primary["low"] <= 0:
                  reasons.append(f"primary interval {primary['low']:+.4g} to {primary['high']:+.4g} includes zero")
          else:
              decision = "ship"
              reasons.append(f"primary lift {primary['diff']:+.4g} (interval {primary['low']:+.4g} to {primary['high']:+.4g}); every guardrail cleared")
          return {"decision": decision, "reasons": reasons, "primary": primary, "guardrails": guardrails, "mde": mde}


      def memo(result):
          lines = [f"Decision: {result['decision']}", f"Minimum effect that pays for the change: {result['mde']:+.4g}"]
          p = result["primary"]
          lines.append(f"Primary: {p['diff']:+.4g} (interval {p['low']:+.4g} to {p['high']:+.4g})")

          lines += [f"Reason: {r}" for r in result["reasons"]]
